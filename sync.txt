/**
 * Installs background triggers (onOpen + 5-minute two-way auto-sync) once authorized.
 */
function ensureBackgroundTriggers_() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const props = PropertiesService.getScriptProperties();
    const triggers = ScriptApp.getProjectTriggers();
    const revOk = props.getProperty(TRIGGER_REV_KEY) === TRIGGER_REV;
    let hasOpenTrigger = false;
    let hasAutoSyncTrigger = false;

    for (let i = 0; i < triggers.length; i++) {
      const fn = triggers[i].getHandlerFunction();
      if (fn === 'setupCenterSyncMenu' || fn === 'taoMenuDongBo') hasOpenTrigger = true;
      if (fn === 'autoSyncTodayAttendanceAfterClass' || fn === 'tuDongDongBoSauGioHoc') {
        if (revOk) hasAutoSyncTrigger = true;
        else ScriptApp.deleteTrigger(triggers[i]);
      }
    }

    if (!hasOpenTrigger && ss) {
      ScriptApp.newTrigger('setupCenterSyncMenu').forSpreadsheet(ss).onOpen().create();
    }
    if (!hasAutoSyncTrigger) {
      ScriptApp.newTrigger('autoSyncTodayAttendanceAfterClass').timeBased().everyMinutes(5).create();
      props.setProperty(TRIGGER_REV_KEY, TRIGGER_REV);
    }
  } catch (e) {}
}

function setupCenterSyncMenu() {
  buildCenterSyncMenuOnly_();
  ensureBackgroundTriggers_();
}

// 3. BUTTON 1: INTERACTIVE DIALOG TO SYNC STUDENT ROSTERS (GOOGLE SHEET / .XLSX)
// ============================================================================
function syncStudentRosters() {
  ensureBackgroundTriggers_();

  const masterSs = SpreadsheetApp.getActiveSpreadsheet();
  const configSheet = masterSs.getSheetByName('Classes_Config');
  if (!configSheet) {
    showAlert_('⚠️ Sheet \'Classes_Config\' was not found. Please initialize or add a class first.');
    return;
  }

  const cfgData = configSheet.getDataRange().getValues();
  const classItems = [];
  const props = PropertiesService.getScriptProperties();
  const syncMetaMap = JSON.parse(props.getProperty(ROSTER_SYNC_META_PROP_KEY) || '{}');
  let metaDirty = false;
  let scriptCache = null;
  try {
    scriptCache = CacheService.getScriptCache();
  } catch (e) {}

  for (let i = 1; i < cfgData.length; i++) {
    const classId = String(cfgData[i][0] || '').trim();
    const className = String(cfgData[i][1] || '').trim();
    if (!classId) continue;

    // Count existing students and collect existing IDs in the Master QR class tab
    const classSheet = masterSs.getSheetByName(classId);
    const existingInfo = collectMasterClassStudentIds_(classSheet);
    const existingStudentCount = existingInfo.count;
    const existingIdsMap = existingInfo.upperMap;

    const alreadySynced = existingStudentCount > 0;
    const targetInfo = resolveClassTargetFile_(masterSs, classId, className);

    let previewCount = existingStudentCount;
    let newStudentsOnDrive = 0;
    let detectedTabName = '';
    let scanError = '';

    const meta = syncMetaMap[classId] || null;
    const fileChangedSinceLastSync = Boolean(
      targetInfo && (
        !meta ||
        meta.fileId !== targetInfo.fileId ||
        meta.syncedCount !== existingStudentCount ||
        (targetInfo.lastUpdatedMs && meta.lastUpdatedMs !== targetInfo.lastUpdatedMs)
      )
    );

    // Pre-scan the target Drive file for new classes OR when the Drive file was modified since last sync
    if (targetInfo && (!alreadySynced || fileChangedSinceLastSync)) {
      try {
        const extractResult = extractRosterFromTargetFile_(targetInfo, classId, className);
        detectedTabName = extractResult.sheetName || '';
        const rosterRows = extractResult.roster || [];

        if (!alreadySynced) {
          previewCount = rosterRows.length;
        } else {
          for (let r = 0; r < rosterRows.length; r++) {
            const sidKey = String(rosterRows[r][1] || '').trim().toUpperCase();
            if (sidKey && !existingIdsMap[sidKey]) {
              newStudentsOnDrive++;
            }
          }
          if (newStudentsOnDrive === 0 && rosterRows.length > 0) {
            syncMetaMap[classId] = {
              fileId: targetInfo.fileId,
              lastUpdatedMs: targetInfo.lastUpdatedMs || 0,
              syncedCount: existingStudentCount
            };
            metaDirty = true;
          }
        }

        if (scriptCache && rosterRows.length > 0) {
          try {
            scriptCache.put(
              ROSTER_PREVIEW_CACHE_PREFIX + classId,
              JSON.stringify({
                fileId: targetInfo.fileId,
                lastUpdatedMs: targetInfo.lastUpdatedMs || 0,
                sheetName: detectedTabName,
                roster: rosterRows
              }),
              600
            );
          } catch (cacheErr) {}
        }
      } catch (err) {
        scanError = err.message;
      }
    }

    classItems.push({
      classId: classId,
      className: className || classId,
      students: previewCount,
      newStudentsOnDrive: newStudentsOnDrive,
      alreadySynced: alreadySynced,
      fileFound: Boolean(targetInfo),
      fileId: targetInfo ? targetInfo.fileId : '',
      fileName: targetInfo ? targetInfo.fileName : '',
      folderPath: targetInfo ? (targetInfo.folderPath || 'My Drive') : '',
      fileUrl: targetInfo ? (targetInfo.fileUrl || '') : '',
      sheetName: detectedTabName,
      scanError: scanError,
      fileTypeLabel: targetInfo ? (targetInfo.isGoogleSheet ? '📊 Google Sheet' : '📗 Excel .XLSX') : '⚠️ Not found on Drive'
    });
  }

  if (metaDirty) {
    try {
      props.setProperty(ROSTER_SYNC_META_PROP_KEY, JSON.stringify(syncMetaMap));
    } catch (e) {}
  }

  if (classItems.length === 0) {
    showAlert_(
      'ℹ️ No classes found in \'Classes_Config\'.\n\n' +
      'Please use \'Attendance Admin -> 2. Add New Class Wizard\' to create a class first.'
    );
    return;
  }

  const rowsTpl = classItems.map(function(c) {
    const isSynced = c.alreadySynced;
    const hasDelta = c.newStudentsOnDrive > 0;
    const shouldPreCheck = (!isSynced && c.fileFound && c.students > 0) || hasDelta;
    const checkedAttr = shouldPreCheck ? 'checked' : '';

    let statusBadge = '';
    if (isSynced && hasDelta) {
      statusBadge =
        '[[span style="background:#dbeafe; color:#1e40af; padding:3px 9px; border-radius:999px; font-size:11.5px; font-weight:700; white-space:nowrap;"]]' +
          '🔄 +' + c.newStudentsOnDrive + ' new (' + c.students + ' → ' + (c.students + c.newStudentsOnDrive) + ')' +
        '[[/span]]';
    } else if (isSynced) {
      statusBadge =
        '[[span style="background:#dcfce7; color:#166534; padding:3px 9px; border-radius:999px; font-size:11.5px; font-weight:700; white-space:nowrap;"]]' +
          '✅ Synced (' + c.students + ' students)' +
        '[[/span]]';
    } else if (!c.fileFound) {
      statusBadge =
        '[[span style="background:#fee2e2; color:#991b1b; padding:3px 9px; border-radius:999px; font-size:11.5px; font-weight:700; white-space:nowrap;"]]' +
          '⚠️ File Not Found' +
        '[[/span]]';
    } else if (c.students > 0) {
      statusBadge =
        '[[span style="background:#dbeafe; color:#1e40af; padding:3px 9px; border-radius:999px; font-size:11.5px; font-weight:700; white-space:nowrap;"]]' +
          '🆕 New Class (' + c.students + ' students)' +
        '[[/span]]';
    } else {
      statusBadge =
        '[[span style="background:#fef3c7; color:#92400e; padding:3px 9px; border-radius:999px; font-size:11.5px; font-weight:700; white-space:nowrap;"]]' +
          '⚠️ 0 students found' +
        '[[/span]]';
    }

    const fileBadgeColor = c.fileFound ? '#374151' : '#dc2626';
    let fileDesc = '';
    let locationDesc = '';
    if (!c.fileFound) {
      fileDesc = '⚠️ No matching Google Sheet or .XLSX file with student list (MSSV + Họ tên) found on Drive';
    } else {
      const tabPart = c.sheetName ? (' • Tab: [[b]]' + escapeHtml_(c.sheetName) + '[[/b]]') : '';
      const errPart = c.scanError ? (' [[span style="color:#dc2626;"]](' + escapeHtml_(c.scanError) + ')[[/span]]') : '';
      fileDesc = escapeHtml_(c.fileTypeLabel) + ': [[b]]' + escapeHtml_(c.fileName) + '[[/b]]' + tabPart + errPart;

      const openLinkHtml = c.fileUrl
        ? (' [[a href="' + escapeHtml_(c.fileUrl) + '" target="_blank" onclick="event.stopPropagation();" ' +
           'style="color:#2563eb; text-decoration:none; font-weight:600; margin-left:6px; white-space:nowrap;"]]↗ Open File[[/a]]')
        : '';
      locationDesc =
        '[[div style="font-size:11.5px; color:#6b7280; margin-top:3px; display:flex; align-items:center; justify-content:space-between; gap:6px; flex-wrap:wrap;"]]' +
          '[[span style="word-break:break-all;"]]📁 Location: [[b style="color:#4b5563; font-weight:600;"]]' + escapeHtml_(c.folderPath) + '[[/b]][[/span]]' +
          openLinkHtml +
        '[[/div]]';
    }

    const cardBg = hasDelta ? '#eff6ff' : (isSynced ? '#f0fdf4' : (c.fileFound && c.students > 0 ? '#eff6ff' : '#fef2f2'));
    const cardBorder = hasDelta ? '#93c5fd' : (isSynced ? '#bbf7d0' : (c.fileFound && c.students > 0 ? '#bfdbfe' : '#fecaca'));

    return (
      '[[label style="display:flex; align-items:flex-start; padding:10px 12px; margin-bottom:8px; background:' + cardBg + '; border:1px solid ' + cardBorder + '; border-radius:8px; cursor:pointer;"]]' +
        '[[input type="checkbox" name="rosterClassItem" value="' + escapeHtml_(c.classId) + '" ' +
          'data-synced="' + (isSynced ? '1' : '0') + '" ' +
          'data-delta="' + c.newStudentsOnDrive + '" ' +
          'data-found="' + (c.fileFound ? '1' : '0') + '" ' +
          'data-count="' + c.students + '" ' +
          checkedAttr + ' ' +
          'onchange="updateSelectionHint()" ' +
          'style="width:18px; height:18px; margin-right:12px; margin-top:2px; cursor:pointer; flex-shrink:0;"]]' +
        '[[div style="flex:1; min-width:0;"]]' +
          '[[div style="display:flex; align-items:center; justify-content:space-between; gap:8px;"]]' +
            '[[div style="font-weight:700; color:#111827; font-size:13.5px; overflow:hidden; text-overflow:ellipsis; white-space:nowrap;"]]' +
              escapeHtml_(c.classId) +
              ' [[span style="font-weight:normal; color:#6b7280; font-size:12px;"]](' + escapeHtml_(c.className) + ')[[/span]]' +
            '[[/div]]' +
            statusBadge +
          '[[/div]]' +
          '[[div style="font-size:12px; color:' + fileBadgeColor + '; margin-top:5px; word-break:break-word;"]]' +
            fileDesc +
          '[[/div]]' +
          locationDesc +
        '[[/div]]' +
      '[[/label]]'
    );
  }).join('');

  const dialogTpl =
    '[[div style="font-family: Arial, sans-serif; padding: 12px; color: #1f2937;"]]' +
      '[[p style="margin-top:0; margin-bottom:10px; font-size:13px; color:#374151; line-height:1.45;"]]' +
        'New classes and classes with [[b]]newly added students on Drive[[/b]] are auto-selected. ' +
        'Tick any [[b]]✅ Synced[[/b]] class to force a fresh re-scan from Drive:' +
      '[[/p]]' +
      '[[div style="display:flex; justify-content:space-between; align-items:center; gap:8px; margin-bottom:10px; padding:6px 10px; background:#f9fafb; border:1px solid #e5e7eb; border-radius:6px; font-size:12px;"]]' +
        '[[button type="button" onclick="tickAllNew()" style="padding:5px 10px; background:#e0e7ff; color:#1e40af; border:1px solid #c7d2fe; border-radius:5px; cursor:pointer; font-weight:600; white-space:nowrap;"]]☑️ Select New / Updated[[/button]]' +
        '[[button type="button" onclick="tickAllClasses()" style="padding:5px 10px; background:#f3f4f6; color:#374151; border:1px solid #d1d5db; border-radius:5px; cursor:pointer; font-weight:600; white-space:nowrap;"]]🔄 Select All Classes (Force Re-scan)[[/button]]' +
      '[[/div]]' +
      '[[div style="max-height:245px; overflow-y:auto; margin-bottom:12px; padding-right:2px;"]]' +
        rowsTpl +
      '[[/div]]' +
      '[[div id="statusMsg" style="display:block; padding:9px 10px; border-radius:6px; font-size:12.5px; margin-bottom:12px; background:#f3f4f6; color:#374151;"]][[/div]]' +
      '[[div style="display:flex; justify-content:flex-end; gap:8px;"]]' +
        '[[button onclick="google.script.host.close()" style="padding:8px 14px; background:#e5e7eb; color:#374151; border:none; border-radius:6px; cursor:pointer; font-weight:600;"]]Cancel[[/button]]' +

/**
 * Sync a single class roster from its target Google Sheet / .XLSX file on Drive into the Master Sheet.
 * Used by both the Sheet Menu (`executeSelectedRostersSync`) and the Lecturer Web App (`TEACHER_REBUILD_REPORT`).
 */
function syncSingleClassRosterFromDrive_(masterSs, classId, className, options) {
  const opts = options || {};
  const targetInfo = resolveClassTargetFile_(masterSs, classId, className, Boolean(opts.forceFreshFileLookup));
  if (!targetInfo) {
    return {
      ok: false,
      summary: '⚠️ ' + classId + ': No matching Google Sheet or .XLSX file with student list found on Google Drive.',
      newStudentsAdded: 0,
      newStudentNames: []
    };
  }

  let classSheet = masterSs.getSheetByName(classId);
  if (!classSheet) {
    classSheet = masterSs.insertSheet(classId);
    classSheet.getRange('B:B').setNumberFormat('@');
    classSheet.getRange(1, 1, 1, 5).setValues([['No.', 'Student_ID', 'Full_Name', 'Total_Present', 'Total_Late']]);
  }

  const existingInfo = collectMasterClassStudentIds_(classSheet);
  const existingIds = existingInfo.upperMap;
  const existingStudentCount = existingInfo.count;

  const props = PropertiesService.getScriptProperties();
  const syncMetaMap = JSON.parse(props.getProperty(ROSTER_SYNC_META_PROP_KEY) || '{}');
  const meta = syncMetaMap[classId] || null;

  // Zero-lag fast path for Web App button when the Drive file has not been modified since last sync
  if (
    opts.onlyIfModified &&
    existingStudentCount > 0 &&
    meta &&
    meta.fileId === targetInfo.fileId &&
    meta.syncedCount === existingStudentCount &&
    targetInfo.lastUpdatedMs &&
    meta.lastUpdatedMs === targetInfo.lastUpdatedMs
  ) {
    return {
      ok: true,
      skippedUnchanged: true,
      summary: 'ℹ️ ' + classId + ': Roster unchanged on Drive (' + existingStudentCount + ' students).',
      newStudentsAdded: 0,
      newStudentNames: []
    };
  }

  let extractResult = opts.extractResult || null;
  if (!extractResult && opts.scriptCache && !opts.bypassPreviewCache) {
    try {
      const cachedRaw = opts.scriptCache.get(ROSTER_PREVIEW_CACHE_PREFIX + classId);
      if (cachedRaw) {
        const parsedCache = JSON.parse(cachedRaw);
        if (
          parsedCache &&
          parsedCache.fileId === targetInfo.fileId &&
          (!targetInfo.lastUpdatedMs || parsedCache.lastUpdatedMs === targetInfo.lastUpdatedMs) &&
          Array.isArray(parsedCache.roster)
        ) {
          extractResult = parsedCache;
        }
      }
    } catch (cacheReadErr) {}
  }

  if (!extractResult) {
    extractResult = extractRosterFromTargetFile_(targetInfo, classId, className);
  }

  const rosterRows = extractResult.roster || [];
  const sourceDesc = (targetInfo.isGoogleSheet ? 'Google Sheet' : '.XLSX') + ' [' + extractResult.sheetName + '] (📁 ' + (targetInfo.folderPath || 'Drive') + ')';

  if (rosterRows.length === 0) {
    return {
      ok: false,
      summary: '⚠️ ' + classId + ': No valid student rows found in ' + sourceDesc + '.',
      newStudentsAdded: 0,
      newStudentNames: []
    };
  }

  let newStudentsAdded = 0;
  const newStudentNames = [];
  let summaryLine = '';

  const rosterLockRes = withMasterWriteLock_(opts, opts.lockWaitMs || 15000, function() {
    if (existingStudentCount === 0) {
      classSheet.getRange(1, 1, 1, 5).setValues([['No.', 'Student_ID', 'Full_Name', 'Total_Present', 'Total_Late']]);
      classSheet.getRange(2, 1, rosterRows.length, 5).setValues(rosterRows);
      SpreadsheetApp.flush();
      newStudentsAdded = rosterRows.length;
      summaryLine = '✅ ' + classId + ': Imported full roster (' + rosterRows.length + ' students) from ' + sourceDesc + '.';
      appendAuditLog_(masterSs, opts.actor || 'SYNC_MENU', 'ROSTER_IMPORT', classId, 'Imported ' + rosterRows.length + ' students from ' + sourceDesc);
    } else {
      const newRows = [];
      for (let r = 0; r < rosterRows.length; r++) {
        const sid = String(rosterRows[r][1]).trim();
        const sidKey = sid.toUpperCase();
        if (!existingIds[sidKey]) {
          const nextNo = existingStudentCount + newRows.length + 1;
          const fullName = String(rosterRows[r][2] || '').trim();
          newRows.push([nextNo, sid, fullName, 0, 0]);
          newStudentNames.push(sid + ' - ' + fullName);
          existingIds[sidKey] = true;
        }
      }

      if (newRows.length > 0) {
        const startRow = classSheet.getLastRow() + 1;
        classSheet.getRange(startRow, 1, newRows.length, 5).setValues(newRows);
        SpreadsheetApp.flush();
        newStudentsAdded = newRows.length;
        summaryLine = '✅ ' + classId + ': Appended ' + newRows.length + ' new student(s) (' + newStudentNames.join(', ') + ' • Total: ' + (existingStudentCount + newRows.length) + ') from ' + sourceDesc + '.';
        appendAuditLog_(masterSs, opts.actor || 'SYNC_MENU', 'ROSTER_APPEND', classId, 'Appended ' + newRows.length + ' new students (' + newStudentNames.join(', ') + ') from ' + sourceDesc);
      } else {
        summaryLine = 'ℹ️ ' + classId + ': Roster is already up to date (' + existingStudentCount + ' students).';
      }
    }
    return { busy: false };
  });

  if (rosterLockRes && rosterLockRes.busy) {
    return {
      ok: false,
      busy: true,
      summary: '⏳ ' + classId + ': Master sheet is busy, please retry in a minute.',
      newStudentsAdded: 0,
      newStudentNames: []
    };
  }

  // Save updated sync metadata so unchanged files are skipped in 0ms next time
  const finalStudentCount = existingStudentCount === 0 ? rosterRows.length : (existingStudentCount + newStudentsAdded);
  syncMetaMap[classId] = {
    fileId: targetInfo.fileId,
    lastUpdatedMs: targetInfo.lastUpdatedMs || 0,
    syncedCount: finalStudentCount
  };
  try {
    props.setProperty(ROSTER_SYNC_META_PROP_KEY, JSON.stringify(syncMetaMap));
  } catch (e) {}

  try {
    if (Array.isArray(extractResult.dateColumns) && extractResult.dateColumns.length > 0) {
      importPastDaysFromSheet_(masterSs, classId, targetInfo, extractResult.dateColumns, opts);
    }
  } catch (importErr) {}

  // Invalidate Master Sheet roster/history cache & rebuild attendance columns whenever Drive file is synced
  try {
    const cache = CacheService.getScriptCache();
    cache.removeAll([
      'CLASS_ATT_STATS_V36_' + classId,
      ROSTER_PREVIEW_CACHE_PREFIX + classId
    ]);
    if (typeof invalidateClassCache === 'function') invalidateClassCache(classId);
    if (typeof rebuildSingleClassGrid === 'function') {
      rebuildSingleClassGrid(masterSs, classId);
    }
  } catch (e) {}

  return {
    ok: true,
    summary: summaryLine,
    newStudentsAdded: newStudentsAdded,
    newStudentNames: newStudentNames,
    totalStudents: finalStudentCount
  };
}

// 4. BUTTON 2: SYNC TODAY'S ATTENDANCE (TO GOOGLE SHEET OR .XLSX)
//    Same two-way merge as the 5-minute trigger, forced (no time gates), resumes paused days.
// ============================================================================
function syncTodayAttendance() {
  ensureBackgroundTriggers_();

  const masterSs = SpreadsheetApp.getActiveSpreadsheet();
  const ui = SpreadsheetApp.getUi();
  const now = new Date();
  const currentDayStr = now.getFullYear() + '-' + (now.getMonth() + 1) + '-' + now.getDate();
  const props = PropertiesService.getScriptProperties();

  const mig = ensureSyncBaseMigrated_(masterSs, props);
  if (mig && mig.busy) {
    showAlert_('⏳ Master sheet is busy with student check-ins. Nothing was synced, please retry in a minute.');
    return;
  }
  const env = makeSyncEnv_(masterSs, now);

  const classNamesMap = getClassNamesMap_(masterSs);
  const groupedByClassDate = getGroupedSessionsByClassDate_(masterSs);
  const sessionsToSync = [];
  const unsyncedPastGroups = [];

  for (let i = 0; i < groupedByClassDate.length; i++) {
    const grp = groupedByClassDate[i];
    const be = env.baseMap[syncBaseKey_(grp.classId, grp.dayKey)] || null;
    if (grp.dateKey === currentDayStr) {
      sessionsToSync.push(grp);
    } else if (!be) {
      // Past date that was NEVER synced even once (e.g. crossed midnight before syncing)
      unsyncedPastGroups.push(grp);
    } else if (be.paused === SYNC_PAUSE_UNDO || be.paused === SYNC_PAUSE_STYLE) {
      // Paused by Undo / mark-style change: pressing Sync resumes it
      sessionsToSync.push(grp);
    }
  }

  // Rescue prompt if there are past sessions that were never synced at all
  if (unsyncedPastGroups.length > 0) {
    const pastDesc = unsyncedPastGroups.map(function(g) {
      return g.classId + ' (' + g.day + '/' + g.month + ')';
    }).join(', ');
    const rescueConfirm = ui.alert(
      'Unsynced Past Session(s) Detected',
      'Found session(s) from a previous date that have NEVER been synced to the target Google Sheet / .XLSX file:\n' +
      '• ' + pastDesc + '\n\n' +
      'Do you want to sync them now? (Already-synced past dates are only updated for cells that changed.)',
      ui.ButtonSet.YES_NO
    );
    if (rescueConfirm === ui.Button.YES) {
      for (let i = 0; i < unsyncedPastGroups.length; i++) {
        sessionsToSync.push(unsyncedPastGroups[i]);
      }
    }
  }

  if (sessionsToSync.length === 0) {
    showAlert_(
      'ℹ️ No QR attendance sessions found for TODAY (' +
      now.getDate() + '/' + (now.getMonth() + 1) + '/' + now.getFullYear() + ').\n\n' +
      'Note: Already-synced past dates are kept in sync automatically (only changed cells are written).'
    );
    return;
  }

  const byClass = {};
  const classOrder = [];
  sessionsToSync.forEach(function(g) {
    if (!byClass[g.classId]) { byClass[g.classId] = []; classOrder.push(g.classId); }
    byClass[g.classId].push(g);
  });

  const undoBatch = [];
  const summaryLines = [];

  for (let c = 0; c < classOrder.length; c++) {
    const classId = classOrder[c];
    const className = classNamesMap[classId] || '';
    let res = null;
    try {
      res = syncClassDays_(masterSs, env, classId, className, byClass[classId], { manual: true, lockWaitMs: 10000 });
    } catch (err) {
      summaryLines.push('❌ ' + classId + ': ' + err.message);
      appendAuditLog_(masterSs, 'MANUAL_SYNC', 'ATTENDANCE_SYNC_ERROR', classId, err.message);
      continue;
    }
    res.lines.forEach(function(l) { summaryLines.push(l); });
    if (res.busy) {
      summaryLines.push('⏳ ' + classId + ': Master sheet busy, please retry in a minute.');
    }
    res.undoEntries.forEach(function(e) { undoBatch.push(e); });
    res.days.forEach(function(dr) {
      const multiNote = dr.g.sessionIds.length > 1 ? ' [Merged ' + dr.g.sessionIds.length + ' sessions]' : '';
      const detailStr = 'Present: ' + dr.counts.P + ' | Late: ' + dr.counts.L +
        (dr.counts.E ? ' | Excused: ' + dr.counts.E : '') + ' | Absent: ' + dr.counts.A +
        ' | Pushed: ' + dr.pushed + ' | Pulled from Sheet: ' + dr.pulled;
      summaryLines.push('✅ ' + classId + ' (Date ' + dr.g.day + '/' + dr.g.month + ')' + multiNote + ' — ' + detailStr);
    });
  }

  if (undoBatch.length > 0) {
    const persistRes = persistUndoBatch_(undoBatch, 10000, true);
    if (persistRes && persistRes.busy) {
      summaryLines.push('⏳ Master sheet busy: sync log not saved (Undo unavailable); files were written and the next sync will refresh them.');
    }
  }

  showAlert_('🎉 ATTENDANCE SYNC RESULTS\n\n' + summaryLines.join('\n'));
}

// 5. BUTTON 3: CONFLICT-FREE UNIFIED UNDO (CLASS DELETE & ATTENDANCE SYNC)
// ============================================================================
function undoLastAction() {
  ensureBackgroundTriggers_();

  const masterSs = SpreadsheetApp.getActiveSpreadsheet();
  const ui = SpreadsheetApp.getUi();
  const props = PropertiesService.getScriptProperties();
  const lastActionType = props.getProperty(LAST_ACTION_TYPE_KEY) || '';

  const deletedSnapshot = readDeletedClassBackup_(masterSs);
  const rawSyncBatch = props.getProperty(LAST_SYNC_PROP_KEY);

  if (!deletedSnapshot && !rawSyncBatch) {
    showAlert_('ℹ️ No recent Attendance Sync or Class Deletion found to undo.');
    return;
  }

  // If BOTH a deleted class backup and an attendance sync backup exist, let the user choose clearly
  if (deletedSnapshot && rawSyncBatch) {
    const classListStr = deletedSnapshot.classIds.join(', ');
    const undoBatch = JSON.parse(rawSyncBatch);
    const syncDescStr = undoBatch.map(function(item) {
      return item.classId + ' (' + item.day + '/' + item.month + ')';
    }).join(', ');

    if (lastActionType === 'CLASS_DELETE') {
      const choice = ui.alert(
        'Undo Deleted Class(es)',
        'Do you want to restore the deleted class(es): ' + classListStr + '?\n\n' +
        '• Click YES to restore deleted class(es): ' + classListStr + '\n' +
        '• Click NO if you want to undo the last Attendance Sync (' + syncDescStr + ') instead.',
        ui.ButtonSet.YES_NO_CANCEL
      );
      if (choice === ui.Button.YES) {
        executeClassDeleteUndo_(masterSs, props, deletedSnapshot, true);
        return;
      } else if (choice === ui.Button.NO) {
        const confirmSync = ui.alert(
          'Undo Last Attendance Sync',
          'Confirm undoing the last attendance sync on the target file(s) for: ' + syncDescStr + '?',
          ui.ButtonSet.YES_NO
        );
        if (confirmSync === ui.Button.YES) {
          executeAttendanceSyncUndo_(masterSs, props, undoBatch, true);
        }
        return;
      }
      return;
    } else {
      const choice = ui.alert(
        'Undo Last Attendance Sync',
        'Do you want to undo the last attendance sync for: ' + syncDescStr + '?\n\n' +
        '• Click YES to undo the Attendance Sync (' + syncDescStr + ')\n' +
        '• Click NO if you want to restore the deleted class(es) (' + classListStr + ') instead.',
        ui.ButtonSet.YES_NO_CANCEL
      );
      if (choice === ui.Button.YES) {
        executeAttendanceSyncUndo_(masterSs, props, undoBatch, true);
        return;
      } else if (choice === ui.Button.NO) {
        const confirmDel = ui.alert(
          'Undo Class Deletion',
          'Confirm restoring the deleted class(es): ' + classListStr + '?',
          ui.ButtonSet.YES_NO
        );
        if (confirmDel === ui.Button.YES) {
          executeClassDeleteUndo_(masterSs, props, deletedSnapshot, true);
        }
        return;
      }
      return;
    }
  }

  // Only deletedSnapshot exists
  if (deletedSnapshot) {
    const classListStr = deletedSnapshot.classIds.join(', ');
    const confirm = ui.alert(
      'Undo Class Deletion',
      'Do you want to restore the recently deleted class(es): ' + classListStr + '?\n\n' +
      'This will restore their Class Tab(s), Classes_Config rows, Sessions_Log, Attendance_Log, and Device_Registry.',
      ui.ButtonSet.YES_NO
    );
    if (confirm === ui.Button.YES) {
      executeClassDeleteUndo_(masterSs, props, deletedSnapshot, false);
    }
    return;
  }

  // Only rawSyncBatch exists
  if (rawSyncBatch) {
    const undoBatch = JSON.parse(rawSyncBatch);
    const descList = undoBatch.map(function(item) {
      return item.classId + ' (' + item.day + '/' + item.month + ')';
    }).join(', ');

    const confirm = ui.alert(
      'Undo Last Attendance Sync',
      'Do you want to undo the last attendance sync on the target Google Sheet / .XLSX file(s) for:\n' + descList + '?',
      ui.ButtonSet.YES_NO
    );
    if (confirm === ui.Button.YES) {
      executeAttendanceSyncUndo_(masterSs, props, undoBatch, false);
    }
  }
}

function executeClassDeleteUndo_(masterSs, props, deletedSnapshot, hasOtherUndo) {
  const classListStr = deletedSnapshot.classIds.join(', ');
  const restoreRes = restoreDeletedClassBackup_(masterSs, deletedSnapshot);
  if (restoreRes && restoreRes.busy) {
    showAlert_('⏳ Master sheet is busy with student check-ins. Nothing was restored, please retry in a minute.');
    return;
  }
  clearDeletedClassBackup_(masterSs);
  props.setProperty(LAST_ACTION_TYPE_KEY, hasOtherUndo ? 'ATTENDANCE_SYNC' : '');
  appendAuditLog_(masterSs, 'SYNC_MENU', 'UNDO_CLASS_DELETE', classListStr, 'Restored deleted class tabs and logs');
  showAlert_('✅ RESTORE COMPLETED\n\nSuccessfully restored class(es): ' + classListStr + '.');
}

function executeAttendanceSyncUndo_(masterSs, props, undoBatch, hasOtherUndo) {
  const restoredLines = [];
  const undone = [];
  const styleNow = syncStyleKey_();

  for (let i = 0; i < undoBatch.length; i++) {
    const item = undoBatch[i];
    const targetInfo = {
      fileId: item.xlsxFileId,
      isGoogleSheet: Boolean(item.isGoogleSheet)
    };
    const res = updateTargetAttendanceFile_(
      targetInfo,
      item.classId,
      item.className || '',
      item.day,
      item.month,
      {},
      item.previousCells || []
    );

    let dayKey = String(item.dayKey || '');
    if (!dayKey && item.groupKey) {
      const gk = String(item.groupKey);
      const dm = gk.slice(gk.lastIndexOf('__') + 2).match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
      if (dm) dayKey = dm[1] + '-' + syncPad2_(dm[2]) + '-' + syncPad2_(dm[3]);
    }
    // Base = the restored column (blank counts as present); unreadable cells keep their stored base value
    const restored = {};
    (item.previousCells || []).forEach(function(pc) {
      if (!pc || !pc.sid) return;
      const sidU = String(pc.sid).toUpperCase();
      const m = readDayColumnMarks_([[pc.val]], [[pc.bg || '']], [sidU], getAttendanceStyle_())[sidU];
      if (m === 'BLANK') restored[sidU] = 'P';
      else if (m === 'P' || m === 'L' || m === 'E' || m === '-') restored[sidU] = m;
    });
    undone.push({ classId: item.classId, dayKey: dayKey, fileId: item.xlsxFileId, restored: restored, afterMs: Number(res && res.afterModifiedMs) || 0 });

    restoredLines.push('↩️ ' + item.classId + ': Restored date column ' + item.day + '/' + item.month + ' to its previous state.');
    appendAuditLog_(masterSs, 'SYNC_MENU', 'UNDO_ATTENDANCE_SYNC', item.classId + ' (' + item.day + '/' + item.month + ')', 'Restored previous column values');
  }

  // Pause these days (push and pull) until the manual Sync button is used again
  const undoPersist = withMasterWriteLock_({}, 10000, function() {
    const cur = readSyncBaseAll_(masterSs).map;
    const upserts = [];
    undone.forEach(function(u) {
      if (!u.dayKey) return;
      const prev = cur[syncBaseKey_(u.classId, u.dayKey)];
      upserts.push({
        classId: u.classId,
        dayKey: u.dayKey,
        fileId: u.fileId,
        base: Object.assign({}, prev ? prev.base : {}, u.restored),
        sheetModifiedMs: u.afterMs,
        styleHash: styleNow,
        paused: SYNC_PAUSE_UNDO
      });
    });
    if (upserts.length > 0) writeSyncBaseBatch_(masterSs, upserts, []);
    props.deleteProperty(LAST_SYNC_PROP_KEY);
    props.setProperty(LAST_ACTION_TYPE_KEY, hasOtherUndo ? 'CLASS_DELETE' : '');
    return { busy: false };
  });
  if (undoPersist && undoPersist.busy) {
    showAlert_('⏳ Files were restored but the Master sheet is busy, so the sync log was not updated. Please run Undo again in a minute.');
    return;
  }
  showAlert_('✅ UNDO COMPLETED\n\n' + restoredLines.join('\n') + '\n\nAuto-sync of these dates is paused until you press "Sync Today\'s Attendance".');
}

function deleteSelectedClasses(selectedClassIds) {
  if (!selectedClassIds || selectedClassIds.length === 0) {
    return { cancelled: true };
  }

  const ui = SpreadsheetApp.getUi();
  const classListStr = selectedClassIds.join(', ');
  const confirm = ui.alert(
    'Confirm Class Deletion',
    'Are you sure you want to delete the selected class(es): ' + classListStr + '?\n\n' +
    '• Class tab(s), Classes_Config, Sessions_Log, Attendance_Log, and Device_Registry for these classes will be cleaned up.\n' +
    '• You can undo this deletion anytime via \'3. Undo Last Action\'.',
    ui.ButtonSet.YES_NO
  );

  if (confirm !== ui.Button.YES) {
    return { cancelled: true };
  }

  const masterSs = SpreadsheetApp.getActiveSpreadsheet();
  const targetMap = {};
  for (let i = 0; i < selectedClassIds.length; i++) {
    targetMap[String(selectedClassIds[i]).trim()] = true;
  }

  const snapshot = {
    deletedAt: new Date().toISOString(),
    classIds: selectedClassIds,
    configRows: [],
    classTabs: {},
    sessionRows: [],
    attendanceRows: [],
    deviceRows: []
  };

  const deleteLockRes = withMasterWriteLock_({}, 20000, function() {
    const deletedClassStudents = {};
    const remainingClassStudents = {};

    const cfgSheet = masterSs.getSheetByName('Classes_Config');
    if (cfgSheet) {
      const cfgData = cfgSheet.getDataRange().getValues();
      for (let i = 1; i < cfgData.length; i++) {
        const cid = String(cfgData[i][0] || '').trim();
        if (!cid) continue;
        const cSheet = masterSs.getSheetByName(cid);
        const rawMap = collectMasterClassStudentIds_(cSheet).rawMap;
        const targetBucket = targetMap[cid] ? deletedClassStudents : remainingClassStudents;
        for (const sid in rawMap) {
          if (Object.prototype.hasOwnProperty.call(rawMap, sid)) targetBucket[sid] = true;
        }
      }
    }

    for (let i = 0; i < selectedClassIds.length; i++) {
      const cid = String(selectedClassIds[i]).trim();
      const cSheet = masterSs.getSheetByName(cid);
      if (cSheet) {
        snapshot.classTabs[cid] = cSheet.getDataRange().getValues();
        masterSs.deleteSheet(cSheet);
      }
    }

    partitionAndRewriteSheetByClassCol_(cfgSheet, 0, targetMap, snapshot.configRows);
    partitionAndRewriteSheetByClassCol_(masterSs.getSheetByName('Sessions_Log'), 1, targetMap, snapshot.sessionRows);
    partitionAndRewriteSheetByClassCol_(masterSs.getSheetByName('Attendance_Log'), 3, targetMap, snapshot.attendanceRows);

    const devSheet = masterSs.getSheetByName('Device_Registry');
    if (devSheet) {
      const devData = devSheet.getDataRange().getValues();
      const keepDev = [devData[0]];
      for (let i = 1; i < devData.length; i++) {
        const sid = normalizeStudentId_(devData[i][0]);
        if (sid && deletedClassStudents[sid] && !remainingClassStudents[sid]) {
          snapshot.deviceRows.push(devData[i]);
        } else {
          keepDev.push(devData[i]);
        }
      }
      rewriteSheetRows_(devSheet, keepDev);
    }

    saveDeletedClassBackup_(masterSs, snapshot);
    return { busy: false };
  });

  if (deleteLockRes && deleteLockRes.busy) {
    showAlert_('⏳ Master sheet is busy with student check-ins. Nothing was deleted, please retry in a minute.');
    return { cancelled: true };
  }
  invalidateCachesAfterClassMutation_(snapshot);

  PropertiesService.getScriptProperties().setProperty(LAST_ACTION_TYPE_KEY, 'CLASS_DELETE');
  appendAuditLog_(
    masterSs,
    'SYNC_MENU',
    'DELETE_CLASSES',
    classListStr,
    'Removed ' + Object.keys(snapshot.classTabs).length + ' tabs, ' + snapshot.sessionRows.length + ' sessions, ' + snapshot.attendanceRows.length + ' attendance logs'
  );
  SpreadsheetApp.flush();

  showAlert_(
    '🗑️ DELETION COMPLETED\n\n' +
    'Removed class(es): ' + classListStr + '\n' +
    '• Class tab(s) deleted: ' + Object.keys(snapshot.classTabs).length + '\n' +
    '• Sessions_Log rows removed: ' + snapshot.sessionRows.length + '\n' +
    '• Attendance_Log rows removed: ' + snapshot.attendanceRows.length + '\n' +
    '• Device_Registry rows removed: ' + snapshot.deviceRows.length + '\n\n' +
    '💡 Tip: If you deleted a class by mistake, click \'3. Undo Last Action\' to restore everything.'
  );

  return { cancelled: false };
}

function collectMasterClassStudentIds_(classSheet) {
  let count = 0;
  const upperMap = {};
  const rawMap = {};
  if (classSheet && classSheet.getLastRow() > 1) {
    const sVals = classSheet.getRange(2, 2, classSheet.getLastRow() - 1, 2).getValues();
    for (let r = 0; r < sVals.length; r++) {
      const sid = normalizeStudentId_(sVals[r][0], sVals[r][1]);
      if (sid) {
        upperMap[sid.toUpperCase()] = true;
        rawMap[sid] = true;
        count++;
      }
    }
  }
  return { count: count, upperMap: upperMap, rawMap: rawMap };
}

function partitionAndRewriteSheetByClassCol_(sheet, classColIdx, targetMap, snapshotList) {
  if (!sheet) return;
  const data = sheet.getDataRange().getValues();
  const keepRows = [data[0]];
  for (let i = 1; i < data.length; i++) {
    const cid = String(data[i][classColIdx] || '').trim();
    if (targetMap[cid]) {
      snapshotList.push(data[i]);
    } else {
      keepRows.push(data[i]);
    }
  }
  rewriteSheetRows_(sheet, keepRows);
}

function rewriteSheetRows_(sheet, rows) {
  sheet.clearContents();
  if (rows && rows.length > 0) {
    sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
  }
}

function saveDeletedClassBackup_(masterSs, snapshot) {
  let backupSheet = masterSs.getSheetByName(UNDO_BACKUP_SHEET_NAME);
  if (!backupSheet) {
    backupSheet = masterSs.insertSheet(UNDO_BACKUP_SHEET_NAME);
    backupSheet.hideSheet();
  }
  backupSheet.clearContents();

  const jsonStr = JSON.stringify(snapshot);
  const chunkSize = 35000;
  const chunks = [];
  for (let i = 0; i < jsonStr.length; i += chunkSize) {
    chunks.push([jsonStr.slice(i, i + chunkSize)]);
  }
  if (chunks.length > 0) {
    backupSheet.getRange(1, 1, chunks.length, 1).setValues(chunks);
  }
  backupSheet.hideSheet();
}

function readDeletedClassBackup_(masterSs) {
  const backupSheet = masterSs.getSheetByName(UNDO_BACKUP_SHEET_NAME);
  if (!backupSheet || backupSheet.getLastRow() === 0) return null;
  try {
    const vals = backupSheet.getRange(1, 1, backupSheet.getLastRow(), 1).getValues();
    let jsonStr = '';
    for (let i = 0; i < vals.length; i++) {
      jsonStr += String(vals[i][0] || '');
    }
    if (!jsonStr) return null;
    return JSON.parse(jsonStr);
  } catch (e) {
    return null;
  }
}

function clearDeletedClassBackup_(masterSs) {
  const backupSheet = masterSs.getSheetByName(UNDO_BACKUP_SHEET_NAME);
  if (backupSheet) {
    backupSheet.clearContents();
    backupSheet.hideSheet();
  }
}

function restoreDeletedClassBackup_(masterSs, snapshot) {
  const lockRes = withMasterWriteLock_({}, 20000, function() {
    const tabNames = Object.keys(snapshot.classTabs || {});
    for (let i = 0; i < tabNames.length; i++) {
      const cid = tabNames[i];
      const rows = snapshot.classTabs[cid];
      let cSheet = masterSs.getSheetByName(cid);
      if (!cSheet) {
        cSheet = masterSs.insertSheet(cid);
      } else {
        cSheet.clearContents();
      }
      if (rows && rows.length > 0) {
        if (rows[0].length >= 2) cSheet.getRange(1, 2, rows.length, 1).setNumberFormat('@');
        cSheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
      }
    }

    appendRowsToSheet_(masterSs.getSheetByName('Classes_Config'), snapshot.configRows);
    appendRowsToSheet_(masterSs.getSheetByName('Sessions_Log'), snapshot.sessionRows);
    appendRowsToSheet_(masterSs.getSheetByName('Attendance_Log'), snapshot.attendanceRows);
    appendRowsToSheet_(masterSs.getSheetByName('Device_Registry'), snapshot.deviceRows);
    SpreadsheetApp.flush();
    return { busy: false };
  });
  if (lockRes && lockRes.busy) return { busy: true };
  invalidateCachesAfterClassMutation_(snapshot);
  return { busy: false };
}

function withMasterWriteLock_(opts, waitMs, fn) {
  if (opts && opts.lockHeld) return fn();
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(waitMs)) return { busy: true };
  try {
    return fn();
  } finally {
    lock.releaseLock();
  }
}

function updateSyncedSessionsMap_(sets, deletes, waitMs, extra) {
  return withMasterWriteLock_({}, waitMs, function() {
    const p = PropertiesService.getScriptProperties();
    const cur = JSON.parse(p.getProperty(AUTO_SYNCED_SESSIONS_KEY) || '{}');
    Object.keys(sets || {}).forEach(function(k) { cur[k] = sets[k]; });
    (deletes || []).forEach(function(k) { delete cur[k]; });
    p.setProperty(AUTO_SYNCED_SESSIONS_KEY, JSON.stringify(cur));
    if (typeof extra === 'function') extra(p);
    return { busy: false };
  });
}

function invalidateCachesAfterClassMutation_(snapshot) {
  try {
    const sessionIds = (snapshot.sessionRows || []).map(function(r) { return String(r[0] || '').trim(); }).filter(Boolean);
    if (typeof invalidateSessionAttendanceMaps_ === 'function') invalidateSessionAttendanceMaps_(sessionIds);
    if (typeof invalidateSystemCache === 'function') invalidateSystemCache();
    if (typeof invalidateClassCache === 'function') {
      (snapshot.classIds || []).forEach(function(cid) { invalidateClassCache(String(cid).trim()); });
    }
    const devKeys = (snapshot.deviceRows || [])
      .map(function(r) { return String(r[0] || '').trim().toUpperCase(); })
      .filter(Boolean)
      .map(function(sid) { return 'DEV_HASH_' + sid; });
    if (devKeys.length > 0) CacheService.getScriptCache().removeAll(devKeys);
  } catch (e) {}
}

function appendRowsToSheet_(sheet, rows) {
  if (!sheet || !rows || rows.length === 0) return;
  const startRow = sheet.getLastRow() + 1;
  sheet.getRange(startRow, 1, rows.length, rows[0].length).setValues(rows);
}

// ============================================================================
// 8. BACKGROUND TWO-WAY AUTO-SYNC (RUNS EVERY 5 MINUTES)
//    Per (class, day): 3-way merge of Web (Master Sheet) vs Sheet/XLSX vs the stored base (_Sync_Base).
//    Sheet wins on conflict; only changed cells are written; nothing is read/converted when the file is unchanged.
// ============================================================================
function syncGateToMs_(v) {
  if (v && Object.prototype.toString.call(v) === '[object Date]') return v.getTime();
  const t = Date.parse(String(v || ''));
  return isNaN(t) ? 0 : t;
}

function syncGateDayKey_(d) {
  return d.getFullYear() + '-' + ('0' + (d.getMonth() + 1)).slice(-2) + '-' + ('0' + d.getDate()).slice(-2);
}

// Cheap gate (no Drive/Calendar): true when a class has a session today (+3h after it ends) or is scheduled today within [start-45m, end+3h].
// Returns 'active' | 'idle'. Unknown schedule -> fail-open between 06:00 and 23:00.
function getSync2wGate_(masterSs, props, nowMs) {
  const now = new Date(nowMs);
  const todayKey = syncGateDayKey_(now);
  const nowMin = now.getHours() * 60 + now.getMinutes();

  try {
    const sh = masterSs.getSheetByName('Sessions_Log');
    const last = sh ? sh.getLastRow() : 0;
    if (last > 1) {
      const n = Math.min(SYNC2W_GATE_SESSION_ROWS, last - 1);
      const rows = sh.getRange(last - n + 1, 1, n, 7).getValues();
      for (let i = 0; i < rows.length; i++) {
        const openedMs = syncGateToMs_(rows[i][3]);
        if (!openedMs || syncGateDayKey_(new Date(openedMs)) !== todayKey) continue;
        const endMs = syncGateToMs_(rows[i][6]) || syncGateToMs_(rows[i][5]) || 0;
        if (!endMs || nowMs <= endMs + SYNC2W_GATE_AFTER_MIN * 60000) return 'active';
      }
    }
  } catch (e) {
    return 'active';
  }

  let usable = 0;
  let gateActive = false;
  try {
    const raw = props.getProperty('QR_CLASS_SCHEDULE_META_MAP');
    const map = raw ? JSON.parse(raw) : {};
    Object.keys(map).forEach(function(cid) {
      const en = map[cid];
      if (!en || typeof en !== 'object') return;
      const sd = Array.isArray(en.sessionDates) ? en.sessionDates : [];
      const md = Array.isArray(en.makeUpDates) ? en.makeUpDates : [];
      const dw = Array.isArray(en.daysOfWeek) ? en.daysOfWeek : (typeof en.dayOfWeek === 'number' ? [en.dayOfWeek] : []);
      if (sd.length === 0 && md.length === 0 && dw.length === 0) return;
      usable++;
      const scheduled = sd.indexOf(todayKey) !== -1 || md.indexOf(todayKey) !== -1 || (sd.length === 0 && dw.indexOf(now.getDay()) !== -1);
      if (!scheduled) return;
      let startMin = SYNC2W_GATE_FAILOPEN_FROM_H * 60 + SYNC2W_GATE_BEFORE_MIN;
      let endMin = SYNC2W_GATE_FAILOPEN_TO_H * 60 - SYNC2W_GATE_AFTER_MIN;
      const sp = Number(en.startPeriod), ep = Number(en.endPeriod || en.startPeriod);
      if (sp >= 1 && ep >= sp && ep <= 20) {
        startMin = sp <= 6 ? 390 + (sp - 1) * 50 : 750 + (sp - 7) * 50;
        endMin = ep <= 6 ? 390 + ep * 50 : 750 + (ep - 6) * 50;
      }
      if (nowMin >= startMin - SYNC2W_GATE_BEFORE_MIN && nowMin <= endMin + SYNC2W_GATE_AFTER_MIN) gateActive = true;
    });
  } catch (e) {
    usable = 0;
  }
  if (gateActive) return 'active';
  if (usable === 0) {
    return (now.getHours() >= SYNC2W_GATE_FAILOPEN_FROM_H && now.getHours() < SYNC2W_GATE_FAILOPEN_TO_H) ? 'active' : 'idle';
  }
  return 'idle';
}

function autoSyncTodayAttendanceAfterClass() {
  const masterSs = SpreadsheetApp.getActiveSpreadsheet();
  const props = PropertiesService.getScriptProperties();
  const gateNowMs = Date.now();
  let gate = 'active';
  try { gate = getSync2wGate_(masterSs, props, gateNowMs); } catch (e) { gate = 'active'; }
  if (gate === 'idle') {
    const lastFull = parseInt(props.getProperty(SYNC2W_LAST_FULL_KEY) || '0', 10) || 0;
    if (gateNowMs - lastFull < SYNC2W_IDLE_INTERVAL_MS) return;
  }
  let cache = null;
  try { cache = CacheService.getScriptCache(); } catch (e) {}
  if (cache) {
    try { if (cache.get(SYNC2W_RUNNING_KEY)) return; } catch (e) {}
    try { cache.put(SYNC2W_RUNNING_KEY, String(Date.now()), SYNC2W_RUNNING_TTL_SEC); } catch (e) {}
  }
  try { props.setProperty(SYNC2W_LAST_FULL_KEY, String(gateNowMs)); } catch (e) {}
  const t0 = Date.now();
  let stats = null;
  try {
    stats = runTwoWayAutoSync_(masterSs, t0 + SYNC2W_DEADLINE_MS);
  } catch (err) {
    appendAuditLog_(masterSs, 'AUTO_SYNC', 'ATTENDANCE_SYNC_ERROR', 'SYNC2W', String(err && err.message || err || ''));
  } finally {
    if (cache) { try { cache.remove(SYNC2W_RUNNING_KEY); } catch (e) {} }
  }
  const elapsed = Date.now() - t0;
  if (stats && (stats.worked > 0 || stats.errors > 0 || elapsed >= SYNC2W_LOG_MIN_MS)) {
    appendAuditLog_(
      masterSs,
      'AUTO_SYNC',
      'SYNC2W_RUN',
      'classes=' + stats.classes + ' worked=' + stats.worked,
      'gate=' + gate + ', elapsedMs=' + elapsed + ', pushedCells=' + stats.pushed + ', pulledCells=' + stats.pulled +
      ', busy=' + stats.busy + ', errors=' + stats.errors + (stats.deadlineHit ? ', DEADLINE_HIT' : '')
    );
  }
}

function runTwoWayAutoSync_(masterSs, deadlineMs) {
  const stats = { classes: 0, worked: 0, pushed: 0, pulled: 0, busy: 0, errors: 0, deadlineHit: false };
  const props = PropertiesService.getScriptProperties();
  const now = new Date();

  const mig = ensureSyncBaseMigrated_(masterSs, props);
  if (mig && mig.busy) {
    stats.busy++;
    appendAuditLog_(masterSs, 'AUTO_SYNC', 'AUTO_SYNC_LOCK_BUSY', 'SYNC_BASE_MIGRATION', 'Master sheet busy, will retry on next run');
    return stats;
  }

  const allGroups = getGroupedSessionsByClassDate_(masterSs);
  if (allGroups.length === 0) return stats;

  const byClass = {};
  allGroups.forEach(function(g) { (byClass[g.classId] = byClass[g.classId] || []).push(g); });
  const classIds = Object.keys(byClass).sort();
  const classNamesMap = getClassNamesMap_(masterSs);
  const env = makeSyncEnv_(masterSs, now);

  let startIdx = parseInt(props.getProperty(SYNC2W_CURSOR_KEY) || '0', 10);
  if (!(startIdx >= 0) || startIdx >= classIds.length) startIdx = 0;

  const undoBatch = [];
  let nextCursor = 0;
  for (let k = 0; k < classIds.length; k++) {
    const idx = (startIdx + k) % classIds.length;
    if (Date.now() >= deadlineMs) {
      stats.deadlineHit = true;
      nextCursor = idx;
      break;
    }
    const classId = classIds[idx];
    stats.classes++;
    let res = null;
    try {
      res = syncClassDays_(masterSs, env, classId, classNamesMap[classId] || '', byClass[classId], { manual: false, deadlineMs: deadlineMs });
    } catch (err) {
      stats.errors++;
      appendAuditLog_(masterSs, 'AUTO_SYNC', 'ATTENDANCE_SYNC_ERROR', classId, String(err && err.message || err || ''));
      continue;
    }
    if (res.busy) {
      stats.busy++;
      if (!env.lockBusyLogged) {
        env.lockBusyLogged = true;
        appendAuditLog_(masterSs, 'AUTO_SYNC', 'AUTO_SYNC_LOCK_BUSY', classId, 'Master sheet busy, will retry on next run');
      }
    }
    if (res.worked) stats.worked++;
    stats.pushed += res.pushed;
    stats.pulled += res.pulled;
    stats.errors += res.errors;
    res.undoEntries.forEach(function(e) { undoBatch.push(e); });
  }

  if (stats.deadlineHit || props.getProperty(SYNC2W_CURSOR_KEY) !== null) {
    try { props.setProperty(SYNC2W_CURSOR_KEY, String(stats.deadlineHit ? nextCursor : 0)); } catch (e) {}
  }

  if (undoBatch.length > 0) {
    const persistRes = persistUndoBatch_(undoBatch, 5000, false);
    if (persistRes && persistRes.busy && !env.lockBusyLogged) {
      appendAuditLog_(masterSs, 'AUTO_SYNC', 'AUTO_SYNC_LOCK_BUSY', 'UNDO_LOG', 'Master sheet busy, undo log not saved');
    }
  }
  return stats;
}

function makeSyncEnv_(masterSs, now) {
  return {
    now: now,
    currentDayStr: now.getFullYear() + '-' + (now.getMonth() + 1) + '-' + now.getDate(),
    baseMap: readSyncBaseAll_(masterSs).map,
    styleNow: syncStyleKey_(),
    pre: null,
    lockBusyLogged: false
  };
}

function syncStyleKey_() {
  return 'S' + styleHash_(getAttendanceStyle_());
}

function ensureWebPre_(masterSs, env, classId) {
  if (!env.pre) {
    const a = masterSs.getSheetByName('Attendance_Log');
    env.pre = { attData: a ? a.getDataRange().getValues() : [], classValues: {} };
  }
  if (!Object.prototype.hasOwnProperty.call(env.pre.classValues, classId)) {
    const s = masterSs.getSheetByName(classId);
    env.pre.classValues[classId] = s ? s.getDataRange().getValues() : null;
  }
  return env.pre;
}

function persistUndoBatch_(undoBatch, waitMs, forceType) {
  if (!undoBatch || undoBatch.length === 0) return { busy: false };
  return withMasterWriteLock_({}, waitMs, function() {
    const p = PropertiesService.getScriptProperties();
    const trimmed = undoBatch.slice();
    let json = JSON.stringify(trimmed);
    while (json.length > 8500 && trimmed.length > 1) {
      trimmed.shift();
      json = JSON.stringify(trimmed);
    }
    try {
      p.setProperty(LAST_SYNC_PROP_KEY, json);
      if (forceType || p.getProperty(LAST_ACTION_TYPE_KEY) !== 'CLASS_DELETE') {
        p.setProperty(LAST_ACTION_TYPE_KEY, 'ATTENDANCE_SYNC');
      }
    } catch (e) {}
    return { busy: false };
  });
}

/**
 * Two-way sync of the given session groups of ONE class. Used by the 5-minute trigger and the manual button.
 * Order: decide from cheap data -> (maybe) one convert/read -> merge -> Sheet writes (no lock) -> short lock for
 * Attendance_Log + _Sync_Base. A cell becomes "agreed" (base) only after the other side was written successfully.
 */
function syncClassDays_(masterSs, env, classId, className, groups, opts) {
  opts = opts || {};
  const manual = !!opts.manual;
  const actor = manual ? 'MANUAL_SYNC' : 'AUTO_SYNC';
  const lockWaitMs = opts.lockWaitMs || SYNC2W_LOCK_WAIT_MS;
  const nowMs = env.now.getTime();
  const has = function(o, k) { return Object.prototype.hasOwnProperty.call(o, k); };
  const out = { undoEntries: [], busy: false, lines: [], pushed: 0, pulled: 0, errors: 0, worked: false, days: [] };
  const skipUpserts = [];
  const markSkipDay = function(g) {
    skipUpserts.push({ classId: classId, dayKey: g.dayKey, fileId: '', base: {}, sheetModifiedMs: 0, styleHash: '', paused: SYNC_SKIP_DAY });
  };

  // Mark-settings changed since the base was stored: stop syncing this class until the manual Sync button.
  if (!manual) {
    const prefix = classId + '|';
    const rows = Object.keys(env.baseMap).filter(function(k) { return k.indexOf(prefix) === 0; }).map(function(k) { return env.baseMap[k]; });
    const styleChanged = rows.some(function(r) { return r.paused === 0 && r.styleHash && r.styleHash !== env.styleNow; });
    if (styleChanged) {
      const pauseUpserts = rows.filter(function(r) { return r.paused === 0; }).map(function(r) {
        return { classId: classId, dayKey: r.dayKey, paused: SYNC_PAUSE_STYLE };
      });
      const pr = withMasterWriteLock_({}, lockWaitMs, function() {
        writeSyncBaseBatch_(masterSs, pauseUpserts, []);
        return { busy: false };
      });
      if (pr && pr.busy) { out.busy = true; return out; }
      pauseUpserts.forEach(function(u) { env.baseMap[syncBaseKey_(classId, u.dayKey)].paused = SYNC_PAUSE_STYLE; });
      appendAuditLog_(masterSs, actor, 'SYNC_STYLE_CHANGED', classId, 'Attendance mark settings changed: sync paused for this class until the manual Sync button is used');
      return out;
    }
  }

  // 1. candidate days (cheap, in-memory)
  const days = [];
  groups.forEach(function(g) {
    const be = env.baseMap[syncBaseKey_(classId, g.dayKey)] || null;
    const isToday = g.dateKey === env.currentDayStr;
    if (!manual && be) {
      if (be.paused === SYNC_PAUSE_UNDO || be.paused === SYNC_PAUSE_STYLE) return;
      if (be.paused === SYNC_SKIP_DAY && !isToday) return;
    }
    const hasBase = !!be && be.paused !== SYNC_SKIP_DAY;
    const d = { g: g, be: hasBase ? be : null, hasBase: hasBase, isToday: isToday, skip: false, mode: hasBase ? 'BASE' : 'FIRST' };
    if (!manual) {
      if (isToday && nowMs < getClassEndTimeOnCalendarMs_(masterSs, classId, env.now, g.openedAtMs)) d.skip = true;
      else if (!g.allClosed && nowMs < g.maxClosesAtMs) d.skip = true;
    }
    if (d.skip) return;
    d.web = getMergedAttendanceStatusMap_(masterSs, classId, g, ensureWebPre_(masterSs, env, classId)) || {};
    if (Object.keys(d.web).length === 0) return;
    if (!hasBase) {
      if (!manual) {
        const real = Object.keys(d.web).some(function(sid) { const st = d.web[sid]; return st === 'P' || st === 'L' || st === 'E'; });
        if (!real) { if (!isToday) markSkipDay(g); return; }
      }
    } else {
      d.pushDiff = computePushDiff_(be.base, d.web);
    }
    days.push(d);
  });

  // 2. target file + decide whether anything must be read
  let ti = null;
  if (days.length > 0) {
    ti = resolveClassTargetFile_(masterSs, classId, className);
    if (!ti) {
      if (manual) out.lines.push('⚠️ ' + classId + ': Skipped (Target Google Sheet / .XLSX file not found on Google Drive).');
      days.length = 0;
    }
  }
  const cur = ti ? (ti.lastUpdatedMs || 0) : 0;
  let needRead = false;
  const active = [];
  days.forEach(function(d) {
    const unchanged = !manual && d.mode === 'BASE' && cur > 0 && d.be.sheetModifiedMs === cur;
    d.needRead = !unchanged;
    if (d.needRead) needRead = true;
    else if (Object.keys(d.pushDiff).length === 0) return;
    active.push(d);
  });

  // 3. one conversion/read per class: roster + every date column
  let extract = null;
  if (needRead && opts.deadlineMs && Date.now() >= opts.deadlineMs) return out;
  if (needRead) {
    try {
      extract = extractRosterFromTargetFile_(ti, classId, className);
    } catch (err) {
      out.errors++;
      appendAuditLog_(masterSs, actor, 'ATTENDANCE_SYNC_ERROR', classId, String(err && err.message || err || ''));
      if (manual) out.lines.push('❌ ' + classId + ': ' + String(err && err.message || err || ''));
      active.length = 0;
    }
    if (extract && typeof syncSingleClassRosterFromDrive_ === 'function') {
      try {
        const rr = syncSingleClassRosterFromDrive_(masterSs, classId, className, {
          onlyIfModified: false,
          actor: actor,
          extractResult: extract,
          lockWaitMs: lockWaitMs,
          groups: groups
        });
        if (rr && rr.busy) { out.busy = true; return out; }
      } catch (e) {}
    }
  }

  // 4. merge
  active.forEach(function(d) {
    if (!d.needRead) {
      d.merge = { toSheet: d.pushDiff, toWeb: {}, newBase: Object.assign({}, d.be.base, d.pushDiff), conflicts: [] };
      return;
    }
    const col = ((extract && extract.dateColumns) || []).filter(function(c) { return c.dayKey === d.g.dayKey; })[0] || null;
    let sheetMarks;
    if (col) {
      sheetMarks = col.marks || {};
    } else if (d.mode === 'FIRST') {
      sheetMarks = {};
      Object.keys(d.web).forEach(function(sid) { sheetMarks[sid] = 'BLANK'; });
    } else {
      d.skip = true;
      return;
    }
    d.sheetMarks = sheetMarks;
    d.merge = computeThreeWayMerge_(d.mode === 'FIRST' ? {} : d.be.base, d.web, sheetMarks, { firstPush: d.mode === 'FIRST' });
  });

  // 5. Sheet writes (outside the lock)
  active.forEach(function(d) {
    if (d.skip || !d.merge || Object.keys(d.merge.toSheet).length === 0) return;
    if (opts.deadlineMs && Date.now() >= opts.deadlineMs) { d.skip = true; return; }
    try {
      const res = updateTargetAttendanceFile_(ti, classId, className, d.g.day, d.g.month, d.merge.toSheet, null, d.needRead ? d.sheetMarks : expectedMarksFromBase_(d.be.base, d.merge.toSheet));
      d.written = res;
      d.afterMs = Number(res && res.afterModifiedMs) || 0;
      d.skippedSids = (res && res.skippedSids) || [];
      if (!res.previousCells || res.previousCells.length > 0) out.undoEntries.push(buildAttendanceUndoEntry_(classId, className, ti, res, d.g, null));
      out.worked = true;
    } catch (err) {
      d.skip = true;
      const msg = String(err && err.message || err || '');
      if (msg.indexOf('Date column ') !== -1 && msg.indexOf('not found') !== -1) {
        if (!manual && d.mode === 'FIRST' && !d.isToday) markSkipDay(d.g);
        else if (manual) out.lines.push('❌ ' + classId + ' (Date ' + d.g.day + '/' + d.g.month + '): ' + msg);
      } else {
        out.errors++;
        appendAuditLog_(masterSs, actor, 'ATTENDANCE_SYNC_ERROR', classId + ' (' + d.g.day + '/' + d.g.month + ')', msg);
        if (manual) out.lines.push('❌ ' + classId + ' (Date ' + d.g.day + '/' + d.g.month + '): ' + msg);
      }
    }
  });

  // 6. short lock: pull into Attendance_Log + base + master grid
  const commitDays = active.filter(function(d) { return !d.skip && d.merge; });
  const upserts = skipUpserts.slice();
  const lockRes = (commitDays.length === 0 && upserts.length === 0) ? { busy: false } : withMasterWriteLock_({}, lockWaitMs, function() {
    let touchedWeb = false;
    commitDays.forEach(function(d) {
      const m = d.merge;
      const finalBase = Object.assign({}, m.newBase);
      const revert = function(sid) {
        if (d.mode === 'BASE' && has(d.be.base, sid)) finalBase[sid] = d.be.base[sid];
        else delete finalBase[sid];
      };
      let retry = false;
      // Cells the Sheet writer skipped (edited meanwhile) stay unagreed: merged again next run
      (d.skippedSids || []).forEach(function(sid) { revert(sid); retry = true; });
      // Web changed after the snapshot (e.g. a check-in): do not overwrite it with the older Sheet value
      let toWebNow = m.toWeb;
      if (Object.keys(m.toWeb).length > 0) {
        let fresh = null;
        try { fresh = getMergedAttendanceStatusMap_(masterSs, classId, d.g) || {}; } catch (e) { fresh = null; }
        toWebNow = {};
        Object.keys(m.toWeb).forEach(function(sid) {
          if (fresh && has(fresh, sid) && fresh[sid] !== d.web[sid]) { revert(sid); retry = true; }
          else toWebNow[sid] = m.toWeb[sid];
        });
      }
      const toWebKeys = Object.keys(toWebNow);
      if (toWebKeys.length > 0) {
        let ok = false;
        try {
          const wr = typeof writeManualMarksForDay_ === 'function'
            ? writeManualMarksForDay_(masterSs, classId, d.g.dayKey, toWebNow, { lockHeld: true, actor: 'SYNC2W', requestTag: 'SHEET_PULL' })
            : null;
          ok = !!(wr && wr.ok);
        } catch (e) { ok = false; }
        if (ok) {
          touchedWeb = true;
          d.pulledCount = toWebKeys.length;
        } else {
          out.errors++;
          toWebKeys.forEach(revert);
          retry = true;
        }
      }
      const ms = retry ? (d.be ? d.be.sheetModifiedMs : 0) : (d.written ? d.afterMs : cur);
      const prev = d.be;
      const same = !!prev && prev.paused === 0 && prev.fileId === ti.fileId && prev.styleHash === env.styleNow &&
        prev.sheetModifiedMs === ms && JSON.stringify(prev.base) === JSON.stringify(finalBase);
      if (!same) {
        upserts.push({ classId: classId, dayKey: d.g.dayKey, fileId: ti.fileId, base: finalBase, sheetModifiedMs: ms, styleHash: env.styleNow, paused: 0 });
      }
    });
    if (commitDays.length > 0 && (touchedWeb || commitDays.some(function(d) { return d.written || d.mode === 'FIRST'; }))) {
      const last = commitDays[commitDays.length - 1];
      try { updateMasterClassSheet_(masterSs, classId, last.g, last.web); } catch (e) {}
    }
    if (upserts.length > 0) writeSyncBaseBatch_(masterSs, upserts, []);
    return { busy: false };
  });

  if (lockRes && lockRes.busy) {
    out.busy = true;
    return out;
  }
  upserts.forEach(function(u) {
    const k = syncBaseKey_(classId, u.dayKey);
    env.baseMap[k] = Object.assign({ updatedAt: '' }, env.baseMap[k] || {}, u);
  });

  // 7. audit + summaries
  commitDays.forEach(function(d) {
    const label = classId + ' (' + d.g.day + '/' + d.g.month + ')';
    const pushedN = d.written ? Object.keys(d.merge.toSheet).length - (d.skippedSids || []).length : 0;
    const counts = countStatuses_(d.web);
    out.pushed += pushedN;
    if (d.pulledCount) {
      out.pulled += d.pulledCount;
      out.worked = true;
    }
    out.days.push({ g: d.g, mode: d.mode, pushed: pushedN, pulled: d.pulledCount || 0, counts: counts, conflicts: d.merge.conflicts.length });
    if (d.written) {
      appendAuditLog_(
        masterSs, actor, 'ATTENDANCE_SYNC_SUCCESS', label,
        (ti.isGoogleSheet ? '[Google Sheet] ' : '[.XLSX] ') +
        'Present: ' + counts.P + ', Late: ' + counts.L + (counts.E ? ', Excused: ' + counts.E : '') + ', Absent: ' + counts.A +
        ' (pushed ' + pushedN + ' cell(s)' + (d.mode === 'FIRST' ? ', first sync' : '') + ')'
      );
    }
    if (d.pulledCount) {
      const shown = Object.keys(d.merge.toWeb).slice(0, 10).map(function(sid) { return sid + '=' + d.merge.toWeb[sid]; }).join(', ');
      appendAuditLog_(masterSs, actor, 'SHEET_PULL_APPLIED', label, d.pulledCount + ' cell(s) from Sheet: ' + shown + (d.pulledCount > 10 ? ', ...' : ''));
    }
    if (d.merge.conflicts.length > 0) {
      appendAuditLog_(masterSs, actor, 'SYNC_CONFLICT_SHEET_WINS', label, d.merge.conflicts.slice(0, 10).join(', '));
    }
    if (d.sheetMarks) {
      const unknown = Object.keys(d.sheetMarks).filter(function(sid) { return d.sheetMarks[sid] === 'UNKNOWN'; });
      if (unknown.length > 0) {
        appendAuditLog_(masterSs, actor, 'SYNC_UNKNOWN_MARK', label, 'Unrecognised mark ignored for: ' + unknown.slice(0, 10).join(', '));
      }
    }
  });
  return out;
}

// Sheet unchanged since the stored stamp: each cell is expected to read back as what was last written for its base value
function expectedMarksFromBase_(base, toSheet) {
  const st = getAttendanceStyle_();
  const out = {};
  Object.keys(toSheet || {}).forEach(function(sid) {
    const b = (base || {})[sid];
    if (b === undefined) return;
    const text = b === 'P' ? st.presentText : b === 'L' ? st.lateText : b === 'E' ? st.excusedText : st.absentText;
    const bg = b === 'P' ? st.presentBg : b === 'L' ? st.lateBg : b === 'E' ? st.excusedBg : st.absentBg;
    out[sid] = readDayColumnMarks_([[text]], [[bg || '']], ['X'], st).X;
  });
  return out;
}

function buildAttendanceUndoEntry_(classId, className, targetInfo, result, sessionGroup, prevSig) {
  return {
    classId: classId,
    className: className,
    xlsxFileId: targetInfo.fileId,
    isGoogleSheet: targetInfo.isGoogleSheet,
    sheetName: result.sheetName,
    day: sessionGroup.day,
    month: sessionGroup.month,
    dayKey: sessionGroup.dayKey,
    groupKey: sessionGroup.groupKey,
    previousSyncedSig: prevSig || null,
    previousCells: result.previousCells
  };
}

function syncBaseKey_(classId, dayKey) {
  return classId + '|' + dayKey;
}

function syncPad2_(n) {
  return ('0' + n).slice(-2);
}

function normSyncDayKey_(v) {
  if (v && typeof v.getTime === 'function' && !isNaN(v.getTime())) {
    return v.getFullYear() + '-' + syncPad2_(v.getMonth() + 1) + '-' + syncPad2_(v.getDate());
  }
  return String(v === undefined || v === null ? '' : v).trim();
}

function syncPausedValue_(v) {
  if (v === true) return SYNC_PAUSE_UNDO;
  const n = Number(v);
  return n > 0 ? n : 0;
}

function getSyncBaseSheet_(masterSs, createIfMissing) {
  let sh = masterSs.getSheetByName(SYNC_BASE_SHEET_NAME);
  if (!sh && createIfMissing) {
    sh = masterSs.insertSheet(SYNC_BASE_SHEET_NAME);
    try { sh.getRange('A:F').setNumberFormat('@'); } catch (e) {}
    sh.getRange(1, 1, 1, SYNC_BASE_HEADER.length).setValues([SYNC_BASE_HEADER]);
    try { sh.hideSheet(); } catch (e) {}
  }
  return sh || null;
}

function readSyncBaseAll_(masterSs) {
  const out = { map: {}, order: [] };
  const sh = getSyncBaseSheet_(masterSs, false);
  if (!sh) return out;
  const last = sh.getLastRow();
  if (last < 2) return out;
  const vals = sh.getRange(2, 1, last - 1, SYNC_BASE_HEADER.length).getValues();
  vals.forEach(function(r) {
    const classId = String(r[0] === undefined || r[0] === null ? '' : r[0]).trim();
    const dayKey = normSyncDayKey_(r[1]);
    if (!classId || !dayKey) return;
    const k = syncBaseKey_(classId, dayKey);
    if (out.map[k]) return;
    let base = {};
    try {
      const parsed = JSON.parse(String(r[3] || '{}'));
      if (parsed && typeof parsed === 'object') base = parsed;
    } catch (e) {}
    out.map[k] = {
      classId: classId,
      dayKey: dayKey,
      fileId: String(r[2] || ''),
      base: base,
      sheetModifiedMs: Number(r[4]) || 0,
      styleHash: String(r[5] || ''),
      updatedAt: String(r[6] || ''),
      paused: syncPausedValue_(r[7])
    };
    out.order.push(k);
  });
  return out;
}

/**
 * Batch upsert/remove of base rows (one read, one write). Caller holds the master lock when needed.
 * An upsert is partial: fields that are not given keep their stored value.
 */
function writeSyncBaseBatch_(masterSs, upserts, removeKeys) {
  const cur = readSyncBaseAll_(masterSs);
  const nowIso = new Date().toISOString();
  (upserts || []).forEach(function(e) {
    const dayKey = normSyncDayKey_(e.dayKey);
    const k = syncBaseKey_(e.classId, dayKey);
    const prev = cur.map[k];
    if (!prev) cur.order.push(k);
    const merged = Object.assign({ classId: e.classId, dayKey: dayKey, fileId: '', base: {}, sheetModifiedMs: 0, styleHash: '', paused: 0 }, prev || {}, e);
    merged.dayKey = dayKey;
    merged.updatedAt = nowIso;
    cur.map[k] = merged;
  });
  const rm = {};
  (removeKeys || []).forEach(function(k) { rm[k] = true; });
  const keys = cur.order.filter(function(k) { return cur.map[k] && !rm[k]; });
  const rows = keys.map(function(k) {
    const e = cur.map[k];
    return [e.classId, e.dayKey, e.fileId || '', JSON.stringify(e.base || {}), Number(e.sheetModifiedMs) || 0, e.styleHash || '', e.updatedAt || nowIso, Number(e.paused) || 0];
  });
  const sh = getSyncBaseSheet_(masterSs, true);
  const prevLast = sh.getLastRow();
  if (rows.length > 0) sh.getRange(2, 1, rows.length, SYNC_BASE_HEADER.length).setValues(rows);
  if (prevLast - 1 > rows.length) {
    sh.getRange(2 + rows.length, 1, prevLast - 1 - rows.length, SYNC_BASE_HEADER.length).clearContent();
  }
  return { rows: rows.length };
}

// Used by CANCEL_MISTAKEN_SESSION (the caller already holds the master lock): forget the base of the given days.
function removeSyncBaseRows_(masterSs, classId, dayKeys) {
  if (!getSyncBaseSheet_(masterSs, false)) return 0;
  const cur = readSyncBaseAll_(masterSs);
  const rm = [];
  (dayKeys || []).forEach(function(d) {
    const k = syncBaseKey_(classId, normSyncDayKey_(d));
    if (cur.map[k]) rm.push(k);
  });
  if (rm.length === 0) return 0;
  writeSyncBaseBatch_(masterSs, [], rm);
  return rm.length;
}

function ensureSyncBaseMigrated_(masterSs, props) {
  if (props.getProperty(SYNC_BASE_MIGRATED_KEY)) return { busy: false, migrated: false };
  const res = withMasterWriteLock_({}, SYNC2W_LOCK_WAIT_MS, function() {
    migrateSyncedMapToBase_(masterSs, props);
    return { busy: false, migrated: true };
  });
  return res || { busy: false };
}

// One-time: turn the old per-day status signatures (QR_CENTER_AUTO_SYNCED_SESSIONS) into base rows. SKIPPED_* markers are ignored.
function migrateSyncedMapToBase_(masterSs, props) {
  let map = {};
  try { map = JSON.parse(props.getProperty(AUTO_SYNCED_SESSIONS_KEY) || '{}') || {}; } catch (e) {}
  const upserts = [];
  Object.keys(map).forEach(function(key) {
    const v = map[key];
    if (typeof v !== 'string' || v.indexOf('SKIPPED_') === 0) return;
    const i = key.lastIndexOf('__');
    if (i < 1) return;
    const dm = key.slice(i + 2).match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
    if (!dm) return;
    let parsed = null;
    try { parsed = JSON.parse(v); } catch (e) {}
    if (!parsed || typeof parsed !== 'object') return;
    const base = {};
    Object.keys(parsed).forEach(function(sid) {
      const st = String(parsed[sid]).toUpperCase();
      if (st === 'P' || st === 'L' || st === 'E' || st === '-') base[String(sid).toUpperCase()] = st;
    });
    upserts.push({
      classId: key.slice(0, i),
      dayKey: dm[1] + '-' + syncPad2_(dm[2]) + '-' + syncPad2_(dm[3]),
      fileId: '',
      base: base,
      sheetModifiedMs: 0,
      styleHash: '',
      paused: 0
    });
  });
  if (upserts.length > 0) writeSyncBaseBatch_(masterSs, upserts, []);
  props.setProperty(SYNC_BASE_MIGRATED_KEY, '1');
  return upserts.length;
}

// Past days that have marks on the Sheet but no session at all: create the (fake) session + TEACHER_MANUAL marks once,
// and start their base. Fully blank columns are ignored (blank = present only counts when the column has other marks).
function importPastDaysFromSheet_(masterSs, classId, targetInfo, dateColumns, opts) {
  if (typeof writeManualMarksForDay_ !== 'function') return 0;
  const now = new Date();
  const todayKey = now.getFullYear() + '-' + syncPad2_(now.getMonth() + 1) + '-' + syncPad2_(now.getDate());
  const baseMap = readSyncBaseAll_(masterSs).map;
  const candidates = (dateColumns || []).filter(function(dc) {
    return dc && dc.dayKey && dc.dayKey < todayKey && !baseMap[syncBaseKey_(classId, dc.dayKey)];
  });
  if (candidates.length === 0) return 0;
  const groups = (opts && opts.groups) || getGroupedSessionsByClassDate_(masterSs).filter(function(g) { return g.classId === classId; });
  const hasSession = {};
  groups.forEach(function(g) { hasSession[g.dayKey] = true; });
  const imports = [];
  candidates.forEach(function(dc) {
    if (hasSession[dc.dayKey]) return;
    const marks = {};
    let clear = 0;
    Object.keys(dc.marks || {}).forEach(function(sid) {
      const m = dc.marks[sid];
      if (m === 'BLANK') marks[sid] = 'P';
      else if (m === 'P' || m === 'L' || m === 'E' || m === '-') { marks[sid] = m; clear++; }
    });
    if (clear > 0) imports.push({ dayKey: dc.dayKey, marks: marks });
  });
  if (imports.length === 0) return 0;
  const styleNow = syncStyleKey_();
  const upserts = [];
  const res = withMasterWriteLock_({}, (opts && opts.lockWaitMs) || SYNC2W_LOCK_WAIT_MS, function() {
    imports.forEach(function(it) {
      const wr = writeManualMarksForDay_(masterSs, classId, it.dayKey, it.marks, { lockHeld: true, actor: 'SHEET_IMPORT' });
      if (wr && wr.ok) {
        upserts.push({ classId: classId, dayKey: it.dayKey, fileId: targetInfo.fileId, base: it.marks, sheetModifiedMs: targetInfo.lastUpdatedMs || 0, styleHash: styleNow, paused: 0 });
      }
    });
    if (upserts.length > 0) writeSyncBaseBatch_(masterSs, upserts, []);
    return { busy: false };
  });
  return res && res.busy ? 0 : upserts.length;
}

function getClassNamesMap_(masterSs) {
  const map = {};
  try {
    const cfgSheet = masterSs.getSheetByName('Classes_Config');
    if (!cfgSheet) return map;
    const data = cfgSheet.getDataRange().getValues();
    for (let i = 1; i < data.length; i++) {
      const cid = String(data[i][0] || '').trim();
      const cname = String(data[i][1] || '').trim();
      if (cid) map[cid] = cname;
    }
  } catch (e) {}
  return map;
}

/**
 * Normalize text for fuzzy matching: strips Vietnamese accents, spaces, hyphens, underscores,
 * and converts to uppercase.
 */
function normalizeMatchKey_(str) {
  if (!str) return '';
  return String(str)
    .normalize('NFD')
    .replace(/[\u0300-\u036f]/g, '')
    .replace(/đ/g, 'd')
    .replace(/Đ/g, 'D')
    .toUpperCase()
    .replace(/[^A-Z0-9]/g, '');
}

/**
 * Extract all numeric groups from a string (e.g. "2610AVCB002" -> ["2610", "002"]).
 */
function extractNumericTokens_(str) {
  if (!str) return [];
  const norm = String(str).toUpperCase();
  return norm.match(/\d+/g) || [];
}

/**
 * Score how well `candidateTitle` matches `classId` / `className` with a strict guard
 * on numeric identifiers so class "002" never matches class "003".
 * Returns 0 if incompatible, or a positive score (higher = stronger match).
 */
function scoreClassMatch_(candidateTitle, classId, className) {
  const candNorm = normalizeMatchKey_(candidateTitle);
  const idNorm = normalizeMatchKey_(classId);
  const nameNorm = normalizeMatchKey_(className);
  if (!candNorm || !idNorm) return 0;

  const idNums = extractNumericTokens_(classId);
  const candNums = extractNumericTokens_(candidateTitle);

  // Strict numeric guard: if classId has numbers, the last numeric token (class section/code, e.g. "002")
  // MUST appear in candidateTitle's numeric tokens.
  if (idNums.length > 0) {
    const trailingClassNum = idNums[idNums.length - 1];
    const trailingInt = parseInt(trailingClassNum, 10);
    let hasTrailingMatch = false;
    for (let i = 0; i < candNums.length; i++) {
      if (candNums[i] === trailingClassNum || (trailingClassNum.length >= 2 && parseInt(candNums[i], 10) === trailingInt && candNums[i].length === trailingClassNum.length)) {
        hasTrailingMatch = true;
        break;
      }
    }
    if (!hasTrailingMatch) return 0;

    // If classId also has a leading cohort/semester number (e.g. "2610" in "2610AVCB002")
    // and candidate also has a 4-digit prefix number, they must not conflict (e.g. 2608 vs 2610).
    if (idNums.length >= 2 && idNums[0].length === 4 && candNums.length >= 2 && candNums[0].length === 4) {
      if (idNums[0] !== candNums[0]) return 0;
    }
  }

  // Exact normalized match
  if (candNorm === idNorm) return 100;
  // Candidate contains full normalized classId
  if (candNorm.indexOf(idNorm) !== -1) return 90;
  if (idNorm.indexOf(candNorm) !== -1 && candNorm.length >= 5) return 80;

  // Match core letters + class number (e.g. "AVCB002" inside "2610AVCB002" vs "Lớp AVCB - 002")
  const alphaTokens = String(classId).toUpperCase().match(/[A-Z]{2,}/g) || [];
  if (alphaTokens.length > 0 && idNums.length > 0) {
    const coreCode = alphaTokens[alphaTokens.length - 1] + idNums[idNums.length - 1];
    if (candNorm.indexOf(coreCode) !== -1) return 75;
  }

  // Match against className if provided
  if (nameNorm && nameNorm.length >= 4) {
    if (candNorm === nameNorm) return 70;
    if (candNorm.indexOf(nameNorm) !== -1) return 65;
  }

  return 0;
}

/**
 * Retrieve up to 3 levels of parent folder names on Google Drive so teachers can
 * easily verify where the matched file lives (e.g. "My Drive / HK1_2026 / Điểm danh").
 */
function getDriveFileFolderPath_(file) {
  try {
    const parents = file.getParents();
    if (!parents || !parents.hasNext()) {
      return 'Shared with me / Root';
    }
    const parts = [];
    let current = parents.next();
    let depth = 0;
    while (current && depth < 3) {
      const folderName = String(current.getName() || '').trim();
      if (folderName) {
        parts.unshift(folderName);
      }
      const nextParents = current.getParents();
      if (nextParents && nextParents.hasNext()) {
        current = nextParents.next();
        depth++;
        if (depth >= 3) {
          parts.unshift('...');
        }
      } else {
        break;
      }
    }
    return parts.length > 0 ? parts.join(' / ') : 'My Drive';
  } catch (e) {
    return 'Google Drive';
  }
}

/**
 * Inspect Drive File metadata and return { fileId, fileName, mimeType, isGoogleSheet, folderPath, fileUrl, lastUpdatedMs } if valid.
 * Automatically resolves Google Drive Shortcuts (`application/vnd.google-apps.shortcut`) to their target file.
 */
function inspectDriveFileById_(fileId, masterFileId, fallbackFolderPath) {
  if (!fileId || fileId === masterFileId) return null;
  try {
    const file = DriveApp.getFileById(fileId);
    if (file.isTrashed()) return null;
    const fname = file.getName() || '';
    if (fname.indexOf('_TEMP_') === 0) return null;
    const mime = file.getMimeType();
    if (mime === MIME_DRIVE_SHORTCUT) {
      const targetId = file.getTargetId ? file.getTargetId() : null;
      if (!targetId || targetId === fileId) return null;
      const shortcutFolder = fallbackFolderPath || getDriveFileFolderPath_(file);
      return inspectDriveFileById_(targetId, masterFileId, shortcutFolder);
    }
    if (mime !== MIME_GOOGLE_SHEET && mime !== MIME_EXCEL_XLSX) return null;
    const lastUpdated = file.getLastUpdated ? file.getLastUpdated() : null;
    return {
      fileId: file.getId(),
      fileName: fname,
      mimeType: mime,
      isGoogleSheet: mime === MIME_GOOGLE_SHEET,
      folderPath: fallbackFolderPath || getDriveFileFolderPath_(file),
      fileUrl: file.getUrl() || '',
      lastUpdatedMs: lastUpdated ? lastUpdated.getTime() : 0
    };
  } catch (e) {
    return null;
  }
}

/**
 * Check whether a specific Sheet tab actually contains a student roster (MSSV + Full Name rows).
 * Used to distinguish real Attendance/Roster files from Syllabus/Class Diary sheets that share the same Class_ID.
 */
function probeSheetHasStudentRoster_(sheet) {
  if (!sheet) return false;
  try {
    const lastRow = sheet.getLastRow();
    const lastCol = sheet.getLastColumn();
    if (lastRow < 2 || lastCol < 2) return false;
    const layout = detectSheetLayout_(sheet);
    if (lastRow < layout.dataStartRow) return false;
    const scanRows = Math.min(lastRow - layout.dataStartRow + 1, 20);
    const maxColNeeded = Math.max(layout.idCol, layout.nameCol + (layout.isTwoColName ? 1 : 0));
    if (maxColNeeded > lastCol) return false;
    const sample = sheet.getRange(layout.dataStartRow, 1, scanRows, maxColNeeded).getValues();
    for (let i = 0; i < sample.length; i++) {
      if (extractRowStudentIdAndName_(sample[i], layout).sid) {
        return true;
      }
    }
  } catch (e) {}
  return false;
}

/**
 * Check whether a candidate Drive file (Google Sheet or .XLSX) contains a valid student roster (MSSV + Full Name).
 */
function probeFileHasStudentRoster_(candInfo, classId, className) {
  if (!candInfo) return false;
  try {
    return withTargetSpreadsheet_(candInfo, classId, className, false, function(ss, sheet) {
      return probeSheetHasStudentRoster_(sheet);
    });
  } catch (e) {
    return false;
  }
}

/**
 * Locate the target Google Sheet or .XLSX file on Google Drive for a given class.
 * Universally matches by Class_ID (`Mã lớp`) and verifies student roster structure (`MSSV` + `Họ tên`)
 * when multiple files in a class folder share the same Class_ID (e.g. Attendance file vs. Class Diary file).
 */
function resolveClassTargetFile_(masterSs, classId, className, bypassCache) {
  if (!classId) return null;
  const masterFileId = masterSs ? masterSs.getId() : '';
  const docsHost = ['docs', 'google', 'com'].join('.');
  const driveHost = ['drive', 'google', 'com'].join('.');

  // 1. Check custom link/file ID in Classes_Config row (column J or any column containing a Drive link)
  try {
    const cfgSheet = masterSs.getSheetByName('Classes_Config');
    if (cfgSheet) {
      const cfg = cfgSheet.getDataRange().getValues();
      for (let i = 1; i < cfg.length; i++) {
        if (String(cfg[i][0] || '').trim() === classId) {
          for (let c = 2; c < cfg[i].length; c++) {
            const cellStr = String(cfg[i][c] || '').trim();
            if (!cellStr) continue;
            if (cellStr.indexOf(docsHost) !== -1 || cellStr.indexOf(driveHost) !== -1 || c === 9) {
              const m = cellStr.match(/[-\w]{25,}/);
              if (m) {
                const info = inspectDriveFileById_(m[0], masterFileId);
                if (info) return info;
              }
            }
          }
          break;
        }
      }
    }
  } catch (e) {}

  // 2. Check optional manual override map in code
  if (CUSTOM_CLASS_FILE_IDS[classId]) {
    const info = inspectDriveFileById_(CUSTOM_CLASS_FILE_IDS[classId], masterFileId);
    if (info) return info;
  }

  // 3. Check cached file ID (V3 cache stores verified roster files)
  const props = PropertiesService.getScriptProperties();
  const cacheMap = JSON.parse(props.getProperty(FILE_ID_CACHE_PROP_KEY) || '{}');
  if (!bypassCache && cacheMap[classId]) {
    const info = inspectDriveFileById_(cacheMap[classId], masterFileId);
    if (info) return info;
  }

  // 4. Search Google Drive for matching Google Sheets, .XLSX files, or Drive Shortcuts
  const searchTerms = [classId];
  const alphaTokens = String(classId).toUpperCase().match(/[A-Z]{2,}/g) || [];
  const numTokens = extractNumericTokens_(classId);
  if (alphaTokens.length > 0 && numTokens.length > 0) {
    const coreTerm = alphaTokens[alphaTokens.length - 1] + numTokens[numTokens.length - 1];
    if (coreTerm !== classId && searchTerms.indexOf(coreTerm) === -1) {
      searchTerms.push(coreTerm);
    }
    if (searchTerms.indexOf(alphaTokens[alphaTokens.length - 1]) === -1) {
      searchTerms.push(alphaTokens[alphaTokens.length - 1]);
    }
  }

  const candidates = [];
  const seenFileIds = {};

  for (let t = 0; t < searchTerms.length; t++) {
    const term = searchTerms[t].replace(/'/g, "\\'");
    const query =
      "title contains '" + term + "' and " +
      "(mimeType = '" + MIME_GOOGLE_SHEET + "' or mimeType = '" + MIME_EXCEL_XLSX + "' or mimeType = '" + MIME_DRIVE_SHORTCUT + "') and " +
      "trashed = false";

    try {
      const files = DriveApp.searchFiles(query);
      while (files.hasNext()) {
        const rawFile = files.next();
        const rawId = rawFile.getId();
        if (rawId === masterFileId) continue;

        const rawName = rawFile.getName() || '';
        if (rawName.indexOf('_TEMP_') === 0) continue;

        const score = scoreClassMatch_(rawName, classId, className);
        if (score <= 0) continue;

        const info = inspectDriveFileById_(rawId, masterFileId);
        if (!info || info.fileId === masterFileId) continue;

        const finalScore = Math.max(score, scoreClassMatch_(info.fileName, classId, className));
        if (seenFileIds[info.fileId]) {
          if (finalScore > seenFileIds[info.fileId].score) {
            seenFileIds[info.fileId].score = finalScore;
          }
          continue;
        }

        const candObj = { info: info, score: finalScore };
        seenFileIds[info.fileId] = candObj;
        candidates.push(candObj);
      }
    } catch (e) {}

    // If we found strong matches on the exact classId term, no need to search broader terms
    const hasStrongMatch = candidates.some(function(c) { return c.score >= 90; });
    if (hasStrongMatch) break;
  }

  if (candidates.length === 0) return null;

  // Sort candidates by score descending
  candidates.sort(function(a, b) {
    if (b.score !== a.score) return b.score - a.score;
    return (b.info.lastUpdatedMs || 0) - (a.info.lastUpdatedMs || 0);
  });

  let chosenInfo = null;

  if (candidates.length === 1) {
    chosenInfo = candidates[0].info;
  } else {
    // Disambiguate multiple files matching the same Class_ID (e.g. Syllabus/Diary vs. Attendance Roster):
    // Probe native Google Sheets first (fast ~80ms check without .XLSX conversion) to eliminate non-roster files!
    const topScore = candidates[0].score;
    const topTier = candidates.filter(function(c) { return c.score >= Math.max(65, topScore - 15); });
    const pool = topTier.length > 0 ? topTier : candidates;

    const gsheetCands = pool.filter(function(c) { return c.info.isGoogleSheet; });
    const xlsxCands = pool.filter(function(c) { return !c.info.isGoogleSheet; });

    for (let i = 0; i < gsheetCands.length; i++) {
      if (probeFileHasStudentRoster_(gsheetCands[i].info, classId, className)) {
        chosenInfo = gsheetCands[i].info;
        break;
      }
    }

    if (!chosenInfo) {
      if (xlsxCands.length === 1) {
        // All Google Sheet candidates had 0 students (e.g. Syllabus/Class Diary), so the single .XLSX candidate wins immediately!
        chosenInfo = xlsxCands[0].info;
      } else if (xlsxCands.length > 1) {
        for (let i = 0; i < xlsxCands.length; i++) {
          if (probeFileHasStudentRoster_(xlsxCands[i].info, classId, className)) {
            chosenInfo = xlsxCands[i].info;
            break;
          }
        }
      }
    }

    if (!chosenInfo) {
      chosenInfo = candidates[0].info;
    }
  }

  if (chosenInfo) {
    cacheMap[classId] = chosenInfo.fileId;
    try {
      props.setProperty(FILE_ID_CACHE_PROP_KEY, JSON.stringify(cacheMap));
    } catch (e) {}
    return chosenInfo;
  }

  return null;
}

/**
 * Find the best matching sheet/tab inside a Spreadsheet for `classId` / `className`.
 * Supports both multi-class workbooks (one tab per class) and single-class files,
 * prioritizing tabs that actually contain student roster rows (`MSSV` + `Họ tên`).
 */
function findBestMatchingSheet_(ss, classId, className) {
  const sheets = ss.getSheets();
  if (!sheets || sheets.length === 0) throw new Error('No sheets found in target file.');

  // 1. Try matching tab name to classId / className
  let bestSheet = null;
  let bestScore = 0;
  for (let i = 0; i < sheets.length; i++) {
    const sName = sheets[i].getName();
    if (sName === UNDO_BACKUP_SHEET_NAME || sName.indexOf('_TEMP_') === 0) continue;
    const score = scoreClassMatch_(sName, classId, className);
    if (score > bestScore) {
      bestScore = score;
      bestSheet = sheets[i];
    }
  }
  if (bestSheet && bestScore >= 65 && probeSheetHasStudentRoster_(bestSheet)) {
    return bestSheet;
  }

  // 2. Check standard Attendance sheet tab names that contain student rows
  const standardNames = ['Danh sách mới', 'Attendance sheet', 'Danh sách', 'Danh sach', 'Điểm danh', 'Diem danh'];
  for (let i = 0; i < standardNames.length; i++) {
    const candidate = ss.getSheetByName(standardNames[i]);
    if (candidate && probeSheetHasStudentRoster_(candidate)) return candidate;
  }

  // 3. Check any visible sheet that actually contains a student roster (MSSV + Full Name)
  for (let i = 0; i < sheets.length; i++) {
    const sName = sheets[i].getName();
    if (sName === UNDO_BACKUP_SHEET_NAME || sName.indexOf('_TEMP_') === 0) continue;
    if (!sheets[i].isSheetHidden() && probeSheetHasStudentRoster_(sheets[i])) {
      return sheets[i];
    }
  }

  // 4. Fallback if workbook is an empty template (0 students currently)
  if (bestSheet && bestScore >= 65) return bestSheet;
  for (let i = 0; i < standardNames.length; i++) {
    const candidate = ss.getSheetByName(standardNames[i]);
    if (candidate) return candidate;
  }
  for (let i = 0; i < sheets.length; i++) {
    if (!sheets[i].isSheetHidden()) return sheets[i];
  }
  return sheets[0];
}

/**
 * Pure character-based date string parser (handles "28/10", "28-10", "28/10/2026", "2026-10-28").
 * Avoids regex slashes/dots so text/URL sanitizers never alter the code.
 */
function parseDateString_(s) {
  const str = String(s || '').trim();
  if (!str || str.length > 12) return null;

  const parts = [];
  let cur = '';
  for (let i = 0; i < str.length; i++) {
    const ch = str.charAt(i);
    if (ch === '/' || ch === '-' || ch === '.') {
      parts.push(cur);
      cur = '';
    } else if (ch >= '0' && ch <= '9') {
      cur += ch;
    } else {
      return null;
    }
  }
  parts.push(cur);

  if (parts.length !== 2 && parts.length !== 3) return null;
  if (!parts[0] || !parts[1]) return null;

  const p0 = parseInt(parts[0], 10);
  const p1 = parseInt(parts[1], 10);

  // YYYY-MM-DD format
  if (parts.length === 3 && parts[0].length === 4) {
    if (!parts[2]) return null;
    const p2 = parseInt(parts[2], 10);
    if (p1 >= 1 && p1 <= 12 && p2 >= 1 && p2 <= 31) {
      return { year: p0, month: p1, day: p2 };
    }
    return null;
  }

  // DD/MM or DD/MM/YYYY format
  if (p0 >= 1 && p0 <= 31 && p1 >= 1 && p1 <= 12) {
    let yr = null;
    if (parts.length === 3) {
      if (!parts[2] || parts[2].length < 2) return null;
      yr = parseInt(parts[2], 10);
      if (yr < 100) yr += 2000;
    }
    return { day: p0, month: p1, year: yr };
  }

  return null;
}

/**
 * Validate Student_ID:
 * - Accepts pure numeric IDs (e.g. "24105921") AND alphanumeric IDs (e.g. "SV001", "B21DCCN015", "AVCB-02").
 * - Length 3..20 chars, only letters/digits/dot/underscore/hyphen, and MUST contain at least 2 digits
 *   so footer words ("Tổng cộng", "Giáo viên", "Ghi chú") and 1-digit row numbers are ignored.
 * - If `optionalNameVal` is passed, also ensures the name cell is non-empty and not a footer label.
 */
// Roster-import ID rule. Defaults keep the old behaviour; a lecturer can override it with the
// Script Property STUDENT_ID_RULE_V1 = {"minLen":3,"maxLen":20,"minDigits":2} (e.g. minDigits 0 for letter-only IDs).
function getStudentIdRule_() {
  const def = { minLen: 3, maxLen: 20, minDigits: 2 };
  try {
    const raw = PropertiesService.getScriptProperties().getProperty('STUDENT_ID_RULE_V1');
    const o = raw ? JSON.parse(raw) : {};
    const num = (v, lo, hi, d) => { const n = Number(v); return (isFinite(n) && n >= lo && n <= hi) ? Math.round(n) : d; };
    const minLen = num(o.minLen, 1, 40, def.minLen);
    return { minLen: minLen, maxLen: Math.max(minLen, num(o.maxLen, 1, 40, def.maxLen)), minDigits: num(o.minDigits, 0, 10, def.minDigits) };
  } catch (e) {
    return def;
  }
}

function normalizeStudentId_(val, optionalNameVal) {
  if (val === null || val === undefined || val === '') return '';
  let s = String(val).trim();
  if (s.endsWith('.0')) s = s.slice(0, -2);
  const idRule = getStudentIdRule_();
  if (s.length < idRule.minLen || s.length > idRule.maxLen) return '';

  // Must start and end with alphanumeric and contain only alphanumeric, '-', '_', '.' (single-character IDs allowed only when minLen is 1)
  if (!/^[A-Za-z0-9]([A-Za-z0-9._-]*[A-Za-z0-9])?$/.test(s)) return '';

  // Must contain at least minDigits digits (default 2) to prevent matching normal words or single digits
  const digitMatches = s.match(/\d/g);
  if ((digitMatches ? digitMatches.length : 0) < idRule.minDigits) return '';

  // Reject date-like strings (e.g. "28-10-2026" or "28/10")
  if (parseDateString_(s) !== null) return '';

  if (optionalNameVal !== undefined) {
    const nameStr = String(optionalNameVal || '').trim();
    if (!nameStr) return '';
    const nameKey = normalizeMatchKey_(nameStr);
    if (
      nameKey.indexOf('TONGCONG') === 0 ||
      nameKey.indexOf('GIAOVIEN') === 0 ||
      nameKey.indexOf('GIANGVIEN') === 0 ||
      nameKey.indexOf('CHUKY') === 0 ||
      nameKey.indexOf('GHICHU') === 0
    ) {
      return '';
    }
  }

  return typeof normalizeId === 'function' ? normalizeId(s) : String(s).trim().toUpperCase();
}

/**
 * Check if a cell value/display string represents a valid attendance date header (Date object or dd/mm).
 */
function parseDateCell_(rawVal, dispVal) {
  if (rawVal instanceof Date && !isNaN(rawVal.getTime())) {
    return { day: rawVal.getDate(), month: rawVal.getMonth() + 1, year: rawVal.getFullYear() };
  }
  const parsed = parseDateString_(dispVal);
  if (parsed) {
    return { day: parsed.day, month: parsed.month, year: parsed.year || null };
  }
  return null;
}

/**
 * Automatically scan the top 20 rows of `sheet` to detect:
 *   - idCol (1-indexed column of Student_ID / MSSV)
 *   - nameCol (1-indexed column of Full_Name)
 *   - isTwoColName (true if First/Last name spans nameCol and nameCol + 1)
 *   - dateRow (1-indexed row containing session dates)
 *   - dateStartCol (1-indexed first column of attendance dates)
 *   - dataStartRow (1-indexed first row of student records)
 * Falls back gracefully to the standard template (Row 9 dates, Row 10+ students, Col B ID, Col C-D Name).
 */
function detectSheetLayout_(sheet) {
  const maxScanRows = Math.min(sheet.getLastRow(), 20);
  const maxScanCols = Math.min(sheet.getLastColumn(), 100);

  if (maxScanRows < 2 || maxScanCols < 2) {
    return {
      idCol: 2,
      nameCol: 3,
      isTwoColName: true,
      dateRow: 9,
      dateStartCol: 5,
      dataStartRow: 10
    };
  }

  const rawGrid = sheet.getRange(1, 1, maxScanRows, maxScanCols).getValues();
  const dispGrid = sheet.getRange(1, 1, maxScanRows, maxScanCols).getDisplayValues();

  const idKeywords = ['MSSV', 'MSHV', 'MASV', 'MAHV', 'MASO', 'MASINHVIEN', 'MAHOCVIEN', 'STUDENTID', 'LEARNERID', 'IDSV', 'IDHV', 'ID'];
  const nameKeywords = ['HOVATEN', 'HOTEN', 'TENSV', 'TENHV', 'TENHOCVIEN', 'TENSINHVIEN', 'FULLNAME', 'STUDENTNAME', 'LEARNERNAME', 'NAME'];

  let detectedHeaderRowIdx = -1; // 0-indexed
  let detectedIdColIdx = -1;     // 0-indexed
  let detectedNameColIdx = -1;   // 0-indexed

  for (let r = 0; r < Math.min(maxScanRows, 16); r++) {
    let rowIdCol = -1;
    let rowNameCol = -1;

    for (let c = 0; c < Math.min(maxScanCols, 10); c++) {
      const cellKey = normalizeMatchKey_(dispGrid[r][c]);
      if (!cellKey) continue;

      if (rowIdCol === -1) {
        for (let k = 0; k < idKeywords.length; k++) {
          const kw = idKeywords[k];
          if (cellKey === kw || (kw !== 'ID' && cellKey.indexOf(kw) !== -1)) {
            rowIdCol = c;
            break;
          }
        }
      }

      if (rowNameCol === -1 && c !== rowIdCol) {
        for (let k = 0; k < nameKeywords.length; k++) {
          const kw = nameKeywords[k];
          if (cellKey === kw || (kw !== 'NAME' && cellKey.indexOf(kw) !== -1)) {
            rowNameCol = c;
            break;
          }
        }
      }
    }

    if (rowIdCol !== -1) {
      detectedHeaderRowIdx = r;
      detectedIdColIdx = rowIdCol;
      detectedNameColIdx = rowNameCol !== -1 ? rowNameCol : (rowIdCol + 1);
      break;
    }
  }

  // Fallback to standard template coordinates if no explicit header row was found
  if (detectedHeaderRowIdx === -1) {
    const e9Date = (maxScanRows >= 9 && maxScanCols >= 5)
      ? parseDateCell_(rawGrid[8][4], dispGrid[8][4])
      : null;
    return {
      idCol: 2,
      nameCol: 3,
      isTwoColName: Boolean(e9Date),
      dateRow: 9,
      dateStartCol: 5,
      dataStartRow: 10
    };
  }

  // Determine whether Full Name is split across 2 columns (e.g. Col C + Col D merged header)
  let isTwoColName = false;
  const nextColIdx = detectedNameColIdx + 1;
  if (nextColIdx < maxScanCols) {
    const nextHeaderKey = normalizeMatchKey_(dispGrid[detectedHeaderRowIdx][nextColIdx]);
    const nextHeaderIsDate = Boolean(parseDateCell_(rawGrid[detectedHeaderRowIdx][nextColIdx], dispGrid[detectedHeaderRowIdx][nextColIdx]));

    if (!nextHeaderIsDate) {
      if (nextHeaderKey === 'TEN' || nextHeaderKey === 'FIRSTNAME') {
        isTwoColName = true;
      } else if (nextHeaderKey === '') {
        // Header cell is blank (common when "Họ và tên" is merged across 2 columns).
        // Check if rows below nextColIdx contain name words rather than attendance marks or numbers.
        let sampleNameCells = 0;
        for (let rr = detectedHeaderRowIdx + 1; rr < Math.min(maxScanRows, detectedHeaderRowIdx + 8); rr++) {
          const sampleVal = String(dispGrid[rr][nextColIdx] || '').trim();
          if (sampleVal.length >= 2 && !/^(P|L|A|X|V|-|\d+)$/i.test(sampleVal) && !parseDateCell_(rawGrid[rr][nextColIdx], sampleVal)) {
            sampleNameCells++;
          }
        }
        if (sampleNameCells >= 1) {
          isTwoColName = true;
        }
      }
    }
  }

  const dateStartColIdx = isTwoColName ? (detectedNameColIdx + 2) : (detectedNameColIdx + 1);

  // Detect which row near detectedHeaderRowIdx holds the session dates (supports 1-tier or 2-tier headers)
  let bestDateRowIdx = detectedHeaderRowIdx;
  let maxDateMatches = 0;
  const candidateRows = [detectedHeaderRowIdx, detectedHeaderRowIdx + 1, detectedHeaderRowIdx - 1];

  for (let i = 0; i < candidateRows.length; i++) {
    const rIdx = candidateRows[i];
    if (rIdx < 0 || rIdx >= maxScanRows) continue;
    let dateCount = 0;
    for (let c = dateStartColIdx; c < maxScanCols; c++) {
      if (parseDateCell_(rawGrid[rIdx][c], dispGrid[rIdx][c])) {
        dateCount++;
      }
    }
    if (dateCount > maxDateMatches) {
      maxDateMatches = dateCount;
      bestDateRowIdx = rIdx;
    }
  }

  // Find the first actual student data row below the header/date rows
  let dataStartRowIdx = Math.max(detectedHeaderRowIdx, bestDateRowIdx) + 1;
  for (let r = dataStartRowIdx; r < maxScanRows; r++) {
    const candidateId = dispGrid[r][detectedIdColIdx];
    const candidateName = dispGrid[r][detectedNameColIdx];
    if (normalizeStudentId_(candidateId, candidateName)) {
      dataStartRowIdx = r;
      break;
    }
  }

  return {
    idCol: detectedIdColIdx + 1,
    nameCol: detectedNameColIdx + 1,
    isTwoColName: isTwoColName,
    dateRow: bestDateRowIdx + 1,
    dateStartCol: dateStartColIdx + 1,
    dataStartRow: dataStartRowIdx + 1
  };
}

/**
 * Construct API URLs at runtime without literal domain strings in source code.
 */
function buildApiUrl_(hostParts, path) {
  return 'https' + '://' + hostParts.join('.') + path;
}

/**
 * Helper that opens either a native Google Sheet directly or a temporary converted
 * Google Sheet for an .XLSX file, runs `callback(spreadsheet, sheet)`, and (if `isWrite`
 * is true on an .XLSX file) exports and patches the original .XLSX file before cleaning up.
 */
function withTargetSpreadsheet_(targetInfo, classId, className, isWrite, callback) {
  if (targetInfo.isGoogleSheet) {
    const ss = SpreadsheetApp.openById(targetInfo.fileId);
    const sheet = findBestMatchingSheet_(ss, classId, className);
    const res = callback(ss, sheet);
    if (isWrite) {
      SpreadsheetApp.flush();
      if (res && typeof res === 'object') {
        try { res.afterModifiedMs = DriveApp.getFileById(targetInfo.fileId).getLastUpdated().getTime(); } catch (e) {}
      }
    }
    return res;
  }

  // .XLSX workflow via Drive API v3 temporary conversion
  const token = ScriptApp.getOAuthToken();
  const getModifiedTime_ = () => {
    const r = UrlFetchApp.fetch(buildApiUrl_(['www', 'googleapis', 'com'], '/drive/v3/files/' + targetInfo.fileId + '?fields=modifiedTime&supportsAllDrives=true'), {
      headers: { Authorization: 'Bearer ' + token },
      muteHttpExceptions: true
    });
    if (r.getResponseCode() >= 300) return '';
    try { return String(JSON.parse(r.getContentText()).modifiedTime || ''); } catch (e) { return ''; }
  };
  const modifiedBefore = isWrite ? getModifiedTime_() : '';
  const copyUrl = buildApiUrl_(['www', 'googleapis', 'com'], '/drive/v3/files/' + targetInfo.fileId + '/copy?supportsAllDrives=true');
  const copyRes = UrlFetchApp.fetch(copyUrl, {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + token },
    payload: JSON.stringify({
      name: '_TEMP_' + (isWrite ? 'SYNC_' : 'ROSTER_') + classId + '_' + new Date().getTime(),
      mimeType: MIME_GOOGLE_SHEET
    }),
    muteHttpExceptions: true
  });

  if (copyRes.getResponseCode() >= 300) {
    throw new Error('Cannot open .XLSX file (HTTP ' + copyRes.getResponseCode() + '): ' + copyRes.getContentText());
  }

  const tempFileId = JSON.parse(copyRes.getContentText()).id;
  try {
    const tempSs = SpreadsheetApp.openById(tempFileId);
    const sheet = findBestMatchingSheet_(tempSs, classId, className);
    const res = callback(tempSs, sheet);

    if (isWrite) {
      SpreadsheetApp.flush();

      const exportUrl = buildApiUrl_(['docs', 'google', 'com'], '/spreadsheets/d/' + tempFileId + '/export?format=xlsx');
      const exportRes = UrlFetchApp.fetch(exportUrl, {
        headers: { Authorization: 'Bearer ' + token },
        muteHttpExceptions: true
      });

      if (exportRes.getResponseCode() >= 300) {
        throw new Error('Failed to export .XLSX: ' + exportRes.getContentText());
      }

      // M11: never overwrite an .XLSX that the lecturer edited while this sync was converting it.
      const modifiedNow = getModifiedTime_();
      if (!modifiedBefore || !modifiedNow) {
        throw new Error('XLSX_VERSION_UNKNOWN: could not read the target .XLSX modified time, so nothing was overwritten; try again.');
      }
      if (modifiedBefore !== modifiedNow) {
        throw new Error('XLSX_MODIFIED_DURING_SYNC: the target .XLSX changed while syncing; nothing was overwritten, try again.');
      }

      const updateUrl = buildApiUrl_(['www', 'googleapis', 'com'], '/upload/drive/v3/files/' + targetInfo.fileId + '?uploadType=media&supportsAllDrives=true');
      const updateRes = UrlFetchApp.fetch(updateUrl, {
        method: 'patch',
        contentType: MIME_EXCEL_XLSX,
        headers: { Authorization: 'Bearer ' + token },
        payload: exportRes.getBlob().getBytes(),
        muteHttpExceptions: true
      });

      if (updateRes.getResponseCode() >= 300) {
        throw new Error('Failed to update target .XLSX file: ' + updateRes.getContentText());
      }
      if (res && typeof res === 'object') {
        const afterMs = Date.parse(getModifiedTime_());
        if (afterMs) res.afterModifiedMs = afterMs;
      }
    }

    return res;
  } finally {
    try {
      DriveApp.getFileById(tempFileId).setTrashed(true);
    } catch (cleanupErr) {}
  }
}

/**
 * Smart Color-Family Classifier for Google Sheets & Excel default palettes:
 * Recognizes all shades of:
 *   - 'ORANGE' (Excused E): #ff9900, #f6b26b, #f9cb9c, #fce5cd, #e69138, #b45f06, #ff6d01, #ffa500, #ffc000, #ed7d31, etc. (Hue 17°..46°)
 *   - 'YELLOW' (Late L):    #ffff00, #fff2cc, #ffe599, #ffd966, #f1c232, #bf9000, #ffeb3b, etc. (Hue 47°..69°)
 *   - 'RED'    (Exam Ban):  #ff0000, #e06666, #ea9999, #f4cccc, #cc0000, #990000, #d93025, #ea4335, etc. (Hue <= 16° or >= 342°)
 *   - 'EMPTY':  white / transparent / neutral gray
 */
function classifySheetCellColor_(hexColor) {
  if (!hexColor) return 'EMPTY';
  let h = String(hexColor).trim().toLowerCase();
  if (!h || h === 'none' || h === 'transparent' || h === '#ffffff' || h === '#00000000' || h === '00000000' || h === 'ffffffff') {
    return 'EMPTY';
  }
  if (h.charAt(0) === '#') h = h.slice(1);
  if (h.length === 8) h = h.slice(2); // Strip ARGB alpha prefix if present
  if (h.length === 3) {
    h = h.charAt(0) + h.charAt(0) + h.charAt(1) + h.charAt(1) + h.charAt(2) + h.charAt(2);
  }
  if (h.length !== 6) return 'EMPTY';

  // Fast exact palette lookup for Google Sheets & Excel standard swatches
  const orangeSwatches = {
    'ff9900': 1, 'f6b26b': 1, 'f9cb9c': 1, 'fce5cd': 1, 'e69138': 1, 'b45f06': 1,
    'ff6d01': 1, 'ffa500': 1, 'ffc000': 1, 'ed7d31': 1, 'f4b084': 1, 'f8cbad': 1,
    'fbe5d6': 1, 'fb8c00': 1, 'f57c00': 1, 'ffab40': 1, 'ffd180': 1, 'ff9800': 1,
    'ffb74d': 1, 'ffe0b2': 1, 'fff3e0': 1, 'f97316': 1, 'fdba74': 1, 'fed7aa': 1
  };
  if (orangeSwatches[h]) return 'ORANGE';

  const yellowSwatches = {
    'ffff00': 1, 'fff2cc': 1, 'ffe599': 1, 'ffd966': 1, 'f1c232': 1, 'bf9000': 1,
    'ffeb3b': 1, 'fff59d': 1, 'ffee58': 1, 'fdd835': 1, 'fbc02d': 1, 'fef08a': 1,
    'fde047': 1, 'eab308': 1, 'fef9c3': 1
  };
  if (yellowSwatches[h]) return 'YELLOW';

  const redSwatches = {
    'ff0000': 1, 'e06666': 1, 'ea9999': 1, 'f4cccc': 1, 'cc0000': 1, '990000': 1,
    'd93025': 1, 'ea4335': 1, 'ef4444': 1, 'dc2626': 1, 'b91c1c': 1, 'f87171': 1,
    'fca5a5': 1, 'fee2e2': 1, 'fecaca': 1, 'c00000': 1, 'ff5252': 1
  };
  if (redSwatches[h]) return 'RED';

  const r = parseInt(h.slice(0, 2), 16) / 255;
  const g = parseInt(h.slice(2, 4), 16) / 255;
  const b = parseInt(h.slice(4, 6), 16) / 255;
  if (isNaN(r) || isNaN(g) || isNaN(b)) return 'EMPTY';

  const max = Math.max(r, g, b);
  const min = Math.min(r, g, b);
  const d = max - min;
  const sat = max === 0 ? 0 : d / max;
  const val = max;

  if (sat < 0.14 || val < 0.25) return 'EMPTY';

  let hue = 0;
  if (d !== 0) {
    if (max === r) {
      hue = ((g - b) / d) % 6;
    } else if (max === g) {
      hue = (b - r) / d + 2;
    } else {
      hue = (r - g) / d + 4;
    }
    hue = Math.round(hue * 60);
    if (hue < 0) hue += 360;
  }

  if (hue <= 16 || hue >= 342) return 'RED';
  if (hue >= 17 && hue <= 46) return 'ORANGE';
  if (hue >= 47 && hue <= 69) return 'YELLOW';
  return 'OTHER';
}

function saveManualExamBanForClass_(classId, bannedStudentMap) {
  if (!classId) return;
  try {
    const props = PropertiesService.getScriptProperties();
    const allBans = JSON.parse(props.getProperty('QR_MANUAL_EXAM_BAN_MAP') || '{}');
    allBans[classId] = bannedStudentMap || {};
    props.setProperty('QR_MANUAL_EXAM_BAN_MAP', JSON.stringify(allBans));
  } catch (e) {}
}

function extractRowStudentIdAndName_(row, layout) {
  const rawId = row[layout.idCol - 1];
  const namePart1 = String(row[layout.nameCol - 1] || '').trim();
  const namePart2 = layout.isTwoColName ? String(row[layout.nameCol] || '').trim() : '';
  const fullName = (layout.isTwoColName ? (namePart1 + ' ' + namePart2) : namePart1)
    .replace(/\s+/g, ' ')
    .trim();
  const sid = normalizeStudentId_(rawId, fullName);
  return { sid: sid, fullName: fullName };
}

function normalizeHexKey_(v) {
  let h = String(v === undefined || v === null ? '' : v).trim().toLowerCase();
  if (h.charAt(0) === '#') h = h.slice(1);
  if (h.length === 8) h = h.slice(2);
  if (h.length === 3) h = h.charAt(0) + h.charAt(0) + h.charAt(1) + h.charAt(1) + h.charAt(2) + h.charAt(2);
  return /^[0-9a-f]{6}$/.test(h) && h !== 'ffffff' ? h : '';
}

// Reads a date-column cell back using the lecturer's own Attendance Mark Settings (text and background
// they chose), so CUSTOM symbols/colors round-trip. Returns 'E' | 'L' | 'A_MANUAL' | 'P', or '' when
// the cell does not match any configured mark (the caller then falls back to the built-in recognizers).
function classifyMarkFromStyle_(cellUpper, bgRaw, style) {
  const st = style || {};
  const t = String(cellUpper || '').trim().toUpperCase();
  const eq = v => { const s = String(v === undefined || v === null ? '' : v).trim().toUpperCase(); return s !== '' && s === t; };
  if (eq(st.excusedText)) return 'E';
  if (eq(st.lateText)) return 'L';
  if (eq(st.absentText)) return 'A_MANUAL';
  if (eq(st.presentText)) return 'P';
  const bg = normalizeHexKey_(bgRaw);
  if (bg) {
    const same = v => { const k = normalizeHexKey_(v); return k !== '' && k === bg; };
    if (same(st.excusedBg)) return 'E';
    if (same(st.lateBg)) return 'L';
    if (same(st.absentBg)) return 'A_MANUAL';
    if (same(st.presentBg)) return 'P';
  }
  return '';
}

function isStudentRowHighlightedRed_(infoBgs, infoFontColors, rowIdx, maxCols) {
  if (!infoBgs && !infoFontColors) return false;
  for (let c = 0; c < maxCols; c++) {
    const bgType = infoBgs && infoBgs[rowIdx] ? classifySheetCellColor_(infoBgs[rowIdx][c]) : 'EMPTY';
    const fcType = infoFontColors && infoFontColors[rowIdx] ? classifySheetCellColor_(infoFontColors[rowIdx][c]) : 'EMPTY';
    if (bgType === 'RED' || fcType === 'RED') return true;
  }
  return false;
}

/**
 * Extract student roster (No., Student_ID, Full_Name) from either a Google Sheet or .XLSX file.
 * Also scans student identity cells (STT, MSSV, Họ tên) for manual Red Exam-Ban highlights (#ff0000, #e06666...)
 * and full-sheet attendance cells (x, Yellow, Orange) so pre-existing sheet data is synced to Master Sheet.
 */
function extractRosterFromTargetFile_(targetInfo, classId, className) {
  return withTargetSpreadsheet_(targetInfo, classId, className, false, function(ss, sheet) {
    const layout = detectSheetLayout_(sheet);
    const lastRow = Math.min(sheet.getLastRow(), 2000);
    if (lastRow < layout.dataStartRow) {
      return { roster: [], sheetName: sheet.getName() };
    }

    const totalCols = Math.min(sheet.getLastColumn(), 100);
    let detectedDateCols = []; // [{ colIdx0, colNumber, day, month, year, dayKey, displayDate, label }]

    // Extract period & session date schedule metadata from top rows (rows 1..dateRow)
    try {
      const metaRows = Math.min(layout.dateRow, sheet.getLastRow(), 12);
      const rawTop = sheet.getRange(1, 1, metaRows, totalCols).getValues();
      const dispTop = sheet.getRange(1, 1, metaRows, totalCols).getDisplayValues();
      const headerTexts = [sheet.getName(), className || ''];
      for (let r = 0; r < metaRows; r++) {
        for (let c = 0; c < Math.min(totalCols, 12); c++) {
          const cellStr = String(dispTop[r][c] || '').trim();
          if (cellStr && cellStr.length <= 120) headerTexts.push(cellStr);
        }
      }
      let sessionDates = [];
      const dRowIdx = layout.dateRow - 1;
      if (dRowIdx >= 0 && dRowIdx < metaRows) {
        const scannedDates = scanSheetDateColumns_(rawTop[dRowIdx], dispTop[dRowIdx], layout.dateStartCol - 1, totalCols, layout.dateStartCol);
        sessionDates = scannedDates.sessionDates;
        detectedDateCols = scannedDates.columns;
      }
      saveClassScheduleMetaFromSheet_(classId, headerTexts.join(' | '), sessionDates);
    } catch (metaErr) {}

    const numRows = lastRow - layout.dataStartRow + 1;
    const identityCols = Math.max(layout.idCol, layout.nameCol + (layout.isTwoColName ? 1 : 0));
    const maxColNeeded = Math.max(identityCols, totalCols);
    const infoRange = sheet.getRange(layout.dataStartRow, 1, numRows, maxColNeeded);
    const rows = infoRange.getValues();
    const idDispCol = sheet.getRange(layout.dataStartRow, layout.idCol, numRows, 1).getDisplayValues();
    for (let i = 0; i < rows.length; i++) {
      rows[i][layout.idCol - 1] = typeof pickStudentIdCell_ === 'function' ? pickStudentIdCell_(rows[i][layout.idCol - 1], idDispCol[i][0]) : rows[i][layout.idCol - 1];
    }
    let infoBgs = null;
    let infoFontColors = null;
    try {
      infoBgs = infoRange.getBackgrounds();
      infoFontColors = infoRange.getFontColors();
    } catch (e) {}

    const roster = [];
    const manualBans = {};
    const validRowStudentIds = [];

    for (let i = 0; i < rows.length; i++) {
      const parsedRow = extractRowStudentIdAndName_(rows[i], layout);
      const sid = parsedRow.sid;
      const fullName = parsedRow.fullName;
      if (!sid || !fullName) {
        validRowStudentIds.push('');
        continue;
      }

      const sidUpper = sid.toUpperCase();
      validRowStudentIds.push(sidUpper);

      // Check if STT, MSSV, or Full Name cell is highlighted in Red (Manual Exam Ban on Sheet)
      if (isStudentRowHighlightedRed_(infoBgs, infoFontColors, i, identityCols)) {
        manualBans[sidUpper] = true;
      }

      const stt = roster.length + 1;
      roster.push([stt, sid, fullName, 0, 0]);
    }

    saveManualExamBanForClass_(classId, manualBans);

    // Read every date column once (same conversion as the roster) so the two-way sync can reuse it
    const dateColumns = [];
    try {
      const readMarkStyle = getAttendanceStyle_();
      for (let d = 0; d < detectedDateCols.length; d++) {
        const dc = detectedDateCols[d];
        const colVals = [];
        const colBgs = [];
        for (let r = 0; r < rows.length; r++) {
          colVals.push([rows[r][dc.colIdx0]]);
          colBgs.push([infoBgs && infoBgs[r] ? infoBgs[r][dc.colIdx0] : '']);
        }
        dateColumns.push({
          dayKey: dc.dayKey,
          displayDate: dc.displayDate,
          label: dc.label,
          marks: readDayColumnMarks_(colVals, colBgs, validRowStudentIds, readMarkStyle)
        });
      }
    } catch (colErr) {}

    return {
      roster: roster,
      sheetName: sheet.getName(),
      dateColumns: dateColumns
    };
  });
}

function saveClassScheduleMetaFromSheet_(classId, combinedHeaderText, sessionDates) {
  if (!classId) return;
  const s = String(combinedHeaderText || '');
  let startPeriod = null;
  let endPeriod = null;

  const compactMatch = s.match(/(?:ti[eế]t|t|ca)?\s*\b(123|456|789|101112|131415|161718)\b/i);
  if (compactMatch) {
    const map = {
      '123': [1, 3],
      '456': [4, 6],
      '789': [7, 9],
      '101112': [10, 12],
      '131415': [13, 15],
      '161718': [16, 18]
    };
    const pair = map[compactMatch[1]];
    if (pair) {
      startPeriod = pair[0];
      endPeriod = pair[1];
    }
  } else {
    const rangeMatch = s.match(/(?:ti[eế]t|t|periods?)?\s*[:.]?\s*\b(\d{1,2})\s*[-–→to]\s*(\d{1,2})\b/i);
    if (rangeMatch) {
      const sp = parseInt(rangeMatch[1], 10);
      const ep = parseInt(rangeMatch[2], 10);
      if (sp >= 1 && ep >= sp && ep <= 18 && (ep - sp) <= 6) {
        startPeriod = sp;
        endPeriod = ep;
      }
    }
  }

  const daysOfWeek = [];
  const addDay = function(d) {
    if (typeof d === 'number' && d >= 0 && d <= 6 && daysOfWeek.indexOf(d) === -1) {
      daysOfWeek.push(d);
    }
  };
  const thuRegex = /(?:^|[\s,;|(\[/])(?:th[uứ]\s*|t)([2-7](?:\s*(?:[&,+]\s*(?:th[uứ]\s*|t)?|[-–]\s*(?:th[uứ]\s*|t))[2-7])*)\b(?!\s*[-–→to]\s*\d)/ig;
  let m;
  while ((m = thuRegex.exec(s)) !== null) {
    const digits = m[1].match(/[2-7]/g) || [];
    for (let i = 0; i < digits.length; i++) {
      addDay(parseInt(digits[i], 10) - 1);
    }
  }
  if (/\b(ch[uủ]\s*nh[aậ]t|cn|sunday|sun)\b/i.test(s)) addDay(0);
  if (/\b(monday|mon)\b/i.test(s)) addDay(1);
  if (/\b(tuesday|tue)\b/i.test(s)) addDay(2);
  if (/\b(wednesday|wed)\b/i.test(s)) addDay(3);
  if (/\b(thursday|thu)\b/i.test(s)) addDay(4);
  if (/\b(friday|fri)\b/i.test(s)) addDay(5);
  if (/\b(saturday|sat)\b/i.test(s)) addDay(6);

  const dayOfWeek = daysOfWeek.length > 0 ? daysOfWeek[0] : null;
  mergeAndSaveClassSessionDates_(classId, sessionDates, {
    startPeriod: startPeriod,
    endPeriod: endPeriod,
    dayOfWeek: dayOfWeek,
    daysOfWeek: daysOfWeek
  });
}

function scanSheetDateColumns_(rawRow, dispRow, startIdx0, endIdx0, baseColNumber) {
  const sessionDates = [];
  const columns = [];
  let runningYear = new Date().getFullYear();
  let prevMonth = null;
  for (let idx = startIdx0; idx < endIdx0; idx++) {
    const parsed = parseDateCell_(rawRow[idx], dispRow[idx]);
    if (parsed) {
      let yr = parsed.year;
      if (!yr) {
        if (prevMonth !== null && parsed.month < prevMonth && (prevMonth - parsed.month) >= 6) {
          runningYear++;
        }
        yr = runningYear;
      } else {
        runningYear = yr;
      }
      prevMonth = parsed.month;
      const dd = ('0' + parsed.day).slice(-2);
      const mm = ('0' + parsed.month).slice(-2);
      const dayKey = yr + '-' + mm + '-' + dd;
      const colNumber = baseColNumber + (idx - startIdx0);
      sessionDates.push(dayKey);
      columns.push({
        colIdx0: colNumber - 1,
        colNumber: colNumber,
        day: parsed.day,
        month: parsed.month,
        year: yr,
        dayKey: dayKey,
        displayDate: dd + '/' + mm + '/' + yr,
        label: dd + '/' + mm
      });
    }
  }
  return { sessionDates: sessionDates, columns: columns };
}

function mergeAndSaveClassSessionDates_(classId, sessionDates, extraMetaFields) {
  if (!classId) return;
  const props = PropertiesService.getScriptProperties();
  const existingMap = JSON.parse(props.getProperty('QR_CLASS_SCHEDULE_META_MAP') || '{}');
  const prevEntry = existingMap[classId] || {};
  const makeUpDates = Array.isArray(prevEntry.makeUpDates) ? prevEntry.makeUpDates : [];
  const mergedDates = (sessionDates || []).slice();
  makeUpDates.forEach(function(md) {
    if (md && mergedDates.indexOf(md) === -1) {
      mergedDates.push(md);
    }
  });
  mergedDates.sort(function(a, b) { return String(a).localeCompare(String(b)); });

  if (extraMetaFields) {
    existingMap[classId] = {
      startPeriod: extraMetaFields.startPeriod,
      endPeriod: extraMetaFields.endPeriod,
      dayOfWeek: extraMetaFields.dayOfWeek,
      daysOfWeek: extraMetaFields.daysOfWeek,
      sessionDates: mergedDates,
      makeUpDates: makeUpDates
    };
  } else {
    prevEntry.sessionDates = mergedDates;
    existingMap[classId] = prevEntry;
  }
  props.setProperty('QR_CLASS_SCHEDULE_META_MAP', JSON.stringify(existingMap));
}

/**
 * Update attendance marks for (targetDay, targetMonth) on either a Google Sheet or .XLSX file.
 * `cellsToWrite` = { STUDENT_ID: 'P'|'L'|'E'|'-' }: only those cells are written (grouped RangeList writes),
 * `previousCells` lists only the touched cells, and `afterModifiedMs` is the file's modified time after the write.
 * `restoreCells` (Undo) keeps the legacy cell-by-cell restore.
 */
function updateTargetAttendanceFile_(targetInfo, classId, className, targetDay, targetMonth, cellsToWrite, restoreCells, expectedSheetMarks) {
  const markStyle = getAttendanceStyle_();

  return withTargetSpreadsheet_(targetInfo, classId, className, true, function(ss, sheet) {
    const layout = detectSheetLayout_(sheet);
    const lastCol = Math.max(sheet.getLastColumn(), layout.dateStartCol + 35);
    const numCols = Math.min(lastCol - layout.dateStartCol + 1, 100);

    const dateRowValues = sheet.getRange(layout.dateRow, layout.dateStartCol, 1, numCols).getValues()[0];
    const dateRowDisplay = sheet.getRange(layout.dateRow, layout.dateStartCol, 1, numCols).getDisplayValues()[0];
    const scannedDates = scanSheetDateColumns_(dateRowValues, dateRowDisplay, 0, numCols, layout.dateStartCol);
    const refreshedSessionDates = scannedDates.sessionDates;
    const dateColPositions = scannedDates.columns;
    let targetCol = -1;

    for (let i = 0; i < dateColPositions.length; i++) {
      if (dateColPositions[i].day === targetDay && dateColPositions[i].month === targetMonth) {
        targetCol = dateColPositions[i].colNumber;
        break;
      }
    }

    // If targetCol was not found on the Sheet, check if this date is a confirmed Make-up session
    if (targetCol === -1 && restoreCells === null) {
      try {
        const props = PropertiesService.getScriptProperties();
        const existingMap = JSON.parse(props.getProperty('QR_CLASS_SCHEDULE_META_MAP') || '{}');
        const curEntry = existingMap[classId] || {};
        const makeUpDates = Array.isArray(curEntry.makeUpDates) ? curEntry.makeUpDates : [];
        const ddTarget = ('0' + targetDay).slice(-2);
        const mmTarget = ('0' + targetMonth).slice(-2);
        let matchedMakeUpKey = '';
        for (let m = 0; m < makeUpDates.length; m++) {
          const mk = String(makeUpDates[m] || '');
          if (mk.slice(5, 7) === mmTarget && mk.slice(8, 10) === ddTarget) {
            matchedMakeUpKey = mk;
            break;
          }
        }

        if (matchedMakeUpKey) {
          if (dateColPositions.length > 0) {
            let insertBeforeCol = -1;
            for (let p = 0; p < dateColPositions.length; p++) {
              if (dateColPositions[p].dayKey > matchedMakeUpKey) {
                insertBeforeCol = dateColPositions[p].colNumber;
                break;
              }
            }
            if (insertBeforeCol !== -1) {
              sheet.insertColumnBefore(insertBeforeCol);
              targetCol = insertBeforeCol;
            } else {
              const lastDateCol = dateColPositions[dateColPositions.length - 1].colNumber;
              sheet.insertColumnAfter(lastDateCol);
              targetCol = lastDateCol + 1;
            }
          } else {
            targetCol = layout.dateStartCol;
          }
          sheet.getRange(layout.dateRow, targetCol).setValue(ddTarget + '/' + mmTarget);
          if (refreshedSessionDates.indexOf(matchedMakeUpKey) === -1) {
            refreshedSessionDates.push(matchedMakeUpKey);
            refreshedSessionDates.sort(function(a, b) { return String(a).localeCompare(String(b)); });
          }
        }
      } catch (insertErr) {}
    }

    if (refreshedSessionDates.length > 0) {
      try {
        mergeAndSaveClassSessionDates_(classId, refreshedSessionDates, null);
      } catch (e) {}
    }

    if (targetCol === -1) {
      throw new Error(
        'Date column ' + targetDay + '/' + targetMonth +
        ' not found in row ' + layout.dateRow + ' of sheet "' + sheet.getName() + '" (' + classId + ').'
      );
    }

    const stats = {
      sheetName: sheet.getName(),
      presentCount: 0,
      lateCount: 0,
      excusedCount: 0,
      absentCount: 0,
      previousCells: [],
      skippedSids: []
    };

    if (restoreCells !== null) {
      for (let i = 0; i < restoreCells.length; i++) {
        const item = restoreCells[i];
        const cell = sheet.getRange(item.row, targetCol);
        cell.setValue(item.val || '');
        cell.setBackground(item.bg || null);
      }
      return stats;
    }

    const lastRow = Math.min(sheet.getLastRow(), 2000);
    if (lastRow < layout.dataStartRow) {
      return stats;
    }

    const numRows = lastRow - layout.dataStartRow + 1;
    const targetColRange = sheet.getRange(layout.dataStartRow, targetCol, numRows, 1);
    const maxColNeeded = Math.max(layout.idCol, layout.nameCol + (layout.isTwoColName ? 1 : 0));
    const infoRange = sheet.getRange(layout.dataStartRow, 1, numRows, maxColNeeded);
    const idAndNameRows = infoRange.getValues();
    const idDispCol = sheet.getRange(layout.dataStartRow, layout.idCol, numRows, 1).getDisplayValues();
    for (let i = 0; i < idAndNameRows.length; i++) {
      idAndNameRows[i][layout.idCol - 1] = typeof pickStudentIdCell_ === 'function' ? pickStudentIdCell_(idAndNameRows[i][layout.idCol - 1], idDispCol[i][0]) : idAndNameRows[i][layout.idCol - 1];
    }
    let infoBgs = null;
    let infoFontColors = null;
    try {
      infoBgs = infoRange.getBackgrounds();
      infoFontColors = infoRange.getFontColors();
    } catch (e) {}

    const oldColValues = targetColRange.getDisplayValues();
    const oldColBgs = targetColRange.getBackgrounds();

    // Normalize keys to uppercase so alphanumeric IDs match case-insensitively
    const normCells = {};
    Object.keys(cellsToWrite || {}).forEach(function(k) { normCells[String(k).toUpperCase()] = cellsToWrite[k]; });

    const manualBans = {};
    const writeGroups = {};
    const colLetter = colLetterSync_(targetCol);

    for (let idx = 0; idx < numRows; idx++) {
      const rowNumber = layout.dataStartRow + idx;
      const sid = extractRowStudentIdAndName_(idAndNameRows[idx], layout).sid;
      if (!sid) continue;

      const sidKey = sid.toUpperCase();
      if (isStudentRowHighlightedRed_(infoBgs, infoFontColors, idx, maxColNeeded)) {
        manualBans[sidKey] = true;
      }
      if (!Object.prototype.hasOwnProperty.call(normCells, sidKey)) continue;

      // Optimistic check: the cell must still look like it did when it was read for the merge
      if (expectedSheetMarks && Object.prototype.hasOwnProperty.call(expectedSheetMarks, sidKey)) {
        const nowMark = readDayColumnMarks_([[oldColValues[idx][0]]], [[oldColBgs[idx][0]]], [sidKey], markStyle)[sidKey];
        if (nowMark !== expectedSheetMarks[sidKey]) {
          stats.skippedSids.push(sidKey);
          continue;
        }
      }

      const oldBg = oldColBgs[idx][0];
      const oldBgColorClass = classifySheetCellColor_(oldBg);
      stats.previousCells.push({
        row: rowNumber,
        val: oldColValues[idx][0],
        bg: (oldBg && oldBg !== '#ffffff' && oldBg !== '#FFFFFF') ? oldBg : null,
        sid: sidKey
      });

      const status = normCells[sidKey];
      let text, bg;
      if (status === 'P') {
        text = markStyle.presentText;
        bg = markStyle.presentBg ? markStyle.presentBg : null;
        stats.presentCount++;
      } else if (status === 'L') {
        text = markStyle.lateText;
        bg = markStyle.lateBg ? markStyle.lateBg : null;
        stats.lateCount++;
      } else if (status === 'E') {
        text = markStyle.excusedText !== undefined ? markStyle.excusedText : '';
        bg = markStyle.excusedBg ? markStyle.excusedBg : (oldBgColorClass === 'ORANGE' ? oldBg : '#ff9900');
        stats.excusedCount++;
      } else {
        text = markStyle.absentText;
        bg = markStyle.absentBg ? markStyle.absentBg : null;
        stats.absentCount++;
      }

      // A red cell means "no information" for the lecturer: write the mark text but keep the red background
      const keepBg = oldBgColorClass === 'RED';
      const gKey = text + '' + (keepBg ? 'KEEP' : (bg || ''));
      if (!writeGroups[gKey]) writeGroups[gKey] = { text: text, bg: bg, keepBg: keepBg, a1: [] };
      writeGroups[gKey].a1.push(colLetter + rowNumber);
    }

    saveManualExamBanForClass_(classId, manualBans);

    Object.keys(writeGroups).forEach(function(k) {
      const grp = writeGroups[k];
      const rl = sheet.getRangeList(grp.a1);
      rl.setValue(grp.text);
      if (!grp.keepBg) rl.setBackground(grp.bg || null);
      rl.setHorizontalAlignment('center');
    });

    return stats;
  });
}

function colLetterSync_(n) {
  let s = '';
  let x = Number(n);
  while (x > 0) {
    const m = (x - 1) % 26;
    s = String.fromCharCode(65 + m) + s;
    x = Math.floor((x - 1) / 26);
  }
  return s;
}

function normalizeAttendanceStatusCodeSync_(val) {
  const s = String(val === undefined || val === null ? '' : val).trim().toUpperCase();
  if (s === 'P' || s === 'PRESENT' || s === 'CÓ MẶT' || s === 'CO MAT') return 'P';
  if (s === 'L' || s === 'LATE' || s === 'TRỄ' || s === 'TRE' || s === 'ĐI TRỄ' || s === 'DI TRE') return 'L';
  if (s === 'E' || s === 'EXCUSED' || s === 'CÓ PHÉP' || s === 'CO PHEP') return 'E';
  return '-';
}

function mergeAttendanceSync_(p1, p2, p2Opened) {
  if (typeof mergeAttendance === 'function') {
    return mergeAttendance(p1, p2, p2Opened);
  }
  const s1 = normalizeAttendanceStatusCodeSync_(p1);
  if (p2Opened === undefined) {
    const rawP2 = String(p2 === undefined || p2 === null ? '' : p2).trim();
    if (rawP2 === '') return s1;
  } else if (!p2Opened) {
    return s1;
  }
  if (s1 === 'E') return 'E';
  const s2Raw = normalizeAttendanceStatusCodeSync_(p2);
  if (s2Raw === 'E') return 'E';
  const s2Present = (s2Raw === 'P' || s2Raw === 'L');
  if (!s2Present) return '-';
  if (s1 === 'P') return 'P';
  return 'L';
}

// ---- Two-way sync pure helpers (no Sheet/Drive/Cache access) ----
function readDayColumnMarks_(vals, bgs, rowSids, style) {
  const out = {};
  const one = function(a, i) { const x = a ? a[i] : ''; return Array.isArray(x) ? x[0] : x; };
  for (let r = 0; r < (rowSids || []).length; r++) {
    const sid = String(rowSids[r] || '').trim().toUpperCase();
    if (!sid) continue;
    const raw = one(vals, r);
    const bgRaw = one(bgs, r);
    const cu = String(raw === undefined || raw === null ? '' : raw).trim().toUpperCase();
    const sm = classifyMarkFromStyle_(cu, bgRaw, style);
    const bgType = classifySheetCellColor_(bgRaw);
    let st;
    if (sm === 'E' || sm === 'L' || sm === 'P') st = sm;
    else if (sm === 'A_MANUAL') st = '-';
    else if (bgType === 'RED') st = 'RED_SKIP';
    else if (bgType === 'ORANGE' || cu === 'E' || cu === 'P_EXCUSED') st = 'E';
    else if (bgType === 'YELLOW' || cu === 'L' || cu === 'M') st = 'L';
    else if (cu === 'X' || cu === 'V' || cu === 'A' || cu === '-') st = '-';
    else if (cu === 'P' || cu === '1' || cu === '✓') st = 'P';
    else if (cu === '') st = 'BLANK';
    else st = 'UNKNOWN';
    out[sid] = st;
  }
  return out;
}

function computeThreeWayMerge_(base, web, sheet, opts) {
  const b0 = base || {}, w0 = web || {}, s0 = sheet || {};
  const firstPush = !!(opts && opts.firstPush);
  const has = function(o, k) { return Object.prototype.hasOwnProperty.call(o, k); };
  const isClear = function(v) { return v !== undefined && v !== 'BLANK' && v !== 'UNKNOWN' && v !== 'RED_SKIP'; };
  const res = { toSheet: {}, toWeb: {}, newBase: {}, conflicts: [] };
  Object.keys(b0).forEach(function(k) { res.newBase[k] = b0[k]; });
  const sids = {};
  [b0, w0, s0].forEach(function(o) { Object.keys(o).forEach(function(k) { sids[k] = 1; }); });
  Object.keys(sids).forEach(function(sid) {
    const w = has(w0, sid) ? w0[sid] : undefined;
    const raw = has(s0, sid) ? s0[sid] : undefined;
    if (raw === undefined || raw === 'UNKNOWN' || raw === 'RED_SKIP') return;
    const inBase = has(b0, sid);
    if (firstPush) {
      if (raw === 'BLANK') {
        if (w !== undefined) { res.toSheet[sid] = w; res.newBase[sid] = w; }
      } else {
        if (w !== raw) res.toWeb[sid] = raw;
        res.newBase[sid] = raw;
      }
      return;
    }
    let sh = raw;
    if (raw === 'BLANK') {
      if (!inBase) return;
      sh = 'P';
    }
    if (!inBase) {
      if (w === undefined) { res.toWeb[sid] = sh; }
      else if (w !== sh) { res.toWeb[sid] = sh; res.conflicts.push(sid); }
      res.newBase[sid] = sh;
      return;
    }
    const b = b0[sid];
    const webChanged = w !== undefined && w !== b;
    const sheetChanged = sh !== b;
    if (sheetChanged) {
      if (webChanged && w === sh) { res.newBase[sid] = sh; return; }
      res.toWeb[sid] = sh;
      res.newBase[sid] = sh;
      if (webChanged) res.conflicts.push(sid);
    } else if (webChanged) {
      res.toSheet[sid] = w;
      res.newBase[sid] = w;
    }
  });
  return res;
}

function computePushDiff_(base, web) {
  const out = {};
  const b0 = base || {}, w0 = web || {};
  Object.keys(b0).forEach(function(sid) {
    if (Object.prototype.hasOwnProperty.call(w0, sid) && w0[sid] !== b0[sid]) out[sid] = w0[sid];
  });
  return out;
}

function countStatuses_(map) {
  const c = { P: 0, L: 0, E: 0, A: 0 };
  Object.keys(map || {}).forEach(function(k) {
    const v = map[k];
    if (v === 'P') c.P++; else if (v === 'L') c.L++; else if (v === 'E') c.E++; else c.A++;
  });
  return c;
}

function isDayEligibleForPull_(hasSession, hasBase, isPaused) {
  return !!hasSession && !!hasBase && !isPaused;
}

function styleHash_(style) {
  const st = style || {};
  const s = Object.keys(st).sort().map(function(k) { return k + '=' + String(st[k] === undefined || st[k] === null ? '' : st[k]); }).join('|');
  let h = 5381;
  for (let i = 0; i < s.length; i++) h = ((h * 33) ^ s.charCodeAt(i)) >>> 0;
  return h.toString(16);
}

}

function getGroupedSessionsByClassDate_(masterSs) {
  const sessSheet = masterSs.getSheetByName('Sessions_Log');
  if (!sessSheet) throw new Error('Sheet \'Sessions_Log\' not found.');
  const data = sessSheet.getDataRange().getValues();
  const displayData = sessSheet.getDataRange().getDisplayValues();

  // Pre-scan Attendance_Log to filter out accidental/test sessions where 0 students had P/L/E/TEACHER_MANUAL
  const attSheet = masterSs.getSheetByName('Attendance_Log');
  const activeSessionIds = {};
  if (attSheet && attSheet.getLastRow() > 1) {
    const attRows = attSheet.getRange(2, 1, attSheet.getLastRow() - 1, 10).getValues();
    for (let a = 0; a < attRows.length; a++) {
      const sId = String(attRows[a][2] || '').trim();
      const st = String(attRows[a][5] || '').trim().toUpperCase();
      const src = String(attRows[a][9] || '').trim();
      if (sId && (st === 'P' || st === 'L' || st === 'E' || src === 'TEACHER_MANUAL')) {
        activeSessionIds[sId] = true;
      }
    }
  }

  const groupsMap = {};
  const groupOrder = [];

  for (let i = 1; i < data.length; i++) {
    const sessionId = String(data[i][0] || '').trim();
    const classId = String(data[i][1] || '').trim();
    const sessionLabel = String(data[i][2] || '').trim();
    if (!sessionId || !classId) continue;

    const openedAtRaw = data[i][3];
    const openedAtStr = String(displayData[i][3] || '').trim();
    const closesAtRaw = data[i][5];
    const closedManuallyStr = String(displayData[i][6] || '').trim();

    let day = null, month = null, year = null, openedAtMs = 0;
    if (openedAtRaw instanceof Date && !isNaN(openedAtRaw.getTime())) {
      day = openedAtRaw.getDate();
      month = openedAtRaw.getMonth() + 1;
      year = openedAtRaw.getFullYear();
      openedAtMs = openedAtRaw.getTime();
    } else {
      const m = openedAtStr.match(/(\d{4})-(\d{1,2})-(\d{1,2})/);
      if (m) {
        year = parseInt(m[1], 10);
        month = parseInt(m[2], 10);
        day = parseInt(m[3], 10);
      }
    }
    if (!day || !month) continue;

    let closesAtMs = 0;
    if (closesAtRaw instanceof Date && !isNaN(closesAtRaw.getTime())) {
      closesAtMs = closesAtRaw.getTime();
    }
    const isClosed = closedManuallyStr.length > 0 || (closesAtMs > 0 && Date.now() > closesAtMs);

    const isR2 = isRound2SessionSync_(sessionId, sessionLabel);
    // Skip closed ghost/test P1 sessions that have 0 real attendance marks, but NEVER skip a P2 session (Edge Case E.1: P2 opened with 0 scans)
    if (isClosed && !activeSessionIds[sessionId] && !isR2) continue;

    const dateKey = year + '-' + month + '-' + day;
    const groupKey = classId + '__' + dateKey;

    if (!groupsMap[groupKey]) {
      groupsMap[groupKey] = {
        groupKey: groupKey,
        classId: classId,
        day: day,
        month: month,
        dateKey: dateKey,
        dayKey: year + '-' + ('0' + month).slice(-2) + '-' + ('0' + day).slice(-2),
        sessionIds: [],
        p1SessionIds: [],
        p2SessionIds: [],
        p2Opened: false,
        sessionLabels: [],
        primaryLabel: sessionLabel.replace(/_P2$/i, ''),
        openedAtMs: openedAtMs,
        allClosed: true,
        maxClosesAtMs: closesAtMs
      };
      groupOrder.push(groupKey);
    }

    const grp = groupsMap[groupKey];
    grp.sessionIds.push(sessionId);
    if (isR2) {
      grp.p2SessionIds.push(sessionId);
      grp.p2Opened = true;
    } else {
      grp.p1SessionIds.push(sessionId);
    }
    if (sessionLabel && grp.sessionLabels.indexOf(sessionLabel) === -1) {
      grp.sessionLabels.push(sessionLabel);
    }
    grp.primaryLabel = grp.sessionLabels[0] || sessionLabel || grp.primaryLabel;
    if (openedAtMs > grp.openedAtMs) grp.openedAtMs = openedAtMs;
    if (!isClosed) grp.allClosed = false;
    if (closesAtMs > grp.maxClosesAtMs) grp.maxClosesAtMs = closesAtMs;
  }

  return groupOrder.map(function(k) { return groupsMap[k]; });
}

function getClassEndTimeOnCalendarMs_(masterSs, classId, now, openedAtMs) {
  try {
    const cal = CalendarApp.getDefaultCalendar();
    const events = cal.getEventsForDay(now);
    for (let i = 0; i < events.length; i++) {
      const title = events[i].getTitle() || '';
      if (title.indexOf(classId) !== -1) {
        return events[i].getEndTime().getTime();
      }
    }
  } catch (e) {}

  let endHour = 17, endMin = 30;
  try {
    const cfgSheet = masterSs.getSheetByName('Classes_Config');
    if (cfgSheet) {
      const cfg = cfgSheet.getDataRange().getValues();
      for (let i = 1; i < cfg.length; i++) {
        if (String(cfg[i][0] || '').trim() === classId) {
          const cname = String(cfg[i][1] || '');
          if (cname.indexOf('13-15') !== -1 || cname.indexOf('131415') !== -1 || cname.toLowerCase().indexOf('evening') !== -1 || cname.toLowerCase().indexOf('tối') !== -1) {
            endHour = 20; endMin = 0;
          } else if (cname.indexOf('10-12') !== -1 || cname.indexOf('101112') !== -1) {
            endHour = 17; endMin = 30;
          } else if (cname.indexOf('7-9') !== -1 || cname.indexOf('789') !== -1) {
            endHour = 15; endMin = 0;
          } else if (cname.indexOf('4-6') !== -1 || cname.indexOf('456') !== -1) {
            endHour = 11; endMin = 30;
          } else if (cname.indexOf('1-3') !== -1 || cname.indexOf('123') !== -1) {
            endHour = 9; endMin = 0;
          }
          break;
        }
      }
    }
  } catch (e) {}

  if (openedAtMs > 0) {
    const openedDate = new Date(openedAtMs);
    const h = openedDate.getHours() + openedDate.getMinutes() / 60;
    if (h >= 17.25) {
      endHour = 20; endMin = 0;
    } else if (h >= 14.75) {
      endHour = 17; endMin = 30;
    } else if (h >= 12.25) {
      endHour = 15; endMin = 0;
    } else if (h >= 8.75) {
      endHour = 11; endMin = 30;
    } else {
      endHour = 9; endMin = 0;
    }
  }

  const fallbackEnd = new Date(now.getFullYear(), now.getMonth(), now.getDate(), endHour, endMin, 0);
  return Math.max(fallbackEnd.getTime(), openedAtMs || 0);
}

function getMergedAttendanceStatusMap_(masterSs, classId, sessionGroup, pre) {
  const statusMap = {};
  const p1Marks = {};
  const p2Marks = {};
  const p1Manual = {};
  const p2Manual = {};
  const allKeys = {};

  const preClassValues = pre && pre.classValues ? pre.classValues[classId] : null;
  const classSheet = pre && pre.classValues ? null : masterSs.getSheetByName(classId);
  if (classSheet || preClassValues) {
    const values = preClassValues || classSheet.getDataRange().getValues();
    if (values.length > 1) {
      const headers = values[0].map(function(h) { return String(h).trim(); });
      const matchedFinalCols = [];

      for (let c = 5; c < headers.length; c++) {
        if (/_P1$|_Trạng_thái_P2$/i.test(headers[c])) continue;
        for (let s = 0; s < sessionGroup.sessionLabels.length; s++) {
          const lbl = String(sessionGroup.sessionLabels[s] || '').replace(/_P2$/i, '');
          if (lbl && headers[c].indexOf(lbl) !== -1) {
            matchedFinalCols.push(c);
            break;
          }
        }
      }

      for (let r = 1; r < values.length; r++) {
        const sid = normalizeStudentId_(values[r][1], values[r][2]);
        if (!sid) continue;
        const key = sid.toUpperCase();
        allKeys[key] = true;
        statusMap[key] = '-';
        for (let k = 0; k < matchedFinalCols.length; k++) {
          const st = String(values[r][matchedFinalCols[k]] || '').trim().toUpperCase();
          if (st === 'P' || st === 'E' || st === 'L') {
            p1Marks[key] = st;
          }
        }
      }
    }
  }

  const sessionIdSet = {};
  let hasP2Session = Boolean(sessionGroup.p2Opened);
  for (let i = 0; i < sessionGroup.sessionIds.length; i++) {
    const sId = sessionGroup.sessionIds[i];
    sessionIdSet[sId] = true;
    if (isRound2SessionSync_(sId, '')) hasP2Session = true;
  }

  const preAttData = pre && pre.attData ? pre.attData : null;
  const attSheet = preAttData ? null : masterSs.getSheetByName('Attendance_Log');
  if (attSheet || preAttData) {
    const attData = preAttData || attSheet.getDataRange().getValues();
    for (let i = 1; i < attData.length; i++) {
      const rowSessId = String(attData[i][2] || '').trim();
      const rowClassId = String(attData[i][3] || '').trim();
      if (sessionIdSet[rowSessId] && rowClassId === classId) {
        const sid = normalizeStudentId_(attData[i][4]);
        const st = String(attData[i][5] || '').trim().toUpperCase();
        const src = String(attData[i][9] || '').trim();
        if (sid && (st === 'P' || st === 'E' || st === 'L' || st === '-')) {
          const key = sid.toUpperCase();
          allKeys[key] = true;
          const isManual = (src === 'TEACHER_MANUAL');
          const isR2 = isRound2SessionSync_(rowSessId, '');
          if (isR2) {
            if (isManual) {
              p2Marks[key] = (st === 'P' || st === 'L') ? 'P' : (st === 'E' ? 'E' : '-');
              p2Manual[key] = true;
            } else if (!p2Manual[key] && !p2Marks[key]) {
              p2Marks[key] = (st === 'P' || st === 'L') ? 'P' : (st === 'E' ? 'E' : '-');
            }
          } else {
            if (isManual) {
              p1Marks[key] = (st === 'P' || st === 'L' || st === 'E') ? st : '-';
              p1Manual[key] = true;
            } else if (!p1Manual[key]) {
              const cur = p1Marks[key] || '-';
              if (st === 'P') p1Marks[key] = 'P';
              else if (st === 'E' && cur !== 'P') p1Marks[key] = 'E';
              else if (st === 'L' && cur !== 'P' && cur !== 'E') p1Marks[key] = 'L';
            }
          }
        }
      }
    }
  }

  Object.keys(allKeys).forEach(function(key) {
    const p1St = p1Marks[key] || '-';
    const p2St = hasP2Session ? (p2Marks[key] || '-') : '';
    statusMap[key] = mergeAttendanceSync_(p1St, p2St, hasP2Session);
  });

  return statusMap;
}

function updateMasterClassSheet_(masterSs, classId, sessionGroup, statusMap) {
  if (typeof rebuildSingleClassGrid === 'function') {
    try {
      rebuildSingleClassGrid(masterSs, classId);
      CacheService.getScriptCache().remove('CLASS_ATT_STATS_V36_' + classId);
      SpreadsheetApp.flush();
      return;
    } catch (e) {}
  }

  const classSheet = masterSs.getSheetByName(classId);
  if (!classSheet) return;
  const values = classSheet.getDataRange().getValues();
  if (values.length <= 1) return;

  const headers = values[0].map(function(h) { return String(h).trim(); });
  let sessionColIdx = -1;

  for (let c = 5; c < headers.length; c++) {
    for (let s = 0; s < sessionGroup.sessionLabels.length; s++) {
      const lbl = sessionGroup.sessionLabels[s];
      if (lbl && headers[c].indexOf(lbl) !== -1) {
        sessionColIdx = c;
        break;
      }
    }
    if (sessionColIdx !== -1) break;
  }

  if (sessionColIdx === -1) {
    sessionColIdx = Math.max(headers.length, 5);
    classSheet.getRange(1, sessionColIdx + 1).setValue(sessionGroup.primaryLabel);
    headers[sessionColIdx] = sessionGroup.primaryLabel;
  }

  const numStudents = values.length - 1;
  const sessionColOut = [];
  const totalsOut = [];

  for (let r = 1; r < values.length; r++) {
    const sid = normalizeStudentId_(values[r][1], values[r][2]);
    const st = sid ? (statusMap[sid.toUpperCase()] || '-') : String(values[r][sessionColIdx] || '-');
    sessionColOut.push([st]);

    let totalP = 0;
    let totalL = 0;
    for (let c = 5; c < headers.length; c++) {
      const cellVal = (c === sessionColIdx) ? st : String(values[r][c] || '').trim().toUpperCase();
      if (cellVal === 'P') totalP++;
      if (cellVal === 'L') totalL++;
    }
    totalsOut.push([totalP, totalL]);
  }

  if (numStudents > 0) {
    classSheet.getRange(2, sessionColIdx + 1, numStudents, 1).setValues(sessionColOut);
    classSheet.getRange(2, 4, numStudents, 2).setValues(totalsOut);
  }
  try {
    CacheService.getScriptCache().remove('CLASS_ATT_STATS_V36_' + classId);
  } catch (e) {}
  SpreadsheetApp.flush();
}

/**
 * Converts `[[tag ...]]` template syntax into real HTML tags at runtime using
 * character codes so plain-text / Canvas HTML sanitizers never corrupt quotes.
 */
function renderDialogHtml_(tpl, width, height) {
  const openAngle = String.fromCharCode(60);
  const closeAngle = String.fromCharCode(62);
  const htmlStr = String(tpl || '')
    .replace(/\[\[/g, openAngle)
    .replace(/\]\]/g, closeAngle);
  return HtmlService.createHtmlOutput(htmlStr).setWidth(width).setHeight(height);
}

function escapeHtml_(str) {
  const openAngle = String.fromCharCode(60);
  const closeAngle = String.fromCharCode(62);
  return String(str || '')
    .replace(/&/g, '&amp;')
    .split(openAngle).join('&lt;')
    .split(closeAngle).join('&gt;')
    .replace(/"/g, '&quot;')
    .replace(/'/g, '&#39;');
}

// Reads class data through Code.gs helpers and returns plain data for the mail.
function buildWeeklyReportData_(ss, cfg, nowMs, prevIds) {
  const todayKey = reportDayKey_(nowMs);
  const startKey = reportDayKey_(nowMs - 6 * 86400000);
  const rulesMap = getClassRulesMap_();
  const classes = getClassesMap(ss, true);
  const wanted = cfg.classIds.length ? cfg.classIds : Object.keys(classes);
  const out = { todayKey: todayKey, classes: [], ids: [], totals: { ban: 0, warn: 0, attn: 0, excused: 0 }, hadPrev: Array.isArray(prevIds) };
  const prevSet = {};
  (prevIds || []).forEach(function (k) { prevSet[k] = true; });

  wanted.forEach(function (cid) {
    if (!classes[cid]) return;
    const stats = getClassAttendanceStatsMap_(ss, cid, true) || {};
    const roster = getClassRosterMap(ss, cid, false) || {};
    const sids = Object.keys(stats);
    const weekDays = {};
    let firstDay = '';
    sids.forEach(function (sid) {
      (stats[sid].history || []).forEach(function (h) {
        if (!h || !h.dayKey) return;
        if (!firstDay || h.dayKey < firstDay) firstDay = h.dayKey;
        if (h.dayKey >= startKey && h.dayKey <= todayKey && h.finalStatus !== 'OPEN') weekDays[h.dayKey] = true;
      });
    });
    const weekCount = Object.keys(weekDays).length;
    if (!weekCount) return;

    const rules = getClassRules_(cid);
    const banAbs = effectiveBanAbsences_(rules);
    const isDefault = !rulesMap[cid];
    let present = 0, counted = 0, excused = 0;
    const ban = [], warnGroups = {}, warnOrder = [], attn = [];
    let warnTotal = 0;

    sids.forEach(function (sid) {
      const st = stats[sid];
      const name = (roster[sid] && roster[sid].fullName) || sid;
      const hist = (st.history || []).filter(function (h) { return h && h.finalStatus !== 'OPEN'; });
      hist.forEach(function (h) {
        if (h.dayKey < startKey || h.dayKey > todayKey) return;
        counted++;
        if (h.finalStatus === 'P' || h.finalStatus === 'L') present++;
        if (h.finalStatus === 'E') excused++;
      });
      const ev = evaluateAttendanceRules_(st.P, st.L, st.E, st.A, st.manualExamBan, rules);
      const id = cid + ':' + sid;
      const isNew = out.hadPrev && !prevSet[id];
      if (ev.examStatus === 'BAN') {
        if (!st.manualExamBan) ban.push({ id: id, sid: sid, name: name, isNew: isNew, note: 'vắng quy đổi ' + ev.equivAbsences + ', điểm CC ' + ev.attendanceScore });
        return;
      }
      if (ev.examStatus === 'WARN') {
        const useAbs = rules.banMode === 'BOTH' || rules.banMode === 'ABSENCES';
        const absWarn = useAbs && ev.equivAbsences >= banAbs - rules.warnAbsMargin && ev.equivAbsences <= banAbs;
        const key = absWarn ? ('vắng quy đổi ' + ev.equivAbsences + '/' + banAbs + ' buổi (cấm thi khi vượt)')
          : ('điểm chuyên cần ' + ev.attendanceScore + ' (cấm thi khi dưới ' + rules.banScoreBelow + ')');
        if (!warnGroups[key]) { warnGroups[key] = []; warnOrder.push(key); }
        warnGroups[key].push({ id: id, sid: sid, name: name, isNew: isNew });
        warnTotal++;
        return;
      }
      const sts = hist.map(function (h) { return h.finalStatus; });
      const absStreak = reportTrailingStreak_(sts, function (x) { return x === '-' || x === 'A'; });
      const lateStreak = reportTrailingStreak_(sts, function (x) { return x === 'L'; });
      if (absStreak >= 2) attn.push({ id: id, sid: sid, name: name, isNew: isNew, note: 'vắng ' + absStreak + ' buổi gần nhất liên tiếp' });
      else if (lateStreak >= 3) attn.push({ id: id, sid: sid, name: name, isNew: isNew, note: 'trễ ' + lateStreak + ' buổi liên tiếp' });
    });

    let shown = 0;
    const warnLines = [];
    warnOrder.forEach(function (key) {
      const list = warnGroups[key].slice(0, Math.max(0, REPORT_MAX_ROWS - shown));
      shown += list.length;
      if (list.length) warnLines.push({ reason: key, students: list });
    });
    const banShown = ban.slice(0, REPORT_MAX_ROWS);
    const attnShown = attn.slice(0, REPORT_MAX_ROWS);
    banShown.concat(attnShown).forEach(function (x) { out.ids.push(x.id); });
    warnLines.forEach(function (g) { g.students.forEach(function (x) { out.ids.push(x.id); }); });

    out.totals.ban += ban.length;
    out.totals.warn += warnTotal;
    out.totals.attn += attn.length;
    out.totals.excused += excused;
    out.classes.push({
      classId: cid, className: classes[cid].className || '', size: sids.length, weekCount: weekCount,
      presentPct: counted ? Math.round(present * 100 / counted) : 0,
      banCount: ban.length, warnCount: warnTotal, attnCount: attn.length, excused: excused,
      ban: banShown, banMore: ban.length - banShown.length,
      warnLines: warnLines, warnMore: warnTotal - shown,
      attn: attnShown, attnMore: attn.length - attnShown.length,
      isDefaultRules: isDefault, presetId: rules.presetId, firstDay: firstDay
    });
  });
  return out;
}

function renderWeeklyReportMail_(data) {
  const dd = data.todayKey.slice(8, 10) + '/' + data.todayKey.slice(5, 7);
  const t = data.totals;
  const subject = '[CCS] Chuyên cần tuần ' + dd + ': ' + t.ban + ' cấm thi, ' + t.warn + ' sát nút, ' + t.attn + ' chú ý';
  const E = escapeHtml_;
  const dot = function (c) { return '<span style="color:' + c + ';">&#9679;</span> '; };
  const NEW_H = ' <span style="background:#2563eb;color:#fff;font-size:10px;padding:1px 4px;border-radius:3px;">MỚI</span>';
  const NEW_T = ' [MỚI]';
  const h = [], x = [];
  h.push('<div style="font-family:Arial,sans-serif;max-width:560px;margin:0 auto;font-size:14px;color:#111827;">');
  h.push('<p>Xin chào, đây là báo cáo chuyên cần tuần này (đến ' + E(reportDisplayDate_(data.todayKey)) + ').</p>');
  x.push('Xin chào, đây là báo cáo chuyên cần tuần này (đến ' + reportDisplayDate_(data.todayKey) + ').', '');

  h.push('<h3 style="margin:14px 0 6px;">1) Tóm tắt</h3><table style="border-collapse:collapse;width:100%;font-size:13px;"><tr style="background:#f3f4f6;text-align:left;"><th style="padding:4px;">Lớp</th><th>Sĩ số</th><th>Có mặt</th><th>Cấm thi</th><th>Sát nút</th><th>Chú ý</th></tr>');
  x.push('1) TÓM TẮT (Lớp | Sĩ số | Có mặt tuần | Cấm thi | Sát nút | Chú ý)');
  data.classes.forEach(function (c) {
    h.push('<tr style="border-top:1px solid #e5e7eb;"><td style="padding:4px;">' + E(c.classId) + '</td><td>' + c.size + '</td><td>' + c.presentPct + '% (' + c.weekCount + ' buổi)</td><td>' + c.banCount + '</td><td>' + c.warnCount + '</td><td>' + c.attnCount + '</td></tr>');
    x.push(c.classId + ' | ' + c.size + ' | ' + c.presentPct + '% (' + c.weekCount + ' buổi) | ' + c.banCount + ' | ' + c.warnCount + ' | ' + c.attnCount);
  });
  h.push('</table>');
  x.push('');

  const sect = function (title, color, label, pick) {
    let any = false;
    const hh = [], xx = [];
    data.classes.forEach(function (c) {
      const r = pick(c);
      if (!r.rows.length) return;
      any = true;
      hh.push('<p style="margin:8px 0 2px;"><b>' + E(c.classId) + '</b></p>');
      xx.push(c.classId + ':');
      r.rows.forEach(function (row) {
        hh.push('<div style="margin:2px 0 2px 8px;">' + dot(color) + row.h + '</div>');
        xx.push('  - ' + row.t);
      });
      if (r.more > 0) {
        hh.push('<div style="margin:2px 0 2px 8px;color:#6b7280;">+' + r.more + ' sinh viên khác, xem trên web</div>');
        xx.push('  +' + r.more + ' sinh viên khác, xem trên web');
      }
    });
    h.push('<h3 style="margin:14px 0 6px;">' + title + '</h3>');
    x.push(label);
    if (!any) { h.push('<div style="color:#6b7280;">Không có.</div>'); x.push('  Không có.'); } else { h.push(hh.join('')); x.push.apply(x, xx); }
    x.push('');
  };

  sect('2) Cần xử lý ngay: vượt ngưỡng cấm thi, chưa đánh dấu', '#dc2626', '2) CẦN XỬ LÝ NGAY (vượt ngưỡng cấm thi, chưa đánh dấu)', function (c) {
    return { more: c.banMore, rows: c.ban.map(function (s) {
      return { h: E(s.name) + ' (' + E(s.sid) + ') - ' + E(s.note) + (s.isNew ? NEW_H : ''), t: s.name + ' (' + s.sid + ') - ' + s.note + (s.isNew ? NEW_T : '') };
    }) };
  });
  sect('3) Sát nút', '#f97316', '3) SÁT NÚT', function (c) {
    return { more: c.warnMore, rows: c.warnLines.map(function (g) {
      const names = g.students.map(function (s) { return s.name + (s.isNew ? NEW_T : ''); }).join(', ');
      const namesH = g.students.map(function (s) { return E(s.name) + (s.isNew ? NEW_H : ''); }).join(', ');
      return { h: namesH + ' - ' + E(g.reason), t: names + ' - ' + g.reason };
    }) };
  });
  sect('4) Chú ý hành vi (chuỗi đang diễn ra)', '#eab308', '4) CHÚ Ý HÀNH VI (chuỗi đang diễn ra)', function (c) {
    return { more: c.attnMore, rows: c.attn.map(function (s) {
      return { h: E(s.name) + ' (' + E(s.sid) + ') - ' + E(s.note) + (s.isNew ? NEW_H : ''), t: s.name + ' (' + s.sid + ') - ' + s.note + (s.isNew ? NEW_T : '') };
    }) };
  });

  const safe = data.classes.filter(function (c) { return !c.banCount && !c.warnCount && !c.attnCount; }).map(function (c) { return c.classId; });
  const foot = [];
  foot.push('Vắng có phép tuần này: ' + t.excused + ' lượt.');
  foot.push('Lớp an toàn (không ai cần chú ý): ' + (safe.length ? safe.join(', ') : 'không có') + '.');
  data.classes.forEach(function (c) {
    foot.push('Lớp ' + c.classId + ': ' + (c.isDefaultRules ? 'quy chế mặc định' : 'quy chế đã đặt (' + c.presetId + ')') + ', dữ liệu tính từ ngày ' + (c.firstDay ? reportDisplayDate_(c.firstDay) : 'chưa có') + '.');
  });
  h.push('<h3 style="margin:14px 0 6px;">5) Ghi chú</h3>');
  x.push('5) GHI CHÚ');
  foot.forEach(function (l) { h.push('<div style="font-size:12px;color:#374151;margin:2px 0;">' + E(l) + '</div>'); x.push('  ' + l); });
  h.push('</div>');
  return { subject: subject, htmlBody: h.join(''), body: x.join('\n') };
}

function sendWeeklyReportNow_(ss, cfg, nowMs) {
  const props = PropertiesService.getScriptProperties();
  let prev = null;
  try { const raw = props.getProperty(REPORT_LAST_IDS_KEY); if (raw) { const p = JSON.parse(raw); if (Array.isArray(p)) prev = p; } } catch (e) {}
  const data = buildWeeklyReportData_(ss, cfg, nowMs, prev);
  if (!data.classes.length) return 'NO_SESSIONS';
  const to = getReportRecipient_(cfg);
  if (!to) return 'NO_RECIPIENT';
  const mail = renderWeeklyReportMail_(data);
  const prevSent = props.getProperty(REPORT_LAST_SENT_KEY);
  props.setProperty(REPORT_LAST_SENT_KEY, data.todayKey);
  try {
    MailApp.sendEmail({ to: to, subject: mail.subject, body: mail.body, htmlBody: mail.htmlBody });
  } catch (e) {
    if (prevSent === null) props.deleteProperty(REPORT_LAST_SENT_KEY); else props.setProperty(REPORT_LAST_SENT_KEY, prevSent);
    throw e;
  }
  props.setProperty(REPORT_LAST_IDS_KEY, JSON.stringify(data.ids));
  return 'SENT';
}

// Hourly time-driven entry point. Never throws, so a mail error cannot break other triggers.
function sendWeeklyAttendanceReportTick() {
  let lock = null;
  try {
    const cfg = getReportCfg_();
    if (!cfg.enabled) return 'DISABLED';
    const nowMs = Date.now();
    const tz = reportTz_();
    const dow = Number(Utilities.formatDate(new Date(nowMs), tz, 'u')) % 7;
    const hour = Number(Utilities.formatDate(new Date(nowMs), tz, 'H'));
    if (dow !== cfg.dow || hour < cfg.hour) return 'NOT_YET';
    if (PropertiesService.getScriptProperties().getProperty(REPORT_LAST_SENT_KEY) === reportDayKey_(nowMs)) return 'ALREADY_SENT';
    lock = LockService.getScriptLock();
    if (!lock.tryLock(10000)) return 'BUSY';
    if (PropertiesService.getScriptProperties().getProperty(REPORT_LAST_SENT_KEY) === reportDayKey_(nowMs)) return 'ALREADY_SENT';
    return sendWeeklyReportNow_(SpreadsheetApp.getActiveSpreadsheet(), cfg, nowMs);
  } catch (e) {
    try { Logger.log('Weekly report failed: ' + (e && e.message ? e.message : e)); } catch (e2) {}
    return 'ERROR';
  } finally {
    try { if (lock) lock.releaseLock(); } catch (e) {}
  }
}
