  } catch (err) {}

function normalizeDashboardFrontendUrlB5_(sheet) {
  let rawB5 = String(sheet.getRange('B5').getValue() || '').trim();
  if (/tntrung8321\.github\.io\/ClassCheck-GPS\/?$/i.test(rawB5)) {
    rawB5 = 'https://tntrung8321.github.io/ccs/';
    sheet.getRange('B5').setValue(rawB5);
  }
  return rawB5;
}

function syncDashboardVersionBadge_(ss) {
  if (!ss) return;
  const sheet = ss.getSheetByName(TAB_DASHBOARD);
  if (!sheet) return;
  normalizeDashboardFrontendUrlB5_(sheet);
  const apiUrl = String(sheet.getRange('B4').getValue() || '').trim();
  if (apiUrl && apiUrl.startsWith('https://script.google.com/')) {
    const expectedText = '✅ Verified & Ready (v' + SYSTEM_VERSION + ')';
    const currentB6 = String(sheet.getRange('B6').getValue() || '').trim();
    if (currentB6 !== expectedText) {
      sheet.getRange('B6').setValue(expectedText).setFontColor('#15803d').setFontWeight('bold');
    }
  }
}

/**
 * 2. NON-DESTRUCTIVE INITIALIZATION & SETUP VERIFICATION
 */
function initializeAndVerifySystem() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const props = PropertiesService.getScriptProperties();
  const ui = SpreadsheetApp.getUi();

  const sheetId = ss.getId();
  const boundSheetId = props.getProperty('BOUND_SHEET_ID');

  // If a colleague clicked "File -> Make a Copy", generate fresh isolated credentials
  if (boundSheetId !== sheetId) {
    props.setProperty('BOUND_SHEET_ID', sheetId);
    props.setProperty('INSTALLATION_ID', 'INST_' + generateRandomToken(10));
    props.setProperty('TEACHER_MASTER_KEY', 'TKEY_' + generateRandomToken(24));
    props.setProperty('TEMP_PC_CODE', generateNumericCode(6));
    props.setProperty('TEMP_PC_EXPIRES', String(Date.now() + 7200000));
    props.setProperty('FAILED_AUTH_COUNT', '0');
  }

  props.setProperty('FAILED_AUTH_COUNT', '0');
  props.deleteProperty('AUTH_LOCKOUT_UNTIL');

  let installId = props.getProperty('INSTALLATION_ID');
  if (!installId) {
    installId = 'INST_' + generateRandomToken(10);
    props.setProperty('INSTALLATION_ID', installId);
  }
  let teacherKey = props.getProperty('TEACHER_MASTER_KEY') || props.getProperty('TEACHER_MASTER_Key');
  if (!teacherKey) {
    teacherKey = 'TKEY_' + generateRandomToken(24);
    props.setProperty('TEACHER_MASTER_KEY', teacherKey);
  }

  ensureDashboardTab(ss, installId, teacherKey);
  ensureClassesTab(ss);
  ensureDevicesTab(ss);
  ensureSessionsTab(ss);
  ensureAttendanceTab(ss);
  ensureAuditTab(ss);

  const classesSheet = ss.getSheetByName(TAB_CLASSES);
  if (classesSheet.getLastRow() >= 2) {
    const classIds = classesSheet.getRange(2, 1, classesSheet.getLastRow() - 1, 1).getValues().flat();
    classIds.forEach(cid => {
      if (cid && !ss.getSheetByName(String(cid).trim())) {
        createFormattedRosterTab(ss, String(cid).trim(), true);
      }
    });
  }

  invalidateSystemCache();
  updateDashboardQRs(ss);

  ui.alert(
    '✅ System Initialized & Verified (v' + SYSTEM_VERSION + ')',
    'All system tabs and security credentials are ready.\n\n' +
    'NEXT STEPS:\n' +
    '1. If Cell B4 (Deployed Web App URL) is still blank, deploy this script as a Web App and paste the URL into Cell B4.\n' +
    '2. Ensure your Frontend (GitHub Pages) URL is in Cell B5.\n' +
    '3. Run this button again after updating B4/B5 to refresh your QR codes.',
    ui.ButtonSet.OK
  );
}

function updateDashboardQRs(ss) {
  const sheet = ss.getSheetByName(TAB_DASHBOARD);
  if (!sheet) return;
  const props = PropertiesService.getScriptProperties();

  const lecturerName = String(sheet.getRange('B2').getValue()).trim();
  const installId    = String(sheet.getRange('B3').getValue()).trim();
  const apiUrl       = String(sheet.getRange('B4').getValue()).trim();
  const rawFrontendUrl = normalizeDashboardFrontendUrlB5_(sheet);
  const frontendUrl  = rawFrontendUrl.replace(/\/$/, '');
  const teacherKey   = props.getProperty('TEACHER_MASTER_KEY');

  if (!apiUrl || !apiUrl.startsWith('https://script.google.com/')) {
    sheet.getRange('B6').setValue('⚠️ Waiting for valid Apps Script URL in Cell B4').setFontColor('#b45309');
    return;
  }

  sheet.getRange('B6').setValue('✅ Verified & Ready (v' + SYSTEM_VERSION + ')').setFontColor('#15803d').setFontWeight('bold');

  const studentPayload = JSON.stringify({
    type: 'ONBOARD',
    inst: installId,
    name: lecturerName,
    api: apiUrl
  });
  const studentDeepLink = frontendUrl
    ? frontendUrl + '/?onboard=' + encodeURIComponent(Utilities.base64EncodeWebSafe(studentPayload))
    : studentPayload;

  const teacherPayload = JSON.stringify({
    type: 'TEACHER_PAIR',
    inst: installId,
    name: lecturerName,
    api: apiUrl,
    tkey: teacherKey
  });
  const teacherDeepLink = frontendUrl
    ? frontendUrl + '/?pair=' + encodeURIComponent(Utilities.base64EncodeWebSafe(teacherPayload))
    : teacherPayload;

  const studentQrUrl = 'https://quickchart.io/qr?size=220&margin=2&text=' + encodeURIComponent(studentDeepLink);
  const teacherQrUrl = 'https://quickchart.io/qr?size=220&margin=2&dark=1e3a8a&text=' + encodeURIComponent(teacherDeepLink);

  sheet.getRange('A9').setFormula('=IMAGE("' + studentQrUrl + '")');
  sheet.getRange('B9').setFormula('=IMAGE("' + teacherQrUrl + '")');

  const tempCode = refreshTempPcCode(ss, false);
  sheet.getRange('C9').setValue('Temp PC Code:\n' + tempCode + '\n(Valid 2 Hours)').setFontSize(14).setFontWeight('bold');
}

function ensureSystemTabWithHeaders_(ss, tabName, headers, bgColor, options) {
  let sheet = ss.getSheetByName(tabName);
  if (!sheet) {
    sheet = ss.insertSheet(tabName);
    sheet.getRange(1, 1, 1, headers.length).setValues([headers])
      .setBackground(bgColor).setFontColor('#ffffff').setFontWeight('bold');
    sheet.setFrozenRows(1);
    if (options && options.autoResize) {
      sheet.autoResizeColumns(1, headers.length);
    }
    if (options && options.textColRange) {
      sheet.getRange(options.textColRange).setNumberFormat('@');
    }
  }
  return sheet;
}

function doGet(e) {
  const props = PropertiesService.getScriptProperties();
  return jsonResponse({
    ok: true,
    service: 'QR_Attendance_Backend',
    version: SYSTEM_VERSION,
    installationId: props.getProperty('INSTALLATION_ID') || 'UNINITIALIZED',
    serverTime: new Date().toISOString(),
    serverNowMs: Date.now()
  });
}

function doPost(e) {
  const reqArrivalMs = Date.now();
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return jsonResponse({ ok: false, code: 'ERR_EMPTY_BODY', message: 'Empty request payload.' });
    }

    const body = JSON.parse(e.postData.contents);
    const action = String(body.action || '').trim().toUpperCase();
    if (action === 'PING' && body && body.clockOnly) {
      const nowMs = Date.now();
      return jsonResponse({
        ok: true,
        code: 'PONG',
        version: SYSTEM_VERSION,
        serverTimeMs: nowMs,
        serverNowMs: nowMs,
        serverTime: new Date(nowMs).toISOString()
      });
    }
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    switch (action) {
      case 'PING':
        return handlePing(ss, body);
      case 'STUDENT_ONBOARD':
        return handleStudentOnboard(ss, body);
      case 'STUDENT_CHECKIN':
        return handleStudentCheckIn(ss, body, reqArrivalMs);
      case 'CHECK_RECEIPT':
        return handleCheckReceipt(ss, body);

      case 'TEACHER_AUTH':
        return handleTeacherAuth(ss, body);
      case 'TEACHER_GET_STATE':
        return requireTeacherAuth(ss, body, () => handleTeacherGetState(ss, body));
      case 'TEACHER_START_SESSION':
        return requireTeacherAuth(ss, body, () => handleTeacherStartSession(ss, body));
      case 'TEACHER_CONTROL_SESSION':
        return requireTeacherAuth(ss, body, () => handleTeacherControlSession(ss, body));
      case 'TEACHER_MANUAL_MARK':
        return requireTeacherAuth(ss, body, () => handleTeacherManualMark(ss, body));
      case 'TEACHER_RESET_DEVICE':
        return requireTeacherAuth(ss, body, () => handleTeacherResetDevice(ss, body));
      case 'TEACHER_ROTATE_STATIC_QR':
        return requireTeacherAuth(ss, body, () => handleTeacherRotateStaticQr(ss, body));
      case 'TEACHER_TOGGLE_REG':
        return requireTeacherAuth(ss, body, () => handleTeacherToggleReg(ss, body));
      case 'TEACHER_REBUILD_REPORT':
        return requireTeacherAuth(ss, body, () => handleTeacherRebuildReport(ss, body));
      case 'TEACHER_PUBLISH_GROUPS':
        return requireTeacherAuth(ss, body, () => handleTeacherPublishGroups(ss, body));
      case 'TEACHER_LOGOUT':
        return requireTeacherAuth(ss, body, (authRes) => {
          if (authRes && authRes.mode === 'TEMP_PC') refreshTempPcCode(ss, true);
          return jsonResponse({ ok: true });
        });

      default:
        return jsonResponse({ ok: false, code: 'ERR_UNKNOWN_ACTION', message: 'Unknown action: ' + action });
    }
  } catch (err) {
    return jsonResponse({
      ok: false,
      code: 'ERR_SERVER_EXCEPTION',
      message: err && err.message ? err.message : String(err)
    });
  }
}

/**
 * ============================================================================
 * 5. STUDENT HANDLERS (High-Concurrency Cache + Anti-Sabotage + Soft-Flag)
 * ============================================================================
 */
function normalizeAttendanceStatusCode_(val) {
  const s = String(val === undefined || val === null ? '' : val).trim().toUpperCase();
  if (s === 'P' || s === 'PRESENT' || s === 'CÓ MẶT' || s === 'CO MAT') return 'P';
  if (s === 'L' || s === 'LATE' || s === 'TRỄ' || s === 'TRE' || s === 'ĐI TRỄ' || s === 'DI TRE') return 'L';
  if (s === 'E' || s === 'EXCUSED' || s === 'CÓ PHÉP' || s === 'CO PHEP') return 'E';
  return '-';
}

function mergeAttendance(p1, p2, p2Opened) {
  const s1 = normalizeAttendanceStatusCode_(p1);
  if (p2Opened === undefined) {
    const rawP2 = String(p2 === undefined || p2 === null ? '' : p2).trim();
    if (rawP2 === '') return s1;
  } else if (!p2Opened) {
    return s1;
  }
  if (s1 === 'E') return 'E';
  const s2Raw = normalizeAttendanceStatusCode_(p2);
  if (s2Raw === 'E') return 'E';
  const s2Present = (s2Raw === 'P' || s2Raw === 'L');
  if (!s2Present) return '-';
  if (s1 === 'P') return 'P';
  return 'L';
}

function isRound2Session_(sessionId, sessionLabel) {
  const sid = String(sessionId || '').trim();
  const lbl = String(sessionLabel || '').trim();
  return /_R2$/i.test(sid) || /_P2$/i.test(lbl);
}

function getRound2PinSeed_(secret, sessionId) {
  let srvSalt = '';
  try {
    const cache = CacheService.getScriptCache();
    srvSalt = cache.get('SRV_PIN_SALT_V373') || '';
    if (!srvSalt) {
      srvSalt = PropertiesService.getScriptProperties().getProperty('TEACHER_MASTER_KEY') || 'CCS_R2_SRV_SALT';
      cache.put('SRV_PIN_SALT_V373', srvSalt, 21600);
    }
  } catch (e) {
    srvSalt = 'CCS_R2_SRV_SALT';
  }
  return sha256Hex(String(secret || '') + '|R2SEED|' + String(sessionId || '') + '|' + srvSalt).slice(0, 24);
}

function computePin4FromSeed(seed, bucket10s) {
  const str = String(seed || '') + ':' + String(bucket10s);
  let h1 = 0x811c9dc5;
  let h2 = 0x811c9dc5 ^ 0x9e3779b9;
  for (let i = 0; i < str.length; i++) {
    const c = str.charCodeAt(i);
    h1 ^= c;
    h1 = Math.imul(h1, 0x01000193);
    h2 ^= (c + i);
    h2 = Math.imul(h2, 0x01000193);
  }
  const combined = ((h1 >>> 0) ^ (h2 >>> 11)) >>> 0;
  const pinNum = combined % 10000;
  return ('0000' + pinNum).slice(-4);
}

function getTodayClassSessionsBundle_(ss, classId, todayDayKey) {
  const sheet = ss.getSheetByName(TAB_SESSIONS);
  const result = {
    p1Session: null,
    p2Session: null,
    round2OpenedToday: false,
    activeSession: null
  };
  if (!sheet || !classId || sheet.getLastRow() < 2) return result;
  const nowMs = Date.now();
  const dayKey = todayDayKey || Utilities.formatDate(new Date(nowMs), TIMEZONE, 'yyyy-MM-dd');
  const lastRow = sheet.getLastRow();
  const data = sheet.getRange(2, 1, lastRow - 1, 11).getValues();
  for (let i = 0; i < data.length; i++) {
    const cid = String(data[i][1] || '').trim();
    if (cid !== classId) continue;
    const openedAtMs = parseSheetTimeMs(data[i][3]);
    const sesDayKey = openedAtMs > 0
      ? Utilities.formatDate(new Date(openedAtMs), TIMEZONE, 'yyyy-MM-dd')
      : '';
    if (sesDayKey !== dayKey) continue;
    const presentUntilMs = parseSheetTimeMs(data[i][4]);
    const closesAtMs = parseSheetTimeMs(data[i][5]);
    const closedManual = String(data[i][6] || '').trim();
    const isOpen = !closedManual && nowMs <= closesAtMs;
    const sid = String(data[i][0] || '').trim();
    const lbl = String(data[i][2] || '').trim();
    const isR2 = isRound2Session_(sid, lbl);
    const sesObj = {
      rowIndex: i + 2,
      sessionId: sid,
      classId: cid,
      sessionLabel: lbl,
      openedAtMs: openedAtMs,
      presentUntilMs: presentUntilMs,
      closesAtMs: closesAtMs,
      closedManual: closedManual,
      isOpen: isOpen,
      roomLat: Number(data[i][7]),
      roomLng: Number(data[i][8]),
      radiusMeters: Number(data[i][9] || 100),
      qrMode: String(data[i][10] || 'STATIC'),
      round: isR2 ? 2 : 1
    };
    if (isR2) {
      result.p2Session = sesObj;
      result.round2OpenedToday = true;
    } else {
      result.p1Session = sesObj;
    }
  }
  result.activeSession = result.p2Session || result.p1Session || null;
  if (result.activeSession) {
    result.activeSession.round2OpenedToday = result.round2OpenedToday;
    result.activeSession.p1SessionId = result.p1Session ? result.p1Session.sessionId : '';
    result.activeSession.p2SessionId = result.p2Session ? result.p2Session.sessionId : '';
  }
  return result;
}

function handlePing(ss, body) {
  const cache = CacheService.getScriptCache();
  let lecturerName = cache.get('LECTURER_NAME_B2_V35');
  if (!lecturerName) {
    const dash = ss.getSheetByName(TAB_DASHBOARD);
    lecturerName = dash ? String(dash.getRange('B2').getValue()).trim() : 'Lecturer';
    try { cache.put('LECTURER_NAME_B2_V35', lecturerName || 'Lecturer', 300); } catch (e) {}
  }
  const installId = PropertiesService.getScriptProperties().getProperty('INSTALLATION_ID');
  const classes = getClassesMap(ss, false);
  const allLatestSes = getAllLatestSessionsMap(ss, false);
  const scheduleMetaMap = getStoredClassSchedulesMap();
  const publicClasses = Object.values(classes).map(c => {
    const ses = allLatestSes[c.classId];
    const isOpen = Boolean(ses && ses.isOpen);
    const isR2 = Boolean(ses && isRound2Session_(ses.sessionId, ses.sessionLabel));
    return {
      classId: c.classId,
      className: c.className,
      regOpen: c.regOpen,
      sessionOpenNow: isOpen,
      activeRound: isOpen ? (isR2 ? 2 : 1) : 0,
      activeSessionId: isOpen ? String(ses.sessionId || '') : '',
      activeQrMode: isOpen ? String(ses.qrMode || 'STATIC') : 'STATIC',
      scheduleInfo: scheduleMetaMap[c.classId] || parseClassScheduleInfoBackend(c.className),
      activeGroupSplit: getActiveGroupSplitForClass_(c.classId)
    };
  });

  const studentId = body && body.studentId ? normalizeId(body.studentId) : '';
  const studentStats = studentId
    ? getStudentAttendanceStatsMap(ss, studentId, Object.keys(classes), false)
    : undefined;

  const nowPingMs = Date.now();
  return jsonResponse({
    ok: true,
    code: 'PONG',
    version: SYSTEM_VERSION,
    installationId: installId,
    lecturerName: lecturerName,
    classes: publicClasses,
    studentStats: studentStats,
    calendarEvents: getTodayCalendarEventsCached(),
    classSchedules: scheduleMetaMap,
    serverTimeMs: nowPingMs,
    serverNowMs: nowPingMs,
    serverTime: new Date(nowPingMs).toISOString()
  });
}

function collectStudentEnrolledClasses_(ss, classes, scheduleMetaMap, studentId) {
  const enrolledClasses = [];
  let canonicalName = '';
  let anyRegOpen = false;

  Object.keys(classes).forEach(cid => {
    const roster = getClassRosterMap(ss, cid, true);
    if (roster[studentId]) {
      enrolledClasses.push({
        classId: cid,
        className: classes[cid].className,
        activeQrSecret: classes[cid].activeQrSecret,
        sessionOpenNow: Boolean(getLatestOpenSession(ss, cid)),
        scheduleInfo: scheduleMetaMap[cid] || parseClassScheduleInfoBackend(classes[cid].className),
        attendanceStats: getStudentClassAttendanceStats(ss, cid, studentId, false)
      });
      canonicalName = roster[studentId].fullName;
      if (classes[cid].regOpen) anyRegOpen = true;
    }
  });

  return {
    enrolledClasses: enrolledClasses,
    canonicalName: canonicalName,
    anyRegOpen: anyRegOpen
  };
}

function handleStudentOnboard(ss, body) {
  const studentId = normalizeId(body.studentId);
  const clientCred = String(body.credential || '').trim();

  if (!studentId || clientCred.length < 16) {
    return jsonResponse({ ok: false, code: 'ERR_INVALID_INPUT', message: 'Invalid Student ID or device credential.' });
  }

  // Anti-spam rate limit per device credential (prevents automated ID harvesting/hijacking scripts)
  const cache = CacheService.getScriptCache();
  const credHash = sha256Hex(clientCred);
  const rateKey = 'RL_ONB_' + credHash.slice(0, 16);
  const recentAttempts = Number(cache.get(rateKey) || '0');
  if (recentAttempts >= 10) {
    return jsonResponse({
      ok: false,
      code: 'ERR_SERVER_BUSY',
      message: 'Thao tác quá nhanh. Vui lòng đợi 20 giây rồi thử lại.'
    });
  }
  cache.put(rateKey, String(recentAttempts + 1), 20);

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    return jsonResponse({ ok: false, code: 'ERR_SERVER_BUSY', message: 'Server is busy. Please tap retry in 2 seconds.' });
  }

  try {
    // Always bypass cache during onboarding so changes made by external Sync.gs are reflected immediately
    const classes = getClassesMap(ss, true);
    const scheduleMetaMap = getStoredClassSchedulesMap();
    let match = collectStudentEnrolledClasses_(ss, classes, scheduleMetaMap, studentId);

    if (match.enrolledClasses.length === 0 && typeof syncSingleClassRosterFromDrive_ === 'function') {
      // Zero-lag fallback: if a newly added student types their MSSV right after the lecturer
      // updated the .XLSX/Sheet file on Drive, check only modified Drive files to auto-import them.
      let anyDriveUpdated = false;
      Object.keys(classes).forEach(cid => {
        try {
          const syncRes = syncSingleClassRosterFromDrive_(ss, cid, classes[cid].className, {
            onlyIfModified: true,
            actor: 'STUDENT_LOOKUP_RESCUE'
          });
          if (syncRes && syncRes.newStudentsAdded > 0) {
            anyDriveUpdated = true;
          }
        } catch (e) {}
      });

      if (anyDriveUpdated) {
        match = collectStudentEnrolledClasses_(ss, classes, scheduleMetaMap, studentId);
      }
    }

    const enrolledClasses = match.enrolledClasses;
    const canonicalName = match.canonicalName;
    const anyRegOpen = match.anyRegOpen;

    if (enrolledClasses.length === 0) {
      return jsonResponse({
        ok: false,
        code: 'ERR_NOT_ON_ROSTER',
        params: { id: studentId }
      });
    }

    const devSheet = ss.getSheetByName(TAB_DEVICES);
    const devData = devSheet.getLastRow() > 1
      ? devSheet.getRange(2, 1, devSheet.getLastRow() - 1, 7).getValues()
      : [];

    let existingStudentRow = -1;

    for (let i = 0; i < devData.length; i++) {
      const rowSid = normalizeId(devData[i][0]);
      const rowHash = String(devData[i][2]);
      const rowStatus = String(devData[i][4]);

      // Rule A: Has this browser credential already been bound to ANOTHER student?
      if (rowStatus === 'ACTIVE' && rowHash === credHash && rowSid !== studentId) {
        return jsonResponse({
          ok: false,
          code: 'ERR_DEVICE_BOUND_OTHER'
        });
      }
      if (rowSid === studentId) {
        existingStudentRow = i + 2;
      }
    }

    const nowIso = formatIsoNow();

    if (existingStudentRow !== -1) {
      const savedRow = devData[existingStudentRow - 2];
      const savedHash = String(savedRow[2]);
      const savedStatus = String(savedRow[4]);

      if (savedStatus === 'ACTIVE') {
        if (savedHash === credHash) {
          cache.put('DEV_HASH_' + studentId, credHash, 180);
          return jsonResponse({
            ok: true,
            code: 'ONBOARD_CONFIRMED',
            studentId: studentId,
            fullName: canonicalName,
            enrolledClasses: enrolledClasses
          });
        } else {
          // Per-Student First-Come Lock: Once registered, no 2nd browser can overwrite without Lecturer Reset
          return jsonResponse({
            ok: false,
            code: 'ERR_CREDENTIAL_MISMATCH',
            params: { id: studentId }
          });
        }
      } else {
        // Status was REVOKED by Lecturer -> allow re-bind
        devSheet.getRange(existingStudentRow, 2, 1, 4).setValues([[canonicalName, credHash, nowIso, 'ACTIVE']]);
        cache.put('DEV_HASH_' + studentId, credHash, 180);
        return jsonResponse({
          ok: true,
          code: 'ONBOARD_CONFIRMED',
          studentId: studentId,
          fullName: canonicalName,
          enrolledClasses: enrolledClasses
        });
      }
    }

    if (!anyRegOpen && !isAnySessionOpenForClasses(ss, enrolledClasses.map(c => c.classId))) {
      return jsonResponse({
        ok: false,
        code: 'ERR_REG_LOCKED'
      });
    }

    devSheet.appendRow([studentId, canonicalName, credHash, nowIso, 'ACTIVE', 0, '']);
    cache.put('DEV_HASH_' + studentId, credHash, 21600);

    return jsonResponse({
      ok: true,
      code: 'ONBOARD_CONFIRMED',
      studentId: studentId,
      fullName: canonicalName,
      enrolledClasses: enrolledClasses
    });
  } finally {
    try { cache.remove(spamKey); } catch (e) {}
    lock.releaseLock();
  }
}

function handleStudentCheckIn(ss, body, reqArrivalMs) {
  const arrivalMs  = (typeof reqArrivalMs === 'number' && reqArrivalMs > 0) ? reqArrivalMs : Date.now();
  const studentId  = normalizeId(body.studentId);
  let classId      = String(body.classId || '').trim();
  const clientCred = String(body.credential || '').trim();
  const deviceFp   = String(body.deviceFp || '').trim();
  const qrPayload  = String(body.qrPayload || 'BUTTON_GPS_CHECKIN').trim();
  const requestId  = String(body.requestId || '').trim() || ('REQ_' + generateRandomToken(12));
  const lat        = Number(body.lat);
  const lng        = Number(body.lng);
  const accuracy   = Number(body.accuracy || 999);
  const clientMeta = body.clientMeta || {};

  if (!studentId || !clientCred || !qrPayload) {
    return jsonResponse({ ok: false, code: 'ERR_MISSING_FIELDS' });
  }

  const cache = CacheService.getScriptCache();
  const credHash = sha256Hex(clientCred);

  // 0A. FAST PRE-LOCK IDEMPOTENCY CHECK (Returns in ~2ms if double-tapped)
  const cachedByReq = cache.get('REQ_REC_' + requestId);
  if (cachedByReq) {
    try {
      const rec = JSON.parse(cachedByReq);
      return jsonResponse({
        ok: true,
        code: 'CONFIRMED_' + (rec.status === 'P' ? 'PRESENT' : 'LATE'),
        receipt: rec
      });
    } catch (e) {}
  }

  // 0B. ANTI-SPAM RATE LIMIT BEFORE LOCKSERVICE (Allows retries with the same requestId)
  const spamKey = 'RL_CHK_' + credHash.slice(0, 16);
  const lastSpamReq = cache.get(spamKey);
  if (lastSpamReq && lastSpamReq !== requestId) {
    return jsonResponse({
      ok: false,
      code: 'ERR_SERVER_BUSY',
      message: 'Vui lòng chờ 2 giây giữa các lần bấm.'
    });
  }
  cache.put(spamKey, requestId, 2);

  // 1. Validate Class & Roster using row-aware micro-cache (auto-updates if Sync.gs changed rows)
  const classes = getClassesMap(ss, false);
  let classConfig = classId ? classes[classId] : null;
  let roster = classConfig ? getClassRosterMap(ss, classId, false) : null;
  let activeSession = classConfig ? getLatestOpenSession(ss, classId) : null;

  // Smart fallback: if the submitted classId has no open session (or wasn't selected yet),
  // check if the student is on the roster of a class that currently has an open session.
  if (!classConfig || !roster || !roster[studentId] || !activeSession) {
    const enrolledOpen = [];
    const enrolledAny = [];
    Object.keys(classes).forEach(cid => {
      const rMap = getClassRosterMap(ss, cid, false);
      if (rMap && rMap[studentId]) {
        enrolledAny.push({ cid: cid, rMap: rMap });
        const openSes = getLatestOpenSession(ss, cid);
        if (openSes) {
          enrolledOpen.push({ cid: cid, ses: openSes, rMap: rMap });
        }
      }
    });
    if (enrolledOpen.length === 1) {
      classId = enrolledOpen[0].cid;
      classConfig = classes[classId];
      roster = enrolledOpen[0].rMap;
      activeSession = enrolledOpen[0].ses;
    } else if ((!classConfig || !roster || !roster[studentId]) && enrolledAny.length === 1) {
      classId = enrolledAny[0].cid;
      classConfig = classes[classId];
      roster = enrolledAny[0].rMap;
      activeSession = getLatestOpenSession(ss, classId);
    }
  }

  if (!classConfig) {
    cache.remove(spamKey);
    return jsonResponse({ ok: false, code: 'ERR_INVALID_CLASS' });
  }

  if (!roster || !roster[studentId]) {
    cache.remove(spamKey);
    return jsonResponse({ ok: false, code: 'ERR_NOT_ON_ROSTER', params: { id: studentId } });
  }
  const canonicalName = roster[studentId].fullName;

  if (!activeSession) {
    cache.remove(spamKey);
    return jsonResponse({ ok: false, code: 'ERR_SESSION_CLOSED' });
  }

  // Fast Session + Student Idempotency Check in Cache before LockService
  const sesStuKey = 'SES_ATT_' + activeSession.sessionId + '_' + studentId;
  const cachedSesRec = cache.get(sesStuKey);
  if (cachedSesRec) {
    try {
      const rec = JSON.parse(cachedSesRec);
      if (rec && rec.status === '-' && rec.source === 'TEACHER_MANUAL') {
        return jsonResponse({
          ok: false,
          code: 'ERR_REVOKED_BY_TEACHER',
          message: '[ĐÃ HỦY] Kết quả điểm danh của bạn trong phiên này đã bị Giảng viên hủy khi kiểm tra sĩ số.'
        });
      }
      if (rec && rec.status && rec.status !== '-') {
        const canUpgradeLate = rec.status === 'L' && rec.source !== 'TEACHER_MANUAL' && (isRound2Session_(activeSession.sessionId, activeSession.sessionLabel) || Date.now() <= activeSession.presentUntilMs);
        if (!canUpgradeLate) {
          try { cache.remove(spamKey); } catch (e) {}
          if (!rec.attendanceStats) {
            rec.attendanceStats = getStudentClassAttendanceStats(ss, classId, studentId, false);
          }
          return jsonResponse({
            ok: true,
            code: rec.requestId === requestId ? ('CONFIRMED_' + (rec.status === 'P' ? 'PRESENT' : 'LATE')) : 'ALREADY_RECORDED',
            receipt: rec
          });
        }
      }
    } catch (e) {}
  }

  if (activeSession.qrMode === 'PIN_10S' && qrPayload === 'BUTTON_GPS_CHECKIN') {
    cache.remove(spamKey);
    return jsonResponse({
      ok: false,
      code: 'ERR_PIN_REQUIRED',
      round: isRound2Session_(activeSession.sessionId, activeSession.sessionLabel) ? 2 : 1,
      sessionId: activeSession.sessionId,
      qrMode: 'PIN_10S'
    });
  }
  const qrValid = verifyClassroomQr(qrPayload, classConfig, activeSession, arrivalMs, ss, studentId);
  if (!qrValid) {
    cache.remove(spamKey);
    const pinEval = (activeSession && activeSession._lastPinEval) ? activeSession._lastPinEval : null;
    if (activeSession.qrMode === 'PIN_10S' && pinEval && pinEval.code !== 'PIN_LOCKED') {
      logPinCheckAttempt_(ss, activeSession, classConfig, studentId, pinEval.cleanedPin, pinEval.serverBucket, pinEval.studentBucket, pinEval.offsetIntoBucketMs, pinEval.deltaMs, pinEval.graceMs, pinEval.code);
    }
    const pinErrCode = (pinEval && pinEval.code)
      ? pinEval.code
      : 'PIN_WRONG';
    return jsonResponse({
      ok: false,
      code: activeSession.qrMode === 'PIN_10S' ? pinErrCode : 'ERR_INVALID_QR',
      params: pinEval && pinEval.params ? pinEval.params : undefined,
      message: pinEval && pinEval.message ? pinEval.message : undefined,
      pinEval: pinEval || undefined
    });
  }

  // 4. Verify Geolocation Proximity (Accuracy-Aware Haversine + Lecturer GPS / 3-Student Average Anchor)
  if (isNaN(lat) || isNaN(lng) || lat === 0 || lng === 0 || accuracy <= 0) {
    return jsonResponse({ ok: false, code: 'ERR_GPS_UNAVAILABLE' });
  }

  const anchorKey = 'SES_ANCHOR_' + activeSession.sessionId;
  let anchorState = null;
  try {
    const rawAnchor = cache.get(anchorKey);
    if (rawAnchor) anchorState = JSON.parse(rawAnchor);
  } catch (e) {}

  if (anchorState && anchorState.lat && anchorState.lng) {
    activeSession.roomLat = Number(anchorState.lat);
    activeSession.roomLng = Number(anchorState.lng);
  }

  const accuracyAllowance = Math.min(Math.max(0, accuracy), 50);
  let rawDist = Math.round(calculateHaversineMeters(lat, lng, activeSession.roomLat, activeSession.roomLng));
  let effectiveDist = Math.max(0, rawDist - accuracyAllowance);

  if (anchorState && anchorState.mode === 'COLLECTING_AVG_3') {
    // Once at least 1 student has anchored the provisional average, subsequent students must be within radiusMeters of the running average
    if (Array.isArray(anchorState.samples) && anchorState.samples.length > 0 && effectiveDist > activeSession.radiusMeters) {
      cache.remove(spamKey);
      return jsonResponse({
        ok: false,
        code: 'ERR_LOCATION_OUT_OF_RANGE',
        params: {
          dist: rawDist,
          max: activeSession.radiusMeters,
          acc: Math.round(accuracy)
        }
      });
    }
  } else if (effectiveDist > activeSession.radiusMeters) {
    cache.remove(spamKey);
    return jsonResponse({
      ok: false,
      code: 'ERR_LOCATION_OUT_OF_RANGE',
      params: {
        dist: rawDist,
        max: activeSession.radiusMeters,
        acc: Math.round(accuracy)
      }
    });
  }

  // 5. Acquire LockService ONLY for final credential check & appendRow
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(12000)) {
    cache.remove(spamKey);
    return jsonResponse({ ok: false, code: 'ERR_SERVER_BUSY' });
  }

  try {
    const credCheck = verifyStudentCredential(ss, studentId, clientCred, credHash, canonicalName);
    if (!credCheck.ok) {
      return jsonResponse({ ok: false, code: credCheck.code, params: { id: studentId } });
    }

    // Re-check dynamic session anchor inside lock: if in COLLECTING_AVG_3 mode, average the first 3 verified students
    try {
      const freshAnchorRaw = cache.get(anchorKey);
      if (freshAnchorRaw) anchorState = JSON.parse(freshAnchorRaw);
    } catch (e) {}

    if (anchorState && anchorState.mode === 'COLLECTING_AVG_3') {
      if (!Array.isArray(anchorState.samples)) anchorState.samples = [];
      if (anchorState.samples.length > 0 && anchorState.lat && anchorState.lng) {
        activeSession.roomLat = Number(anchorState.lat);
        activeSession.roomLng = Number(anchorState.lng);
        rawDist = Math.round(calculateHaversineMeters(lat, lng, activeSession.roomLat, activeSession.roomLng));
        effectiveDist = Math.max(0, rawDist - accuracyAllowance);
        if (effectiveDist > activeSession.radiusMeters) {
          return jsonResponse({
            ok: false,
            code: 'ERR_LOCATION_OUT_OF_RANGE',
            params: {
              dist: rawDist,
              max: activeSession.radiusMeters,
              acc: Math.round(accuracy)
            }
          });
        }
      }

      if (accuracy <= 100 && !(clientMeta && clientMeta.mockGpsSuspect)) {
        const alreadySampled = anchorState.samples.some(s => s.studentId === studentId);
        if (!alreadySampled && anchorState.samples.length < 3) {
          anchorState.samples.push({ studentId: studentId, lat: lat, lng: lng });
        }
        const n = anchorState.samples.length;
        if (n > 0) {
          const sumLat = anchorState.samples.reduce((acc, item) => acc + Number(item.lat), 0);
          const sumLng = anchorState.samples.reduce((acc, item) => acc + Number(item.lng), 0);
          const avgLat = Number((sumLat / n).toFixed(6));
          const avgLng = Number((sumLng / n).toFixed(6));
          anchorState.lat = avgLat;
          anchorState.lng = avgLng;
          if (n >= 3) {
            anchorState.mode = 'LOCKED';
            anchorState.source = 'STUDENT_AVG_3';
          }
          try { cache.put(anchorKey, JSON.stringify(anchorState), 7200); } catch (e) {}
          activeSession.roomLat = avgLat;
          activeSession.roomLng = avgLng;
          rawDist = Math.round(calculateHaversineMeters(lat, lng, avgLat, avgLng));
          effectiveDist = Math.max(0, rawDist - accuracyAllowance);
          try {
            if (activeSession.rowIndex) {
              ss.getSheetByName(TAB_SESSIONS).getRange(activeSession.rowIndex, 8, 1, 2).setValues([[avgLat, avgLng]]);
            }
            if (n >= 3 && classConfig && classConfig.rowIndex) {
              ss.getSheetByName(TAB_CLASSES).getRange(classConfig.rowIndex, 4, 1, 2).setValues([[avgLat, avgLng]]);
            }
            cache.removeAll(['ALL_LATEST_SES_V373', 'ALL_LATEST_SES_V35']);
          } catch (e) {}
        }
      } else if (!anchorState.samples.length || effectiveDist > activeSession.radiusMeters) {
        return jsonResponse({
          ok: false,
          code: 'ERR_LOCATION_OUT_OF_RANGE',
          params: {
            dist: rawDist,
            max: activeSession.radiusMeters,
            acc: Math.round(accuracy)
          }
        });
      }
    } else if (anchorState && anchorState.mode === 'LOCKED' && anchorState.lat && anchorState.lng) {
      activeSession.roomLat = Number(anchorState.lat);
      activeSession.roomLng = Number(anchorState.lng);
      rawDist = Math.round(calculateHaversineMeters(lat, lng, activeSession.roomLat, activeSession.roomLng));
      effectiveDist = Math.max(0, rawDist - accuracyAllowance);
      if (effectiveDist > activeSession.radiusMeters) {
        return jsonResponse({
          ok: false,
          code: 'ERR_LOCATION_OUT_OF_RANGE',
          params: {
            dist: rawDist,
            max: activeSession.radiusMeters,
            acc: Math.round(accuracy)
          }
        });
      }
    }

    const existingRecord = findExistingAttendance(ss, activeSession.sessionId, studentId, requestId);
    if (existingRecord) {
      if (existingRecord.status === '-' && existingRecord.source === 'TEACHER_MANUAL') {
        cache.put(sesStuKey, JSON.stringify(existingRecord), 1800);
        return jsonResponse({
          ok: false,
          code: 'ERR_REVOKED_BY_TEACHER',
          message: '[ĐÃ HỦY] Kết quả điểm danh của bạn trong phiên này đã bị Giảng viên hủy khi kiểm tra sĩ số.'
        });
      }
      if (existingRecord.status !== '-') {
        const canUpgradeLate = existingRecord.status === 'L' && existingRecord.source !== 'TEACHER_MANUAL' && (isRound2Session_(activeSession.sessionId, activeSession.sessionLabel) || Date.now() <= activeSession.presentUntilMs);
        if (!canUpgradeLate) {
          if (!existingRecord.attendanceStats) {
            existingRecord.attendanceStats = getStudentClassAttendanceStats(ss, classId, studentId, false);
          }
          cache.put(sesStuKey, JSON.stringify(existingRecord), 1800);
          return jsonResponse({
            ok: true,
            code: existingRecord.requestId === requestId ? ('CONFIRMED_' + (existingRecord.status === 'P' ? 'PRESENT' : 'LATE')) : 'ALREADY_RECORDED',
            receipt: existingRecord
          });
        }
      }
    }

    // 6. Server-Side Anti-Cheat & Multi-Browser (Safari + Chrome) Soft-Flag Evaluation
    const nowMs = Date.now();
    let sourceLabel = 'STUDENT_APP';
    const suspectReasons = [];

    const isRound2CheckIn = isRound2Session_(activeSession.sessionId, activeSession.sessionLabel);
    if (clientMeta && clientMeta.mockGpsSuspect) {
      suspectReasons.push('SUSPECT_WITH_DEVTOOLS|Can thiệp định vị (' + (clientMeta.mockReason || 'Mock') + ')');
    }
    if (isRound2CheckIn && activeSession.qrMode === 'PIN_10S' && activeSession._lastPinEval && clientMeta) {
      const firstKeyIdleMs = Number(clientMeta.firstKeyIdleMs || 0);
      const pinAgeMs = Number(activeSession._lastPinEval.pinAgeMs || 0);
      if (firstKeyIdleMs >= 5000 && pinAgeMs >= 6500) {
        suspectReasons.push(
          'SUSPECT_PIN_RELAY|Nghi vấn nhận PIN qua tin nhắn (Chờ ' +
          (firstKeyIdleMs / 1000).toFixed(1) +
          's, nộp cuối chu kỳ ' +
          (pinAgeMs / 1000).toFixed(1) +
          's)'
        );
      }
    }

    const recentKey = 'RECENT_CHK_' + activeSession.sessionId;
    let recentList = [];
    try {
      const rawRecent = cache.get(recentKey);
      if (rawRecent) recentList = JSON.parse(rawRecent);
    } catch (e) {
      recentList = [];
    }

    recentList = recentList.filter(item => (nowMs - item.timeMs) <= 45000 && item.studentId !== studentId);

    for (let i = 0; i < recentList.length; i++) {
      const prev = recentList[i];
      const diffSec = Math.max(0, Math.round((nowMs - prev.timeMs) / 1000));
      const sameExactCoords = Math.abs(prev.lat - lat) < 0.000001 && Math.abs(prev.lng - lng) < 0.000001;
      const sameHardwareFp  = Boolean(deviceFp && prev.deviceFp && deviceFp === prev.deviceFp && deviceFp !== 'FP_DEFAULT');
      const pairDist        = calculateHaversineMeters(lat, lng, prev.lat, prev.lng);

      if (sameExactCoords) {
        suspectReasons.push('SUSPECT_WITH_' + prev.studentId + '|Trùng tọa độ tuyệt đối với ' + prev.studentId);
        break;
      } else if (sameHardwareFp && diffSec <= 20 && pairDist <= 5) {
        suspectReasons.push('SUSPECT_WITH_' + prev.studentId + '|Cùng cấu hình máy (' + diffSec + 's) với ' + prev.studentId);
        break;
      }
    }

    if (suspectReasons.length > 0) {
      sourceLabel = suspectReasons[0];
    }

    recentList.push({
      studentId: studentId,
      fullName: canonicalName,
      deviceFp: deviceFp,
      lat: lat,
      lng: lng,
      rawDist: rawDist,
      timeMs: nowMs
    });
    if (recentList.length > 40) recentList = recentList.slice(-40);
    try { cache.put(recentKey, JSON.stringify(recentList), 600); } catch (e) {}

    const status = isRound2CheckIn ? 'P' : (nowMs <= activeSession.presentUntilMs ? 'P' : 'L');
    const timeStr = Utilities.formatDate(new Date(nowMs), TIMEZONE, 'HH:mm:ss');
    const attendanceId = 'ATT_' + generateRandomToken(10);

    const attSheet = ss.getSheetByName(TAB_ATTEND);
    attSheet.appendRow([
      attendanceId,
      requestId,
      activeSession.sessionId,
      classId,
      studentId,
      status,
      formatIso(nowMs),
      rawDist,
      Math.round(accuracy),
      sourceLabel
    ]);

    updateSessionAttendanceMapCache_(activeSession.sessionId, studentId, {
      rowIndex: 0,
      attendanceId: attendanceId,
      requestId: requestId,
      sessionId: activeSession.sessionId,
      classId: classId,
      studentId: studentId,
      status: status,
      time: timeStr,
      distance: rawDist,
      accuracy: Math.round(accuracy),
      source: sourceLabel
    });

    const updatedStats = updateStudentClassAttendanceStatsCacheInPlace_(classId, activeSession, studentId, status, isRound2CheckIn);

    const receipt = {
      attendanceId: attendanceId,
      requestId: requestId,
      sessionId: activeSession.sessionId,
      sessionLabel: activeSession.sessionLabel,
      round: isRound2CheckIn ? 2 : 1,
      classId: classId,
      studentId: studentId,
      fullName: canonicalName,
      status: status,
      time: timeStr,
      distanceMeters: rawDist,
      accuracyMeters: Math.round(accuracy),
      source: sourceLabel,
      attendanceStats: updatedStats
    };

    try {
      cache.put(sesStuKey, JSON.stringify(receipt), 1800);
      cache.put('REQ_REC_' + requestId, JSON.stringify(receipt), 600);
    } catch (e) {}

    if (activeSession.qrMode === 'PIN_10S' && activeSession._lastPinEval) {
      const pe = activeSession._lastPinEval;
      logPinCheckAttempt_(ss, activeSession, classConfig, studentId, pe.cleanedPin, pe.serverBucket, pe.studentBucket, pe.offsetIntoBucketMs, pe.deltaMs, pe.graceMs, pe.code);
    }

    return jsonResponse({
      ok: true,
      code: status === 'P' ? 'CONFIRMED_PRESENT' : 'CONFIRMED_LATE',
      receipt: receipt
    });
  } finally {
    try { cache.remove(spamKey); } catch (e) {}
    lock.releaseLock();
  }
}

function handleCheckReceipt(ss, body) {
  const requestId = String(body.requestId || '').trim();
  const studentId = normalizeId(body.studentId);
  if (!requestId || !studentId) {
    return jsonResponse({ ok: false, code: 'ERR_MISSING_FIELDS' });
  }

  const cache = CacheService.getScriptCache();
  const cachedByReq = cache.get('REQ_REC_' + requestId);
  if (cachedByReq) {
    try {
      return jsonResponse({
        ok: true,
        code: 'RECEIPT_FOUND',
        receipt: JSON.parse(cachedByReq)
      });
    } catch (e) {}
  }

  const attSheet = ss.getSheetByName(TAB_ATTEND);
  if (attSheet.getLastRow() < 2) {
    return jsonResponse({ ok: false, code: 'ERR_NOT_FOUND' });
  }
  const data = attSheet.getRange(2, 1, attSheet.getLastRow() - 1, 10).getValues();
  for (let i = data.length - 1; i >= 0; i--) {
    if (String(data[i][1]) === requestId && normalizeId(data[i][4]) === studentId) {
      return jsonResponse({
        ok: true,
        code: 'RECEIPT_FOUND',
        receipt: {
          attendanceId: data[i][0],
          requestId: data[i][1],
          sessionId: data[i][2],
          classId: data[i][3],
          studentId: data[i][4],
          status: data[i][5],
          time: String(data[i][6]).slice(11, 19),
          distanceMeters: data[i][7]
        }
      });
    }
  }
  return jsonResponse({ ok: false, code: 'ERR_NOT_FOUND' });
}

function requireTeacherAuth(ss, body, callback) {

function getAuthClientKey_(body) {
  const raw = String((body && (body.credential || body.deviceFp)) || 'ANON').trim();
  return sha256Hex('TEA_AUTH|' + raw).slice(0, 16);
}

function verifyTeacherToken(ss, submittedToken, body) {
  const props = PropertiesService.getScriptProperties();
  const cache = CacheService.getScriptCache();
  const cleanToken = String(submittedToken || '').trim();
  if (!cleanToken) {
    return { ok: false, code: 'ERR_UNAUTHORIZED', message: 'Vui lòng nhập mật khẩu Giảng viên.' };
  }
  const clientKey = getAuthClientKey_(body);
  const failKey = 'AUTH_FAIL_' + clientKey;
  const lockKey = 'AUTH_LOCK_' + clientKey;

  // 1. Master Key ALWAYS works (bypasses and clears any 10-min lockout)
  const masterKey = props.getProperty('TEACHER_MASTER_KEY') || props.getProperty('TEACHER_MASTER_Key');
  if (masterKey && cleanToken === masterKey) {
    cache.removeAll([failKey, lockKey, 'AUTH_LOCKOUT_UNTIL']);
    if (props.getProperty('AUTH_LOCKOUT_UNTIL')) {
      props.deleteProperty('AUTH_LOCKOUT_UNTIL');
    }
    return { ok: true, mode: 'MASTER' };
  }

  // 2. Check per-device 10-minute brute-force lockout for 6-digit code attempts
  const lockoutUntil = Number(cache.get(lockKey) || '0');
  if (Date.now() < lockoutUntil) {
    const waitMins = Math.ceil((lockoutUntil - Date.now()) / 60000);
    return {
      ok: false,
      code: 'ERR_AUTH_LOCKED',
      message: '[TẠM KHÓA] Nhập sai quá 5 lần. Tạm khóa mã 6 số trong ' + waitMins + ' phút (Dùng Master Key hoặc đổi mã mới trên Sheet để vào ngay).'
    };
  }

  // 3. Fast check against ScriptProperties TEMP_PC_CODE first (avoids reading Setup_&_Dashboard on every 2.5s poll)
  const tempCode = String(props.getProperty('TEMP_PC_CODE') || '').trim();
  const tempExp  = Number(props.getProperty('TEMP_PC_EXPIRES') || '0');
  const isTempValid = !tempExp || Date.now() <= tempExp;

  if (isTempValid && tempCode && cleanToken === tempCode) {
    cache.remove(failKey);
    return { ok: true, mode: 'TEMP_PC' };
  }

  // Fallback: Read active 6-digit password from Cell C9 of Setup_&_Dashboard (cached briefly for 15s)
  let sheetCode = String(cache.get('DASH_C9_CODE_V35') || '');
  if (!sheetCode) {
    const dashSheet = ss.getSheetByName(TAB_DASHBOARD);
    if (dashSheet) {
      const c9Text = String(dashSheet.getRange('C9').getValue() || '').trim();
      const match6 = c9Text.match(/\b(\d{6})\b/);
      if (match6) {
        sheetCode = match6[1];
      } else if (c9Text && c9Text.indexOf('Attendance Admin') === -1) {
        sheetCode = c9Text;
      }
      if (sheetCode) {
        try { cache.put('DASH_C9_CODE_V35', sheetCode, 15); } catch (e) {}
      }
    }
  }

  // Accept if submitted token matches visible C9 code (while not expired)
  if (isTempValid && sheetCode && cleanToken === sheetCode) {
    cache.remove(failKey);
    return { ok: true, mode: 'TEMP_PC' };
  }

  // 4. Record per-device failed attempt & lock this device for 10 mins on 5th consecutive wrong try
  const failedCount = Number(cache.get(failKey) || '0');
  const newFailed = failedCount + 1;
  cache.put(failKey, String(newFailed), 600);
  if (newFailed >= 5) {
    const lockUntilMs = String(Date.now() + 10 * 60 * 1000);
    cache.put(lockKey, lockUntilMs, 600);
    logAudit(ss, 'UNKNOWN', 'TEACHER_AUTH_LOCKOUT', clientKey, '5 failed teacher auth attempts on device');
    return {
      ok: false,
      code: 'ERR_AUTH_LOCKED',
      message: '[ĐÃ KHÓA] Nhập sai 5/5 lần. Đã khóa mã 6 số trong 10 phút (Dùng Master Key hoặc đổi mã mới trên Sheet để vào ngay).'
    };
  }

  return {
    ok: false,
    code: 'ERR_UNAUTHORIZED',
    message: '[TỪ CHỐI] Mật khẩu Giảng viên không đúng (' + newFailed + '/5 lần thử).'
  };
}

  try { syncDashboardVersionBadge_(ss); } catch (e) {}

}

function splitVietnameseNameParts_(fullName) {
  const clean = String(fullName || '').trim().replace(/\s+/g, ' ');
  if (!clean) return { givenName: '', familyMiddle: '' };
  const parts = clean.split(' ');
  return {
    givenName: parts[parts.length - 1] || '',
    familyMiddle: parts.slice(0, -1).join(' ')
  };
}

function compareStudentsVietnameseAZ_(a, b) {
  const na = splitVietnameseNameParts_(a && a.fullName);
  const nb = splitVietnameseNameParts_(b && b.fullName);
  const cmpGiven = na.givenName.localeCompare(nb.givenName, 'vi', { sensitivity: 'base' });
  if (cmpGiven !== 0) return cmpGiven;
  const cmpAccent = na.givenName.localeCompare(nb.givenName, 'vi');
  if (cmpAccent !== 0) return cmpAccent;
  const cmpFam = na.familyMiddle.localeCompare(nb.familyMiddle, 'vi', { sensitivity: 'base' });
  if (cmpFam !== 0) return cmpFam;
  return String((a && a.studentId) || '').localeCompare(String((b && b.studentId) || ''), 'vi');
}

function handleTeacherGetState(ss, body) {
  const classId = String(body.classId || '').trim();
  const classes = getClassesMap(ss, false);
  const scheduleMetaMap = getStoredClassSchedulesMap();
  const allLatestSes = getAllLatestSessionsMap(ss, false);
  const classList = Object.values(classes).map(c => Object.assign({}, c, {
    sessionOpenNow: Boolean(allLatestSes[c.classId] && allLatestSes[c.classId].isOpen),
    scheduleInfo: scheduleMetaMap[c.classId] || parseClassScheduleInfoBackend(c.className)
  }));
  const selectedId = classId && classes[classId] ? classId : (classList[0] ? classList[0].classId : '');

  let sessionInfo = null;
  let rosterSummary = [];
  let round2OpenedToday = false;

  if (selectedId) {
    const rosterMap = getClassRosterMap(ss, selectedId, false);
    const classStatsMap = getClassAttendanceStatsMap_(ss, selectedId, false);
    const todayDayKey = Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd');
    const todayBundle = getTodayClassSessionsBundle_(ss, selectedId, todayDayKey);
    round2OpenedToday = Boolean(todayBundle.round2OpenedToday);
    sessionInfo = todayBundle.activeSession || allLatestSes[selectedId] || getLatestSessionForDashboard(ss, selectedId, false);

    let isSessionDayValid = false;
    let isTodaySession = false;
    if (sessionInfo) {
      if (sessionInfo.isOpen || round2OpenedToday) {
        isSessionDayValid = true;
        isTodaySession = true;
      } else {
        const sesDayKey = sessionInfo.openedAtMs > 0
          ? Utilities.formatDate(new Date(sessionInfo.openedAtMs), TIMEZONE, 'yyyy-MM-dd')
          : '';
        const sampleSt = Object.values(classStatsMap)[0];
        if (sesDayKey && sampleSt && Array.isArray(sampleSt.history)) {
          isSessionDayValid = sampleSt.history.some(h => h && h.dayKey === sesDayKey);
        }
        isTodaySession = Boolean(sesDayKey && sesDayKey === todayDayKey && (isSessionDayValid || !sampleSt));
      }
    }

    if (!isTodaySession) {
      sessionInfo = null;
      round2OpenedToday = false;
    } else if (sessionInfo) {
      sessionInfo.round = isRound2Session_(sessionInfo.sessionId, sessionInfo.sessionLabel) ? 2 : 1;
      sessionInfo.round2OpenedToday = round2OpenedToday;
      if (sessionInfo.round === 2 && classes[selectedId]) {
        sessionInfo.pin10sSeed = getRound2PinSeed_(classes[selectedId].activeQrSecret, sessionInfo.sessionId); sessionInfo.pinSeed = sessionInfo.pin10sSeed;
      }
    }

    const p1Checks = (isTodaySession && todayBundle.p1Session)
      ? getSessionAttendanceMap(ss, todayBundle.p1Session.sessionId, false)
      : ((sessionInfo && isTodaySession && !round2OpenedToday) ? getSessionAttendanceMap(ss, sessionInfo.sessionId, false) : {});
    const p2Checks = (isTodaySession && todayBundle.p2Session)
      ? getSessionAttendanceMap(ss, todayBundle.p2Session.sessionId, false)
      : {};

    rosterSummary = Object.values(rosterMap).map(st => {
      const rec1 = p1Checks[st.studentId] || null;
      const rec2 = p2Checks[st.studentId] || null;
      const stStats = classStatsMap[st.studentId] || null;
      const p1Status = (rec1 && rec1.status) ? rec1.status : '-';
      const p2Status = round2OpenedToday
        ? ((rec2 && (rec2.status === 'P' || rec2.status === 'L')) ? 'P' : (rec2 && rec2.status === 'E' ? 'E' : '-'))
        : '';
      const finalStatus = isTodaySession
        ? mergeAttendance(p1Status, p2Status, round2OpenedToday)
        : '-';
      const activeRec = (round2OpenedToday && rec2) ? rec2 : (rec1 || null);
      return {
        studentId: st.studentId,
        fullName: st.fullName,
        status: finalStatus,
        finalStatus: finalStatus,
        p1Status: isTodaySession ? p1Status : '-',
        p2Status: isTodaySession ? p2Status : '',
        p2Opened: round2OpenedToday,
        p1Time: rec1 ? rec1.time : '',
        p1Distance: rec1 && rec1.distance !== undefined ? rec1.distance : '',
        p1Source: rec1 ? rec1.source : '',
        p2Time: rec2 ? rec2.time : '',
        p2Distance: rec2 && rec2.distance !== undefined ? rec2.distance : '',
        p2Source: rec2 ? rec2.source : '',
        time: (rec2 && rec2.time) ? rec2.time : (rec1 ? rec1.time : ''),
        distance: activeRec && activeRec.distance !== undefined ? activeRec.distance : '',
        accuracy: activeRec && activeRec.accuracy !== undefined ? activeRec.accuracy : '',
        source: activeRec ? activeRec.source : '',
        attendanceStats: stStats
      };
    }).sort(compareStudentsVietnameseAZ_);
  }

  return jsonResponse({
    ok: true,
    code: 'TEACHER_STATE',
    serverTimeMs: Date.now(),
    serverNowMs: Date.now(),
    classes: classList,
    calendarEvents: getTodayCalendarEventsCached(),
    classSchedules: scheduleMetaMap,
    selectedClassId: selectedId,
    round2OpenedToday: round2OpenedToday,
    activeSession: sessionInfo,
    activeGroupSplit: selectedId ? getActiveGroupSplitForClass_(selectedId) : null,
    roster: rosterSummary
  });
}

function handleTeacherStartSession(ss, body) {
  const classId       = String(body.classId || '').trim();
  const qrMode        = String(body.qrMode || 'STATIC').toUpperCase();
  const usePhoneGps   = Boolean(body.usePhoneGps);
  const calibrateOnly = Boolean(body.calibrateOnly);
  const phoneLat      = Number(body.lat);
  const phoneLng      = Number(body.lng);
  const anchorMode    = String(body.anchorMode || '').toUpperCase();

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) return jsonResponse({ ok: false, code: 'ERR_SERVER_BUSY' });

  try {
    const cache = CacheService.getScriptCache();
    const classes = getClassesMap(ss, true);
    const cfg = classes[classId];
    if (!cfg) return jsonResponse({ ok: false, code: 'ERR_INVALID_CLASS' });

    const existingOpen = getLatestOpenSession(ss, classId);
    const hasTeacherGps = usePhoneGps && !isNaN(phoneLat) && !isNaN(phoneLng) && phoneLat !== 0 && anchorMode !== 'STUDENT_AVG_3' && anchorMode !== 'CAMPUS_CLUSTER';

    // If lecturer clicked legacy calibrate without opening a new session
    if (calibrateOnly && !existingOpen) {
      if (!isNaN(phoneLat) && !isNaN(phoneLng) && phoneLat !== 0) {
        ss.getSheetByName(TAB_CLASSES).getRange(cfg.rowIndex, 4, 1, 2).setValues([[phoneLat, phoneLng]]);
        const latestSes = getLatestSessionForDashboard(ss, classId, true);
        if (latestSes && latestSes.rowIndex) {
          ss.getSheetByName(TAB_SESSIONS).getRange(latestSes.rowIndex, 8, 1, 2).setValues([[phoneLat, phoneLng]]);
        }
        invalidateClassCache(classId);
      }
      return handleTeacherGetState(ss, { classId: classId });
    }

    if (existingOpen) {
      if (hasTeacherGps) {
        ss.getSheetByName(TAB_SESSIONS).getRange(existingOpen.rowIndex, 8, 1, 2).setValues([[phoneLat, phoneLng]]);
        ss.getSheetByName(TAB_CLASSES).getRange(cfg.rowIndex, 4, 1, 2).setValues([[phoneLat, phoneLng]]);
        try {
          cache.put('SES_ANCHOR_' + existingOpen.sessionId, JSON.stringify({
            mode: 'LOCKED',
            lat: phoneLat,
            lng: phoneLng,
            source: 'TEACHER_GPS'
          }), 7200);
        } catch (e) {}
        invalidateClassCache(classId);
      } else if (anchorMode === 'STUDENT_AVG_3') {
        try {
          cache.put('SES_ANCHOR_' + existingOpen.sessionId, JSON.stringify({
            mode: 'COLLECTING_AVG_3',
            campusLat: cfg.roomLat,
            campusLng: cfg.roomLng,
            maxCampusMeters: 850,
            samples: []
          }), 7200);
        } catch (e) {}
      }
      return handleTeacherGetState(ss, { classId: classId });
    }

    let roomLat = cfg.roomLat;
    let roomLng = cfg.roomLng;

    if (hasTeacherGps) {
      roomLat = phoneLat;
      roomLng = phoneLng;
      const clSheet = ss.getSheetByName(TAB_CLASSES);
      clSheet.getRange(cfg.rowIndex, 4, 1, 2).setValues([[roomLat, roomLng]]);
    }

    const nowMs = Date.now();
    const presentUntilMs = nowMs + cfg.presentMins * 60 * 1000;
    const closesAtMs     = presentUntilMs + cfg.lateMins * 60 * 1000;

    const dateLabel = Utilities.formatDate(new Date(nowMs), TIMEZONE, 'ddMMM_HHmm');
    const sessionId = 'SES_' + classId + '_' + Utilities.formatDate(new Date(nowMs), TIMEZONE, 'yyyyMMdd_HHmmss');
    const todayIsoKey = Utilities.formatDate(new Date(nowMs), TIMEZONE, 'yyyy-MM-dd');

    // If lecturer explicitly confirmed a MAKE-UP session on an unscheduled date, register todayIsoKey into class schedule metadata
    if (Boolean(body.isMakeUpSession)) {
      try {
        const props = PropertiesService.getScriptProperties();
        const schedMap = JSON.parse(props.getProperty('QR_CLASS_SCHEDULE_META_MAP') || '{}');
        const entry = schedMap[classId] || { sessionDates: [], makeUpDates: [] };
        if (!Array.isArray(entry.sessionDates)) entry.sessionDates = [];
        if (!Array.isArray(entry.makeUpDates)) entry.makeUpDates = [];
        if (entry.sessionDates.indexOf(todayIsoKey) === -1) {
          entry.sessionDates.push(todayIsoKey);
          entry.sessionDates.sort((a, b) => String(a).localeCompare(String(b)));
        }
        if (entry.makeUpDates.indexOf(todayIsoKey) === -1) {
          entry.makeUpDates.push(todayIsoKey);
        }
        schedMap[classId] = entry;
        props.setProperty('QR_CLASS_SCHEDULE_META_MAP', JSON.stringify(schedMap));
      } catch (e) {}
    }

    const sesSheet = ss.getSheetByName(TAB_SESSIONS);
    sesSheet.appendRow([
      sessionId,
      classId,
      dateLabel,
      formatIso(nowMs),
      formatIso(presentUntilMs),
      formatIso(closesAtMs),
      '',
      roomLat,
      roomLng,
      cfg.radiusMeters,
      qrMode
    ]);

    try {
      if (hasTeacherGps) {
        cache.put('SES_ANCHOR_' + sessionId, JSON.stringify({
          mode: 'LOCKED',
          lat: roomLat,
          lng: roomLng,
          source: 'TEACHER_GPS'
        }), 7200);
      } else {
        cache.put('SES_ANCHOR_' + sessionId, JSON.stringify({
          mode: 'COLLECTING_AVG_3',
          campusLat: cfg.roomLat,
          campusLng: cfg.roomLng,
          maxCampusMeters: 850,
          samples: []
        }), 7200);
      }
    } catch (e) {}

    invalidateClassCache(classId);
    return handleTeacherGetState(ss, { classId: classId });
  } finally {
    lock.releaseLock();
  }
}

function removeMakeUpDateFromScheduleMeta_(classId, dayKey) {
  if (!classId || !dayKey) return;
  try {
    const props = PropertiesService.getScriptProperties();
    const schedMap = JSON.parse(props.getProperty('QR_CLASS_SCHEDULE_META_MAP') || '{}');
    const entry = schedMap[classId];
    if (!entry) return;
    if (Array.isArray(entry.makeUpDates) && entry.makeUpDates.indexOf(dayKey) !== -1) {
      entry.makeUpDates = entry.makeUpDates.filter(d => d !== dayKey);
      if (Array.isArray(entry.sessionDates)) {
        entry.sessionDates = entry.sessionDates.filter(d => d !== dayKey);
      }
      schedMap[classId] = entry;
      props.setProperty('QR_CLASS_SCHEDULE_META_MAP', JSON.stringify(schedMap));
    }
  } catch (e) {}
}

function handleTeacherControlSession(ss, body) {
  const classId   = String(body.classId || '').trim();
  const subAction = String(body.subAction || '').toUpperCase();
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    return jsonResponse({ ok: false, code: 'ERR_SERVER_BUSY' });
  }
  try {
  const sesSheet  = ss.getSheetByName(TAB_SESSIONS);

  if (sesSheet.getLastRow() < 2) return jsonResponse({ ok: false, code: 'ERR_NO_SESSION' });
  const data = sesSheet.getRange(2, 1, sesSheet.getLastRow() - 1, 11).getValues();

  let targetRow = -1;
  for (let i = data.length - 1; i >= 0; i--) {
    if (String(data[i][1]).trim() === classId) {
      targetRow = i + 2;
      break;
    }
  }
  if (targetRow === -1) return jsonResponse({ ok: false, code: 'ERR_NO_SESSION' });

  const rowVals = data[targetRow - 2];
  const targetSesId = String(rowVals[0] || '').trim();
  const openedMs = parseSheetTimeMs(rowVals[3]);
  const closeMs = parseSheetTimeMs(rowVals[5]);
  const manualCloseMs = parseSheetTimeMs(rowVals[6]);
  const sesDayKey = openedMs > 0
    ? Utilities.formatDate(new Date(openedMs), TIMEZONE, 'yyyy-MM-dd')
    : '';
  const nowMs = Date.now();
  const todayDayKey = Utilities.formatDate(new Date(nowMs), TIMEZONE, 'yyyy-MM-dd');
  const isTargetOpen = !manualCloseMs && nowMs <= closeMs;
  if (!isTargetOpen && sesDayKey !== todayDayKey) {
    return jsonResponse({
      ok: false,
      code: 'ERR_NO_SESSION',
      message: 'Hôm nay chưa có phiên điểm danh nào cho lớp này.'
    });
  }

  if (subAction === 'CANCEL_MISTAKEN_SESSION') {
    const deletedSesIds = {};
    if (targetSesId) deletedSesIds[targetSesId] = true;

    let prevLat = 0, prevLng = 0;
    for (let i = data.length - 1; i >= 0; i--) {
      if (String(data[i][1]).trim() !== classId) continue;
      const rSesId = String(data[i][0] || '').trim();
      const rOpenedMs = parseSheetTimeMs(data[i][3]);
      const rDayKey = rOpenedMs > 0
        ? Utilities.formatDate(new Date(rOpenedMs), TIMEZONE, 'yyyy-MM-dd')
        : '';
      if (rSesId === targetSesId || (sesDayKey && rDayKey === sesDayKey)) {
        if (rSesId) deletedSesIds[rSesId] = true;
        sesSheet.deleteRow(i + 2);
      } else if (!prevLat && Number(data[i][7]) && Number(data[i][8])) {
        prevLat = Number(data[i][7]);
        prevLng = Number(data[i][8]);
      }
    }
    if (prevLat && prevLng) {
      const cfg = getClassesMap(ss, true)[classId];
      if (cfg && cfg.rowIndex) ss.getSheetByName(TAB_CLASSES).getRange(cfg.rowIndex, 4, 1, 2).setValues([[prevLat, prevLng]]);
    }

    const attSheet = ss.getSheetByName(TAB_ATTEND);
    if (attSheet && attSheet.getLastRow() >= 2) {
      const attVals = attSheet.getRange(2, 1, attSheet.getLastRow() - 1, 4).getValues();
      for (let r = attVals.length - 1; r >= 0; r--) {
        const aSesId = String(attVals[r][2] || '').trim();
        if (deletedSesIds[aSesId]) {
          attSheet.deleteRow(r + 2);
        }
      }
    }

    try {
      const cache = CacheService.getScriptCache();
      const cacheKeysToClear = [];
      Object.keys(deletedSesIds).forEach(sid => {
        cacheKeysToClear.push('SES_ATT_MAP_V36_' + sid);
        cacheKeysToClear.push('SES_ANCHOR_' + sid);
      });
      if (cacheKeysToClear.length > 0) cache.removeAll(cacheKeysToClear);
    } catch (e) {}

    if (sesDayKey) {
      removeMakeUpDateFromScheduleMeta_(classId, sesDayKey);
    }
    invalidateClassCache(classId);
    rebuildSingleClassGrid(ss, classId);
    SpreadsheetApp.flush();
    logAudit(ss, 'TEACHER', 'CANCEL_MISTAKEN_SESSION', classId, 'Deleted mistaken session(s): ' + Object.keys(deletedSesIds).join(', '));
  } else if (subAction === 'CLOSE_NOW') {
    const curPresent = parseSheetTimeMs(rowVals[4]);
    const curClose   = parseSheetTimeMs(rowVals[5]);
    const clampedPresent = curPresent > 0 ? Math.min(curPresent, nowMs) : nowMs;
    const clampedClose   = curClose > 0 ? Math.min(curClose, nowMs) : nowMs;
    sesSheet.getRange(targetRow, 5, 1, 3).setValues([[formatIso(clampedPresent), formatIso(clampedClose), formatIso(nowMs)]]);
    if (sesDayKey && targetSesId && !isRound2Session_(targetSesId, rowVals[2])) {
      const checks = getSessionAttendanceMap(ss, targetSesId, true);
      const hasRealMark = Object.values(checks).some(rec => rec && (rec.status === 'P' || rec.status === 'L' || rec.status === 'E'));
      if (!hasRealMark) {
        removeMakeUpDateFromScheduleMeta_(classId, sesDayKey);
      }
    }
    invalidateClassCache(classId);
    rebuildSingleClassGrid(ss, classId);
  } else if (subAction === 'OPEN_ROUND_2') {
    const todayBundle = getTodayClassSessionsBundle_(ss, classId, todayDayKey);
    if (!todayBundle.p1Session) {
      return jsonResponse({
        ok: false,
        code: 'ERR_NO_P1_SESSION',
        message: 'Hôm nay chưa mở Phiên 1 nên chưa thể mở Điểm danh lần 2.'
      });
    }
    if (todayBundle.p1Session.isOpen) {
      return jsonResponse({
        ok: false,
        code: 'ERR_P1_STILL_OPEN',
        message: 'Phiên 1 đang mở. Vui lòng đóng Phiên 1 trước khi mở Điểm danh lần 2.'
      });
    }
    if (todayBundle.round2OpenedToday) {
      return jsonResponse({
        ok: false,
        code: 'ERR_ROUND2_ALREADY_USED',
        message: 'Mỗi buổi học chỉ được mở Điểm danh lần 2 tối đa 1 lần.'
      });
    }
    const classes = getClassesMap(ss, false);
    const cfg = classes[classId] || { roomLat: 0, roomLng: 0, radiusMeters: 100 };
    const p1Ses = todayBundle.p1Session;
    const roomLat = Number(p1Ses.roomLat || cfg.roomLat || 0);
    const roomLng = Number(p1Ses.roomLng || cfg.roomLng || 0);
    const radiusMeters = Number(p1Ses.radiusMeters || cfg.radiusMeters || 100);
    const round2Mins = Math.min(30, Math.max(1, Number(body.durationMinutes || body.round2Mins || 3)));
    const reqQrMode = String(body.qrMode || 'STATIC').toUpperCase();
    const p2QrMode = (reqQrMode === 'PIN_10S' || reqQrMode === 'ROTATING') ? reqQrMode : 'STATIC';
    const presentUntilMs = nowMs + round2Mins * 60 * 1000;
    const closesAtMs = presentUntilMs;
    const p2SessionId = 'SES_' + classId + '_' + Utilities.formatDate(new Date(nowMs), TIMEZONE, 'yyyyMMdd_HHmmss') + '_R2';
    const p2Label = Utilities.formatDate(new Date(nowMs), TIMEZONE, 'ddMMM_HHmm') + '_P2';

    sesSheet.appendRow([
      p2SessionId,
      classId,
      p2Label,
      formatIso(nowMs),
      formatIso(presentUntilMs),
      formatIso(closesAtMs),
      '',
      roomLat,
      roomLng,
      radiusMeters,
      p2QrMode
    ]);

    try {
      const cache = CacheService.getScriptCache();
      const p1AnchorRaw = cache.get('SES_ANCHOR_' + p1Ses.sessionId);
      if (p1AnchorRaw) {
        cache.put('SES_ANCHOR_' + p2SessionId, p1AnchorRaw, 7200);
      } else if (roomLat && roomLng) {
        cache.put('SES_ANCHOR_' + p2SessionId, JSON.stringify({
          mode: 'LOCKED',
          lat: roomLat,
          lng: roomLng,
          source: 'P1_INHERITED'
        }), 7200);
      } else {
        cache.put('SES_ANCHOR_' + p2SessionId, JSON.stringify({
          mode: 'COLLECTING_AVG_3',
          campusLat: cfg.roomLat,
          campusLng: cfg.roomLng,
          maxCampusMeters: 850,
          samples: []
        }), 7200);
      }
    } catch (e) {}

    invalidateClassCache(classId);
    rebuildSingleClassGrid(ss, classId);
    logAudit(ss, 'TEACHER_APP', 'OPEN_ROUND_2', classId, p2SessionId + ' (' + round2Mins + 'm, ' + p2QrMode + ')');
  } else if (subAction === 'EXTEND_2M') {
    const curPresent = parseSheetTimeMs(rowVals[4]);
    const curClose   = parseSheetTimeMs(rowVals[5]);
    const isR2       = isRound2Session_(targetSesId, rowVals[2]);
    const addMs      = 2 * 60 * 1000;
    let newPresent, newClose;

    if (!isTargetOpen) {
      newPresent = nowMs + addMs;
      newClose   = newPresent;
    } else if (isR2) {
      newClose   = curClose + addMs;
      newPresent = newClose;
    } else if (nowMs <= curPresent) {
      const lateTailMs = Math.max(0, curClose - curPresent);
      newPresent = curPresent + addMs;
      newClose   = newPresent + lateTailMs;
    } else {
      newPresent = nowMs + addMs;
      newClose   = Math.max(curClose, newPresent);
    }
    sesSheet.getRange(targetRow, 5, 1, 3).setValues([[formatIso(newPresent), formatIso(newClose), '']]);
    invalidateClassCache(classId);
  } else if (subAction === 'REOPEN_5M') {
    const todayBundle = getTodayClassSessionsBundle_(ss, classId, todayDayKey);
    if (todayBundle.round2OpenedToday || isRound2Session_(targetSesId, rowVals[2])) {
      return jsonResponse({
        ok: false,
        code: 'ERR_REOPEN_BLOCKED_BY_P2',
        message: 'Không thể mở lại Phiên 1 vì Phiên 2 đã được mở trong buổi học này.'
      });
    }
    const newPresent = nowMs + 5 * 60 * 1000;
    const newClose   = newPresent;
    sesSheet.getRange(targetRow, 5, 1, 3).setValues([[formatIso(newPresent), formatIso(newClose), '']]);
    invalidateClassCache(classId);
  } else if (subAction === 'SET_QR_MODE') {
    const newMode = String(body.qrMode || 'STATIC').toUpperCase();
    sesSheet.getRange(targetRow, 11).setValue(newMode);
    invalidateClassCache(classId);
  }

  return handleTeacherGetState(ss, { classId: classId });
  } finally {
    lock.releaseLock();
  }
}

function handleTeacherManualMark(ss, body) {
  const classId         = String(body.classId || '').trim();
  const studentId       = normalizeId(body.studentId);
  const status          = String(body.status || 'P').toUpperCase(); // P | L | E | -
  const targetSessionId = String(body.targetSessionId || '').trim();
  const targetDayKey    = String(body.targetDayKey || '').trim();

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) return jsonResponse({ ok: false, code: 'ERR_SERVER_BUSY' });

  try {
    let targetSessionIds = [];
    let primarySessionId = '';
    const isPastSessionOverride = Boolean(targetSessionId || targetDayKey);

    if (isPastSessionOverride) {
      const sesSheet = ss.getSheetByName(TAB_SESSIONS);
      const sesData = (sesSheet && sesSheet.getLastRow() > 1)
        ? sesSheet.getRange(2, 1, sesSheet.getLastRow() - 1, 4).getValues()
        : [];
      sesData.forEach(r => {
        if (String(r[1]).trim() !== classId) return;
        const sid = String(r[0]).trim();
        const label = String(r[2] || '').trim();
        const openedMs = parseSheetTimeMs(r[3]);
        let dKey = '';
        if (openedMs > 0) {
          dKey = Utilities.formatDate(new Date(openedMs), TIMEZONE, 'yyyy-MM-dd');
        } else if (label) {
          dKey = label.split('_')[0] || sid;
        }
        if ((targetDayKey && dKey === targetDayKey) || (targetSessionId && sid === targetSessionId)) {
          targetSessionIds.push(sid);
          primarySessionId = sid;
        }
      });
      if (!primarySessionId && targetDayKey && /^\d{4}-\d{2}-\d{2}$/.test(targetDayKey)) {
        const ymdCompact = targetDayKey.replace(/-/g, '');
        const dtParts = targetDayKey.split('-');
        const dtObj = new Date(Number(dtParts[0]), Number(dtParts[1]) - 1, Number(dtParts[2]), 15, 0, 0);
        const dtIso = formatIso(dtObj.getTime());
        const dtLabel = Utilities.formatDate(dtObj, TIMEZONE, 'ddMMM') + '_HIST';
        primarySessionId = 'SES_' + classId + '_' + ymdCompact + '_150000';
        targetSessionIds.push(primarySessionId);
        const classes = getClassesMap(ss, false);
        const cfg = classes[classId] || { roomLat: 0, roomLng: 0, radiusMeters: 100 };
        if (sesSheet) {
          sesSheet.appendRow([
            primarySessionId,
            classId,
            dtLabel,
            dtIso,
            dtIso,
            dtIso,
            dtIso,
            cfg.roomLat || 0,
            cfg.roomLng || 0,
            cfg.radiusMeters || 100,
            'STATIC'
          ]);
        }
      }
      if (!primarySessionId && targetSessionId) {
        primarySessionId = targetSessionId;
        targetSessionIds.push(targetSessionId);
      }
      if (!primarySessionId) {
        return jsonResponse({ ok: false, code: 'ERR_NO_SESSION', message: 'Past session not found for the selected date.' });
      }
    } else {
      const todayDayKey = Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd');
      const todayBundle = getTodayClassSessionsBundle_(ss, classId, todayDayKey);
      const targetRound = String(body.targetRound || '').trim().toUpperCase();

      if (!todayBundle.p1Session && !todayBundle.p2Session) {
        return jsonResponse({
          ok: false,
          code: 'ERR_NO_SESSION',
          message: 'Hôm nay chưa mở phiên điểm danh cho lớp này. Vui lòng mở phiên trước hoặc vào mục Lịch sử để sửa buổi cũ.'
        });
      }

      if (targetRound === 'FINAL' || (todayBundle.round2OpenedToday && targetRound !== 'P1' && targetRound !== 'P2')) {
        return jsonResponse({
          ok: false,
          code: 'ERR_FINAL_READ_ONLY',
          message: 'Tab Kết quả cuối chỉ dùng để xem. Vui lòng chuyển sang tab Phiên 1 hoặc Phiên 2 để chấm tay.'
        });
      }

      if (targetRound === 'P2') {
        if (!todayBundle.p2Session) {
          return jsonResponse({
            ok: false,
            code: 'ERR_NO_P2_SESSION',
            message: 'Phiên 2 chưa được mở nên chưa thể chấm tay Phiên 2.'
          });
        }
        primarySessionId = todayBundle.p2Session.sessionId;
        targetSessionIds = [primarySessionId];
      } else {
        const p1Ses = todayBundle.p1Session || todayBundle.activeSession;
        primarySessionId = p1Ses.sessionId;
        targetSessionIds = [primarySessionId];
      }
    }

    const attSheet = ss.getSheetByName(TAB_ATTEND);
    const nowIso = formatIsoNow();
    const timeStr = nowIso.slice(11, 19);
    let foundRow = -1;

    if (!isPastSessionOverride) {
      const sesMap = getSessionAttendanceMap(ss, primarySessionId, false);
      const existing = sesMap ? sesMap[studentId] : null;
      if (existing && existing.rowIndex && existing.rowIndex >= 2) {
        foundRow = Number(existing.rowIndex);
      }
    }

    if (foundRow === -1 && attSheet.getLastRow() > 1) {
      const data = attSheet.getRange(2, 1, attSheet.getLastRow() - 1, 10).getValues();
      const targetSet = {};
      targetSessionIds.forEach(id => { targetSet[id] = true; });
      for (let i = data.length - 1; i >= 0; i--) {
        const rowSesId = String(data[i][2]).trim();
        if (targetSet[rowSesId] && normalizeId(data[i][4]) === studentId) {
          attSheet.getRange(i + 2, 6, 1, 5).setValues([[status, nowIso, 0, 0, 'TEACHER_MANUAL']]);
          if (foundRow === -1) foundRow = i + 2;
        }
      }
    } else if (foundRow !== -1) {
      attSheet.getRange(foundRow, 6, 1, 5).setValues([[status, nowIso, 0, 0, 'TEACHER_MANUAL']]);
    }

    if (foundRow === -1) {
      attSheet.appendRow([
        'ATT_M_' + generateRandomToken(8),
        'MANUAL_' + Date.now(),
        primarySessionId,
        classId,
        studentId,
        status,
        nowIso,
        0,
        0,
        'TEACHER_MANUAL'
      ]);
      foundRow = attSheet.getLastRow();
    }

    const cache = CacheService.getScriptCache();
    const recPayload = {
      rowIndex: foundRow,
      attendanceId: 'ATT_M',
      requestId: 'MANUAL',
      sessionId: primarySessionId,
      classId: classId,
      studentId: studentId,
      status: status,
      time: timeStr,
      distance: 0,
      distanceMeters: 0,
      accuracy: 0,
      source: 'TEACHER_MANUAL'
    };

    targetSessionIds.forEach(sid => {
      const payloadForSid = Object.assign({}, recPayload, { sessionId: sid });
      try {
        cache.put('SES_ATT_' + sid + '_' + studentId, JSON.stringify(payloadForSid), 1800);
      } catch (e) {}
      updateSessionAttendanceMapCache_(sid, studentId, payloadForSid);
    });

    invalidateClassAttendanceStatsCache_(classId);
    try { rebuildSingleClassGrid(ss, classId); } catch (e) {}

    logAudit(ss, 'TEACHER_APP', 'MANUAL_MARK', studentId, classId + ' [' + (targetDayKey || primarySessionId) + '] -> ' + status);
    return handleTeacherGetState(ss, { classId: classId });
  } finally {
    lock.releaseLock();
  }
}

function handleTeacherPublishGroups(ss, body) {
  const classId = String(body.classId || '').trim();
  if (!classId) return jsonResponse({ ok: false, code: 'ERR_INVALID_CLASS', message: 'Missing classId.' });

  const cache = CacheService.getScriptCache();
  const cacheKey = 'GROUP_SPLIT_' + classId;

  if (body.clear) {
    try { cache.remove(cacheKey); } catch (e) {}
    return jsonResponse({ ok: true, code: 'GROUPS_CLEARED', classId: classId, activeGroupSplit: null });
  }

  const groups = Array.isArray(body.groups) ? body.groups : [];
  const groupSize = Number(body.groupSize || 0);
  const payload = {
    classId: classId,
    groupSize: groupSize,
    publishedAt: formatIsoNow(),
    groups: groups.map((g, idx) => ({
      groupNumber: Number(g.groupNumber || (idx + 1)),
      members: Array.isArray(g.members) ? g.members.map(m => ({
        studentId: normalizeId(m.studentId),
        fullName: String(m.fullName || '').trim()
      })) : []
    }))
  };

  try {
    cache.put(cacheKey, JSON.stringify(payload), 14400); // 4 hours in RAM cache
  } catch (e) {}

  logAudit(ss, 'TEACHER_APP', 'PUBLISH_GROUPS', classId, 'Groups: ' + payload.groups.length);
  return jsonResponse({
    ok: true,
    code: 'GROUPS_PUBLISHED',
    classId: classId,
    activeGroupSplit: payload
  });
}

function getActiveGroupSplitForClass_(classId) {
  if (!classId) return null;
  try {
    const raw = CacheService.getScriptCache().get('GROUP_SPLIT_' + classId);
    if (!raw) return null;
    const parsed = JSON.parse(raw);
    if (!parsed || typeof parsed !== 'object') return null;
    const todayDayKey = Utilities.formatDate(new Date(), TIMEZONE, 'yyyy-MM-dd');
    if (parsed.publishedAt && !String(parsed.publishedAt).startsWith(todayDayKey)) {
      return null;
    }
    return parsed;
  } catch (e) {
    return null;
  }
}

function syncAndRebuildClassReport_(ss, classId, className) {
  let added = 0;
  let names = [];
  if (typeof syncSingleClassRosterFromDrive_ === 'function') {
    try {
      const syncRes = syncSingleClassRosterFromDrive_(ss, classId, className, {
        onlyIfModified: true,
        actor: 'TEACHER_APP'
      });
      if (syncRes && syncRes.newStudentsAdded > 0) {
        added = syncRes.newStudentsAdded;
        names = syncRes.newStudentNames || [];
      }
    } catch (e) {}
  }
  rebuildSingleClassGrid(ss, classId);
  invalidateClassAttendanceStatsCache_(classId);
  return { newStudentsAdded: added, newStudentNames: names };
}

function handleTeacherRebuildReport(ss, body) {
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(12000)) return jsonResponse({ ok: false, code: 'ERR_SERVER_BUSY' });
  try {
    const classId = String(body.classId || '').trim();
    invalidateSystemCache();
    const classes = getClassesMap(ss, true);
    let newStudentsAdded = 0;
    let newStudentNames = [];

    const targetIds = classId ? [classId] : Object.keys(classes);
    targetIds.forEach(cid => {
      const cName = classes[cid] ? classes[cid].className : cid;
      const res = syncAndRebuildClassReport_(ss, cid, cName);
      newStudentsAdded += res.newStudentsAdded;
      newStudentNames = newStudentNames.concat(res.newStudentNames);
    });

    return jsonResponse({
      ok: true,
      code: 'REPORT_REBUILT',
      newStudentsAdded: newStudentsAdded,
      newStudentNames: newStudentNames
    });
  } finally {
    lock.releaseLock();
  }
}

function buildBaseCacheKeysToClear_(extraKeys) {
  const keys = ['ALL_LATEST_SES_V373', 'ALL_LATEST_SES_V35', 'CLASSES_MAP_V373'].concat(Array.isArray(extraKeys) ? extraKeys : []);
  for (let r = 1; r <= 30; r++) keys.push('CLASSES_MAP_V32_' + r);
  return keys;
}

function invalidateSystemCache() {
  try {
    CacheService.getScriptCache().removeAll(buildBaseCacheKeysToClear_([]));
  } catch (e) {}
}

  } catch (e) {}
}

}

      classConfig ? String(classConfig.classId || '') : '',
      String(studentId || ''),
      String(submittedPin || ''),
      serverBucket,
      studentBucket === null || studentBucket === undefined ? '' : studentBucket,
      offsetIntoBucketMs,
      deltaMs === null || deltaMs === undefined ? '' : deltaMs,
      graceMs,
      String(resultLabel || '')
    ]);
  } catch (e) {}
}

function evaluateRound2Pin10s_(ss, scannedText, classConfig, activeSession, checkMs, studentId) {
  const secret = classConfig.activeQrSecret;
  const rawStr = String(scannedText || '').trim();
  const cleanedPin = rawStr.replace(/^PIN10:/i, '').trim();
  const graceMs = (typeof PIN_GRACE_MS === 'number' && !isNaN(PIN_GRACE_MS)) ? PIN_GRACE_MS : 2500;
  const serverBucket = Math.floor(checkMs / 10000);
  const offsetIntoBucketMs = ((checkMs % 10000) + 10000) % 10000;

  const cache = CacheService.getScriptCache();
  const guardKey = 'PIN_GUARD_' + String(activeSession.sessionId || '') + '_' + String(studentId || '');
  let guard = { wrongCount: 0, lockedUntilMs: 0 };
  try {
    const rawGuard = cache.get(guardKey);
    if (rawGuard) guard = JSON.parse(rawGuard) || guard;
  } catch (e) {}

  if (guard.lockedUntilMs && checkMs < guard.lockedUntilMs) {
    const waitSec = Math.max(1, Math.ceil((guard.lockedUntilMs - checkMs) / 1000));
    return {
      ok: false,
      code: 'PIN_LOCKED',
      params: { seconds: waitSec },
      message: '[TẠM KHÓA MÃ PIN] Bạn đã nhập sai mã PIN quá nhiều lần. Vui lòng chờ ' + waitSec + ' giây rồi nhìn bảng chiếu nhập lại.',
      serverBucket: serverBucket,
      studentBucket: null,
      offsetIntoBucketMs: offsetIntoBucketMs,
      deltaMs: null,
      graceMs: graceMs,
      pinAgeMs: 0,
      cleanedPin: cleanedPin,
      result: 'PIN_LOCKED'
    };
  }

  if (!/^\d{4}$/.test(cleanedPin)) {
    guard.wrongCount = Number(guard.wrongCount || 0) + 1;
    let outCode = 'PIN_WRONG';
    let outParams = undefined;
    let outMsg = undefined;
    if (guard.wrongCount >= 4) {
      guard.lockedUntilMs = checkMs + 25000;
      guard.wrongCount = 0;
      outCode = 'PIN_LOCKED';
      outParams = { seconds: 25 };
      outMsg = '[TẠM KHÓA MÃ PIN] Bạn đã nhập sai mã PIN quá nhiều lần. Vui lòng chờ 25 giây rồi nhìn bảng chiếu nhập lại.';
    }
    try { cache.put(guardKey, JSON.stringify(guard), 300); } catch (e) {}
    return {
      ok: false,
      code: outCode,
      params: outParams,
      message: outMsg,
      serverBucket: serverBucket,
      studentBucket: null,
      offsetIntoBucketMs: offsetIntoBucketMs,
      deltaMs: null,
      graceMs: graceMs,
      pinAgeMs: 0,
      cleanedPin: cleanedPin,
      result: outCode
    };
  }

  const seed = getRound2PinSeed_(secret, activeSession.sessionId);
  let matchedBucket = null;
  const searchOffsets = [0, -1, 1, -2, -3, -4, -5, -6, 2];
  for (let i = 0; i < searchOffsets.length; i++) {
    const candBucket = serverBucket + searchOffsets[i];
    if (computePin4FromSeed(seed, candBucket) === cleanedPin) {
      matchedBucket = candBucket;
      break;
    }
  }

  let ok = false;
  let code = 'PIN_WRONG';
  let deltaMs = null;

  if (matchedBucket === serverBucket) {
    ok = true;
    code = 'PIN_OK';
    deltaMs = 0;
  } else if (matchedBucket === (serverBucket - 1)) {
    deltaMs = offsetIntoBucketMs;
    if (offsetIntoBucketMs <= graceMs) {
      ok = true;
      code = 'PIN_OK';
    } else {
      ok = false;
      code = 'PIN_EXPIRED';
    }
  } else if (matchedBucket !== null && matchedBucket < (serverBucket - 1)) {
    deltaMs = (serverBucket - (matchedBucket + 1)) * 10000 + offsetIntoBucketMs;
    ok = false;
    code = 'PIN_EXPIRED';
  } else if (matchedBucket === (serverBucket + 1) && (10000 - offsetIntoBucketMs) <= Math.min(graceMs, 2000)) {
    deltaMs = -(10000 - offsetIntoBucketMs);
    ok = true;
    code = 'PIN_OK';
  } else {
    ok = false;
    code = 'PIN_WRONG';
  }

  let outParams = undefined;
  let outMsg = undefined;
  if (ok) {
    if (guard.wrongCount > 0 || guard.lockedUntilMs > 0) {
      try { cache.remove(guardKey); } catch (e) {}
    }
  } else if (code === 'PIN_WRONG') {
    guard.wrongCount = Number(guard.wrongCount || 0) + 1;
    if (guard.wrongCount >= 4) {
      guard.lockedUntilMs = checkMs + 25000;
      guard.wrongCount = 0;
      code = 'PIN_LOCKED';
      outParams = { seconds: 25 };
      outMsg = '[TẠM KHÓA MÃ PIN] Bạn đã nhập sai mã PIN quá nhiều lần. Vui lòng chờ 25 giây rồi nhìn bảng chiếu nhập lại.';
    }
    try { cache.put(guardKey, JSON.stringify(guard), 300); } catch (e) {}
  }

  const pinAgeMs = matchedBucket === serverBucket
    ? offsetIntoBucketMs
    : (matchedBucket === (serverBucket - 1) ? (10000 + offsetIntoBucketMs) : 0);

  return {
    ok: ok,
    code: code,
    params: outParams,
    message: outMsg,
    serverBucket: serverBucket,
    studentBucket: matchedBucket,
    offsetIntoBucketMs: offsetIntoBucketMs,
    deltaMs: deltaMs,
    graceMs: graceMs,
    pinAgeMs: pinAgeMs,
    cleanedPin: cleanedPin,
    result: code
  };
}

function verifyClassroomQr(scannedText, classConfig, activeSession, overrideNowMs, ss, studentId) {
  const secret = classConfig.activeQrSecret;
  const checkMs = typeof overrideNowMs === 'number' && overrideNowMs > 0 ? overrideNowMs : Date.now();

  if (activeSession && activeSession.qrMode === 'PIN_10S') {
    const evalRes = evaluateRound2Pin10s_(ss, scannedText, classConfig, activeSession, checkMs, studentId);
    activeSession._lastPinEval = evalRes;
    return evalRes.ok;
  }

  let parsed = null;
  try {
    parsed = JSON.parse(scannedText);
  } catch (e) {
    parsed = null;
  }

  if (!parsed) {
    if (activeSession.qrMode === 'ROTATING') return false;
    const prefix = 'QR_' + classConfig.classId.toUpperCase() + '_';
    return (
      scannedText === secret ||
      scannedText === 'BUTTON_GPS_CHECKIN' ||
      /^PIN10:\d{4}$/i.test(String(scannedText || '').trim()) ||
      scannedText.toUpperCase().indexOf(prefix) === 0
    );
  }

  if (parsed.cid && parsed.cid !== classConfig.classId) return false;

  if (parsed.mode === 'ROTATING') {
    const bucketNow = Math.floor(checkMs / 15000);
    const expectedNow  = sha256Hex(secret + '|' + bucketNow).slice(0, 12);
    const expectedPrev = sha256Hex(secret + '|' + (bucketNow - 1)).slice(0, 12);
    return parsed.tok === expectedNow || parsed.tok === expectedPrev;
  }

  if (activeSession.qrMode === 'ROTATING') return false;
  return parsed.sec === secret;
}

function verifyStudentCredential(ss, studentId, clientCred, precomputedHash, canonicalName) {
  const credHash = precomputedHash || sha256Hex(clientCred);
  const cache = CacheService.getScriptCache();
  const cachedHash = cache.get('DEV_HASH_' + studentId);
  if (cachedHash && cachedHash === credHash) {
    return { ok: true };
  }

  const devSheet = ss.getSheetByName(TAB_DEVICES);
  const data = devSheet.getLastRow() >= 2
    ? devSheet.getRange(2, 1, devSheet.getLastRow() - 1, 5).getValues()
    : [];

  let existingStudentRow = -1;
  let existingRowData = null;

  for (let i = 0; i < data.length; i++) {
    const rowSid = normalizeId(data[i][0]);
    const rowHash = String(data[i][2]);
    const rowStatus = String(data[i][4]);

    if (rowStatus === 'ACTIVE' && rowHash === credHash && rowSid !== studentId) {
      return { ok: false, code: 'ERR_DEVICE_BOUND_OTHER' };
    }
    if (rowSid === studentId) {
      existingStudentRow = i + 2;
      existingRowData = data[i];
    }
  }

  if (existingStudentRow !== -1 && existingRowData) {
    const rowHash = String(existingRowData[2]);
    const rowStatus = String(existingRowData[4]);
    if (rowStatus === 'ACTIVE') {
      if (rowHash === credHash) {
        cache.put('DEV_HASH_' + studentId, credHash, 21600);
        return { ok: true };
      }
      return { ok: false, code: 'ERR_CREDENTIAL_MISMATCH' };
    }
    // Device was REVOKED (unlocked by Lecturer) -> auto-rebind immediately on check-in!
    const nameToSave = canonicalName || String(existingRowData[1] || studentId);
    devSheet.getRange(existingStudentRow, 2, 1, 4).setValues([[nameToSave, credHash, formatIsoNow(), 'ACTIVE']]);
    cache.put('DEV_HASH_' + studentId, credHash, 21600);
    return { ok: true };
  }

  // Student was not yet in Device_Registry -> auto-bind immediately on check-in
  devSheet.appendRow([studentId, canonicalName || studentId, credHash, formatIsoNow(), 'ACTIVE', 0, '']);
  cache.put('DEV_HASH_' + studentId, credHash, 21600);
  return { ok: true };
}

function resetStudentCredentialInternal(studentId, actor) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const devSheet = ss.getSheetByName(TAB_DEVICES);
  if (devSheet.getLastRow() < 2) {
    return { ok: false, code: 'ERR_STUDENT_NOT_REGISTERED', message: 'Student has not registered a device yet.' };
  }
  const data = devSheet.getRange(2, 1, devSheet.getLastRow() - 1, 7).getValues();
  for (let i = 0; i < data.length; i++) {
    if (normalizeId(data[i][0]) === studentId) {
      const resets = Number(data[i][5] || 0) + 1;
      devSheet.getRange(i + 2, 5, 1, 3).setValues([['REVOKED', resets, formatIsoNow()]]);
      CacheService.getScriptCache().remove('DEV_HASH_' + studentId);
      logAudit(ss, actor, 'RESET_DEVICE_LOCK', studentId, 'Reset count: ' + resets);
      return { ok: true, code: 'DEVICE_RESET_OK', studentId: studentId };
    }
  }
  return { ok: false, code: 'ERR_STUDENT_NOT_REGISTERED', message: 'Student ID ' + studentId + ' not found in Device_Registry.' };
}

function classifySheetCellColorLocal_(hexColor) {
  const hex = String(hexColor || '').trim().toLowerCase();
  if (!hex || hex === '#ffffff' || hex === '#000000' || hex === 'white' || hex === 'none' || hex === 'transparent') {
    return 'NONE';
  }
  const clean = hex.replace('#', '');
  if (!/^[0-9a-f]{6}$/.test(clean)) return 'NONE';

  const r = parseInt(clean.substring(0, 2), 16);
  const g = parseInt(clean.substring(2, 4), 16);
  const b = parseInt(clean.substring(4, 6), 16);

  const max = Math.max(r, g, b);
  const min = Math.min(r, g, b);
  const delta = max - min;
  if (max > 245 && min > 235) return 'NONE';
  if (delta < 28) return 'NONE';

  const exactOranges = {
    '#ff9900': 1, '#ffa500': 1, '#ffc000': 1, '#f6b26b': 1, '#f9cb9c': 1,
    '#fce5cd': 1, '#e69138': 1, '#b45f06': 1, '#ff6d01': 1, '#ffab40': 1,
    '#ffd180': 1, '#ff9100': 1, '#fb8c00': 1, '#f57c00': 1, '#ef6c00': 1
  };
  if (exactOranges[hex]) return 'ORANGE';

  const exactYellows = {
    '#ffff00': 1, '#fff2cc': 1, '#ffe599': 1, '#ffd966': 1, '#f1c232': 1,
    '#bf9000': 1, '#ffeb3b': 1, '#fff59d': 1, '#fff176': 1, '#ffee58': 1,
    '#fdd835': 1, '#fbc02d': 1, '#ffff8d': 1
  };
  if (exactYellows[hex]) return 'YELLOW';

  const exactReds = {
    '#ff0000': 1, '#e06666': 1, '#ea9999': 1, '#f4cccc': 1, '#cc0000': 1,
    '#990000': 1, '#660000': 1, '#d93025': 1, '#f44336': 1, '#ef5350': 1,
    '#e53935': 1, '#c62828': 1, '#ff5252': 1, '#ff1744': 1, '#dc2626': 1
  };
  if (exactReds[hex]) return 'RED';

  let hue = 0;
  if (max === r) {
    hue = ((g - b) / delta) % 6;
  } else if (max === g) {
    hue = (b - r) / delta + 2;
  } else {
    hue = (r - g) / delta + 4;
  }
  hue = Math.round(hue * 60);
  if (hue < 0) hue += 360;

  if ((hue >= 0 && hue <= 16) || hue >= 342) return 'RED';
  if (hue >= 17 && hue <= 46) return 'ORANGE';
  if (hue >= 47 && hue <= 69) return 'YELLOW';
  return 'OTHER';
}

function buildGroupedClassDayRecords_(ss, classId) {
  const sesSheet = ss.getSheetByName(TAB_SESSIONS);
  const sesData = (sesSheet && sesSheet.getLastRow() > 1)
    ? sesSheet.getRange(2, 1, sesSheet.getLastRow() - 1, 7).getValues()
    : [];

  const nowMs = Date.now();
  const todayKey = Utilities.formatDate(new Date(nowMs), TIMEZONE, 'yyyy-MM-dd');
  const dayGroups = {};
  const dayOrder = [];
  const sessionToDayKey = {};

  // 1. Seed any past completed sessions synced from the official Drive attendance sheet (dayKey <= todayKey)
  try {
    const rawPast = PropertiesService.getScriptProperties().getProperty('QR_SHEET_PAST_HISTORY_' + classId);
    if (rawPast) {
      const pastDays = JSON.parse(rawPast);
      if (Array.isArray(pastDays)) {
        pastDays.forEach(pd => {
          const dk = String(pd && pd.dayKey || '').trim();
          if (!dk || dk > todayKey) return;
          const histSesId = String(pd.sessionId || ('HIST_' + classId + '_' + dk.replace(/-/g, ''))).trim();
          const disp = String(pd.displayDate || (dk.slice(8, 10) + '/' + dk.slice(5, 7) + '/' + dk.slice(0, 4)));
          const lbl = String(pd.label || disp.slice(0, 5));
          const marksCopy = {};
          if (pd.marks && typeof pd.marks === 'object') {
            Object.keys(pd.marks).forEach(sid => {
              marksCopy[normalizeId(sid)] = String(pd.marks[sid]);
            });
          }
          dayGroups[dk] = {
            dayKey: dk,
            displayDate: disp,
            label: lbl,
            latestSessionId: histSesId,
            sessionIds: [histSesId],
            p1SessionIds: [histSesId],
            p2SessionIds: [],
            p2Opened: false,
            hasOpenP1: false,
            hasOpenP2: false,
            hasOpenSession: false,
            hasValidCheckIn: true,
            p1Marks: Object.assign({}, marksCopy),
            p2Marks: {},
            marks: marksCopy,
            manualLocked: {}
          };
          dayOrder.push(dk);
          sessionToDayKey[histSesId] = dk;
        });
      }
    }
  } catch (e) {}

  sesData.forEach(r => {
    if (String(r[1]).trim() !== classId) return;
    const sesId = String(r[0]).trim();
    const label = String(r[2] || '').trim();
    const openedMs = parseSheetTimeMs(r[3]);
    const closesMs = parseSheetTimeMs(r[5]);
    const closedManual = String(r[6] || '').trim();
    const isOpen = !closedManual && nowMs <= closesMs;
    const isR2 = isRound2Session_(sesId, label);
    const cleanLabel = label.replace(/_P2$/i, '');

    let dayKey = '';
    let displayDate = '';
    if (openedMs > 0) {
      const dt = new Date(openedMs);
      dayKey = Utilities.formatDate(dt, TIMEZONE, 'yyyy-MM-dd');
      displayDate = Utilities.formatDate(dt, TIMEZONE, 'dd/MM/yyyy');
    } else if (cleanLabel) {
      dayKey = cleanLabel.split('_')[0] || sesId;
      displayDate = dayKey;
    } else {
      dayKey = sesId;
      displayDate = sesId;
    }

    if (!dayGroups[dayKey]) {
      dayGroups[dayKey] = {
        dayKey: dayKey,
        displayDate: displayDate,
        label: cleanLabel || dayKey,
        latestSessionId: sesId,
        sessionIds: [sesId],
        p1SessionIds: isR2 ? [] : [sesId],
        p2SessionIds: isR2 ? [sesId] : [],
        p2Opened: isR2,
        hasOpenP1: !isR2 && isOpen,
        hasOpenP2: isR2 && isOpen,
        hasOpenSession: isOpen,
        hasValidCheckIn: false,
        p1Marks: {},
        p2Marks: {},
        marks: {},
        manualLocked: {}
      };
      dayOrder.push(dayKey);
    } else {
      if (dayGroups[dayKey].sessionIds.indexOf(sesId) === -1) {
        dayGroups[dayKey].sessionIds.push(sesId);
      }
      if (isR2) {
        if (dayGroups[dayKey].p2SessionIds.indexOf(sesId) === -1) {
          dayGroups[dayKey].p2SessionIds.push(sesId);
        }
        dayGroups[dayKey].p2Opened = true;
        if (isOpen) dayGroups[dayKey].hasOpenP2 = true;
      } else {
        if (dayGroups[dayKey].p1SessionIds.indexOf(sesId) === -1) {
          dayGroups[dayKey].p1SessionIds.push(sesId);
        }
        if (cleanLabel) dayGroups[dayKey].label = cleanLabel;
        if (isOpen) dayGroups[dayKey].hasOpenP1 = true;
      }
      dayGroups[dayKey].latestSessionId = sesId;
      if (isOpen) dayGroups[dayKey].hasOpenSession = true;
    }
    sessionToDayKey[sesId] = dayKey;
  });

  if (dayOrder.length === 0) return [];

  const attSheet = ss.getSheetByName(TAB_ATTEND);
  if (attSheet && attSheet.getLastRow() > 1) {
    const attData = attSheet.getRange(2, 1, attSheet.getLastRow() - 1, 10).getValues();
    attData.forEach(r => {
      if (String(r[3]).trim() !== classId) return;
      const sesId = String(r[2]).trim();
      let dayKey = sessionToDayKey[sesId];
      if (!dayKey) {
        const mHist = sesId.match(/^HIST_[A-Za-z0-9_-]+_(\d{4})(\d{2})(\d{2})$/);
        if (mHist) {
          dayKey = mHist[1] + '-' + mHist[2] + '-' + mHist[3];
        }
      }
      if (!dayKey || !dayGroups[dayKey]) return;
      const grp = dayGroups[dayKey];
      const sid = normalizeId(r[4]);
      const st = String(r[5] || '').trim().toUpperCase();
      const src = String(r[9] || '').trim();
      if (!sid) return;

      const isManual = (src === 'TEACHER_MANUAL');
      const isR2 = isRound2Session_(sesId, '') || (grp.p2SessionIds && grp.p2SessionIds.indexOf(sesId) !== -1);
      if (!grp.p1Manual) grp.p1Manual = {};
      if (!grp.p2Manual) grp.p2Manual = {};

      if (isR2) {
        if (isManual) {
          grp.p2Marks[sid] = (st === 'P' || st === 'L') ? 'P' : (st === 'E' ? 'E' : '-');
          grp.p2Manual[sid] = true;
          grp.manualLocked[sid] = true;
        } else if (!grp.p2Manual[sid] && !grp.p2Marks[sid]) {
          // Edge Case E.3: Only record the first P2 scan for a student; P2 'L' counts as 'P'
          grp.p2Marks[sid] = (st === 'P' || st === 'L') ? 'P' : (st === 'E' ? 'E' : '-');
        }
      } else {
        if (isManual) {
          grp.p1Marks[sid] = (st === 'P' || st === 'L' || st === 'E') ? st : '-';
          grp.p1Manual[sid] = true;
          grp.manualLocked[sid] = true;
        } else if (!grp.p1Manual[sid]) {
          const cur = grp.p1Marks[sid];
          if (st === 'P') {
            grp.p1Marks[sid] = 'P';
          } else if (st === 'E' && cur !== 'P') {
            grp.p1Marks[sid] = 'E';
          } else if (st === 'L' && cur !== 'P' && cur !== 'E') {
            grp.p1Marks[sid] = 'L';
          }
        }
      }
    });
  }

  const rosterMapForClass = getClassRosterMap(ss, classId, false);
  const rosterStudentIds = Object.keys(rosterMapForClass || {});

  Object.keys(dayGroups).forEach(dk => {
    const grp = dayGroups[dk];
    const allSidMap = {};
    rosterStudentIds.forEach(id => { allSidMap[id] = true; });
    Object.keys(grp.p1Marks || {}).forEach(id => { allSidMap[id] = true; });
    Object.keys(grp.p2Marks || {}).forEach(id => { allSidMap[id] = true; });

    Object.keys(allSidMap).forEach(sid => {
      const p1St = (grp.p1Marks && grp.p1Marks[sid]) ? grp.p1Marks[sid] : '-';
      const p2St = grp.p2Opened ? ((grp.p2Marks && grp.p2Marks[sid]) ? grp.p2Marks[sid] : '-') : '';
      grp.p1Marks[sid] = p1St;
      grp.p2Marks[sid] = p2St;
      const mergedFinal = mergeAttendance(p1St, p2St, grp.p2Opened);
      const isExplicitManualRevoke =
        (!grp.p2Opened && grp.p1Manual && grp.p1Manual[sid] && p1St === '-') ||
        (grp.p2Opened && grp.p2Manual && grp.p2Manual[sid] && p2St === '-');
      grp.marks[sid] = (mergedFinal === '-' && isExplicitManualRevoke) ? 'A_MANUAL' : mergedFinal;
    });

    if (!grp.hasValidCheckIn) {
      grp.hasValidCheckIn = Boolean(grp.p2Opened) || Object.keys(grp.p1Marks).some(sid => {
        const m = grp.p1Marks[sid];
        return m === 'P' || m === 'L' || m === 'E';
      }) || Object.keys(grp.marks).some(sid => {
        const m = grp.marks[sid];
        return m === 'P' || m === 'L' || m === 'E';
      });
    }
  });

  // Check official Sheet sessionDates (if available) so accidental wrong-class sessions on unscheduled dates are ignored once closed
  const allowedSheetDates = {};
  let hasSheetSchedule = false;
  try {
    const schedMap = getStoredClassSchedulesMap();
    const sDates = (schedMap && schedMap[classId] && Array.isArray(schedMap[classId].sessionDates))
      ? schedMap[classId].sessionDates
      : [];
    if (sDates.length > 0) {
      hasSheetSchedule = true;
      sDates.forEach(raw => {
        const s = String(raw || '').trim();
        if (!s) return;
        allowedSheetDates[s] = true;
        // Also index DD/MM shorthand in case sessionDates was stored as D/M or DD/MM
        const mIso = s.match(/^(\d{4})-(\d{1,2})-(\d{1,2})$/);
        if (mIso) {
          const d = parseInt(mIso[3], 10);
          const m = parseInt(mIso[2], 10);
          allowedSheetDates[d + '/' + m] = true;
          allowedSheetDates[('0' + d).slice(-2) + '/' + ('0' + m).slice(-2)] = true;
        }
      });
    }
  } catch (e) {}

  // Sort chronologically by dayKey and filter out:
  //  (a) closed ghost/test sessions where 0 students had any valid check-in (unless P2 was opened)
  //  (b) closed accidental sessions opened on a date that does NOT exist on the class Sheet
  return dayOrder
    .slice()
    .sort((a, b) => String(a).localeCompare(String(b)))
    .map(k => dayGroups[k])
    .filter(grp => {
      if (grp.hasOpenSession || grp.p2Opened) return true;
      if (!grp.hasValidCheckIn) return false;
      if (hasSheetSchedule) {
        const dk = String(grp.dayKey || '');
        const dmShort = dk.length >= 10 ? (dk.slice(8, 10) + '/' + dk.slice(5, 7)) : '';
        if (!allowedSheetDates[dk] && (!dmShort || !allowedSheetDates[dmShort])) {
          return false;
        }
      }
      return true;
    });
}

function rebuildSingleClassGrid(ss, classId) {
  const rosterSheet = ss.getSheetByName(classId);
  if (!rosterSheet || rosterSheet.getLastRow() < 2) return;

  const validDays = buildGroupedClassDayRecords_(ss, classId);
  const numStudents = rosterSheet.getLastRow() - 1;
  const studentRows = rosterSheet.getRange(2, 1, numStudents, 3).getValues();

  const sessionHeaders = [];
  validDays.forEach((d, idx) => {
    const prefix = 'S' + (idx + 1) + '_' + d.label;
    sessionHeaders.push(prefix + '_P1');
    sessionHeaders.push(prefix + '_Trạng_thái_P2');
    sessionHeaders.push(prefix + '_Kết_quả_cuối');
  });
  const headers = ['No.', 'Student_ID', 'Full_Name', 'Total_Present', 'Total_Late'].concat(sessionHeaders);

  const outRows = studentRows.map(sRow => {
    const stId = normalizeId(sRow[1]);
    let pCount = 0;
    let lCount = 0;
    const sessionCells = [];
    validDays.forEach(grp => {
      const p1Cell = (grp.p1Marks && grp.p1Marks[stId]) ? grp.p1Marks[stId] : '-';
      const p2Cell = grp.p2Opened ? ((grp.p2Marks && grp.p2Marks[stId]) ? grp.p2Marks[stId] : '-') : '';
      const rawFinal = grp.marks[stId] || mergeAttendance(p1Cell, p2Cell, grp.p2Opened);
      const finalSt = (rawFinal === 'A_MANUAL') ? '-' : rawFinal;
      if (finalSt === 'P') pCount++;
      if (finalSt === 'L') lCount++;
      sessionCells.push(p1Cell, p2Cell, finalSt);
    });
    return [sRow[0], stId, sRow[2], pCount, lCount].concat(sessionCells);
  });

  const prevLastCol = rosterSheet.getLastColumn();
  if (prevLastCol > headers.length) {
    rosterSheet.getRange(1, headers.length + 1, rosterSheet.getLastRow(), prevLastCol - headers.length).clearContent().clearFormat();
  }

  rosterSheet.getRange(1, 1, 1, headers.length).setValues([headers])
    .setBackground('#0f766e').setFontColor('#ffffff').setFontWeight('bold');
  rosterSheet.getRange(2, 1, outRows.length, headers.length).setValues(outRows);
  try {
    CacheService.getScriptCache().remove('ROSTER_MAP_V373_' + classId);
  } catch (e) {}
}

function getAllLatestSessionsMap(ss, bypassCache) {
  const cache = CacheService.getScriptCache();
  const fastKey = 'ALL_LATEST_SES_V373';
  const cacheKey = 'ALL_LATEST_SES_V35';
  const nowMs = Date.now();

  if (!bypassCache) {
    const fastCached = cache.get(fastKey);
    if (fastCached) {
      try {
        const parsed = JSON.parse(fastCached);
        if (parsed && parsed.map) {
          const out = {};
          Object.keys(parsed.map).forEach(cid => {
            const ses = parsed.map[cid];
            if (ses) {
              ses.isOpen = !ses.closedManual && nowMs <= ses.closesAtMs;
              out[cid] = ses;
            }
          });
          return out;
        }
      } catch (e) {}
    }
  }

  const sheet = ss.getSheetByName(TAB_SESSIONS);
  if (!sheet || sheet.getLastRow() < 2) return {};

  const lastRow = sheet.getLastRow();

  if (!bypassCache) {
    const cached = cache.get(cacheKey);
    if (cached) {
      try {
        const parsed = JSON.parse(cached);
        if (parsed && parsed._lastRow === lastRow && parsed.map) {
          const out = {};
          Object.keys(parsed.map).forEach(cid => {
            const ses = parsed.map[cid];
            if (ses) {
              ses.isOpen = !ses.closedManual && nowMs <= ses.closesAtMs;
              out[cid] = ses;
            }
          });
          return out;
        }
      } catch (e) {}
    }
  }

  const data = sheet.getRange(2, 1, lastRow - 1, 11).getValues();
  const map = {};

  for (let i = data.length - 1; i >= 0; i--) {
    const cid = String(data[i][1]).trim();
    if (!cid || map[cid]) continue;

    const openedAtMs     = parseSheetTimeMs(data[i][3]);
    const presentUntilMs = parseSheetTimeMs(data[i][4]);
    const closesAtMs     = parseSheetTimeMs(data[i][5]);
    const closedManual   = String(data[i][6]).trim();
    const isOpen         = !closedManual && nowMs <= closesAtMs;

    const sesObj = {
      _lastRow: lastRow,
      rowIndex: i + 2,
      sessionId: String(data[i][0]),
      classId: cid,
      sessionLabel: String(data[i][2]),
      openedAtMs: openedAtMs,
      presentUntilMs: presentUntilMs,
      closesAtMs: closesAtMs,
      closedManual: closedManual,
      isOpen: isOpen,
      roomLat: Number(data[i][7]),
      roomLng: Number(data[i][8]),
      radiusMeters: Number(data[i][9] || 100),
      qrMode: String(data[i][10] || 'STATIC')
    };

    map[cid] = sesObj;
  }

  try {
    const payloadStr = JSON.stringify({ _lastRow: lastRow, map: map });
    cache.put(fastKey, payloadStr, 120);
    cache.put(cacheKey, payloadStr, 120);
  } catch (e) {}

  return map;
}

function parseSheetTimeMs(val) {
  if (!val) return 0;
  if (typeof val.getTime === 'function') return val.getTime();
  if (typeof val === 'number') return val;
  const ms = new Date(val).getTime();
  return isNaN(ms) ? 0 : ms;
}

function formatAttendanceTimeStr_(val) {
  if (!val) return '';
  if (typeof val.getTime === 'function') {
    return Utilities.formatDate(val, TIMEZONE, 'HH:mm');
  }
  const s = String(val).trim();
  const mIso = s.match(/T(\d{2}:\d{2})/);
  if (mIso) return mIso[1];
  const mTime = s.match(/\b(\d{1,2}):(\d{2})(?::\d{2})?\b/);
  if (mTime) return ('0' + mTime[1]).slice(-2) + ':' + mTime[2];
  return s.slice(11, 16);
}

function updateStudentClassAttendanceStatsCacheInPlace_(classId, activeSession, studentId, status, isRound2CheckIn) {
  const sid = normalizeId(studentId);
  const fallback = {
    P: 0, L: 0, E: 0, A: 0, total: 0,
    equivAbsences: 0, attendanceScore: 10.0, manualExamBan: false, examStatus: 'OK', history: []
  };
  if (!classId || !sid) return fallback;
  try {
    const cache = CacheService.getScriptCache();
    const cacheKey = 'CLASS_ATT_STATS_V36_' + classId;
    const raw = cache.get(cacheKey);
    if (!raw) return fallback;
    const statsMap = JSON.parse(raw);
    if (!statsMap || !statsMap[sid]) return fallback;
    const st = statsMap[sid];
    const nowMs = Date.now();
    const openedMs = (activeSession && activeSession.openedAtMs) ? Number(activeSession.openedAtMs) : nowMs;
    const dt = new Date(openedMs);
    const dayKey = Utilities.formatDate(dt, TIMEZONE, 'yyyy-MM-dd');
    const dispDate = Utilities.formatDate(dt, TIMEZONE, 'dd/MM/yyyy');
    const cleanLabel = String((activeSession && activeSession.sessionLabel) || dayKey).replace(/_P2$/i, '');

    if (!Array.isArray(st.history)) st.history = [];
    let dayEntry = st.history.find(h => h && h.dayKey === dayKey);
    const prevFinal = dayEntry ? dayEntry.finalStatus : null;

    if (!dayEntry) {
      dayEntry = {
        dayKey: dayKey,
        date: dispDate,
        label: cleanLabel,
        sessionId: activeSession ? String(activeSession.sessionId || '') : '',
        p1Status: isRound2CheckIn ? '-' : status,
        p2Status: isRound2CheckIn ? 'P' : '',
        p2Opened: Boolean(isRound2CheckIn),
        finalStatus: isRound2CheckIn ? 'L' : status,
        status: isRound2CheckIn ? 'L' : status
      };
      st.history.push(dayEntry);
    } else {
      if (isRound2CheckIn) {
        dayEntry.p2Opened = true;
        dayEntry.p2Status = 'P';
      } else {
        dayEntry.p1Status = status;
      }
      const nextFinal = mergeAttendance(dayEntry.p1Status || '-', dayEntry.p2Status || '', Boolean(dayEntry.p2Opened));
      dayEntry.finalStatus = nextFinal === '-' ? 'OPEN' : nextFinal;
      dayEntry.status = dayEntry.finalStatus;
    }

    const newFinal = dayEntry.finalStatus;
    if (prevFinal !== newFinal) {
      if (prevFinal === 'P') { st.P = Math.max(0, st.P - 1); st.total = Math.max(0, st.total - 1); }
      else if (prevFinal === 'L') { st.L = Math.max(0, st.L - 1); st.total = Math.max(0, st.total - 1); }
      else if (prevFinal === 'E') { st.E = Math.max(0, st.E - 1); st.total = Math.max(0, st.total - 1); }
      else if (prevFinal === '-') { st.A = Math.max(0, st.A - 1); st.total = Math.max(0, st.total - 1); }

      if (newFinal === 'P') { st.P++; st.total++; }
      else if (newFinal === 'L') { st.L++; st.total++; }
      else if (newFinal === 'E') { st.E++; st.total++; }
      else if (newFinal === '-') { st.A++; st.total++; }

      const equivAbsences = Math.round((st.A + (st.E * 0.5)) * 10) / 10;
      const attendanceScore = Math.round((10.0 - (st.A * 1.0) - (st.L * 0.5) - (st.E * 0.5)) * 10) / 10;
      const isBanned = Boolean(st.manualExamBan) || (equivAbsences > 6.0) || (attendanceScore < 0.0);
      const isWarn = !isBanned && ((equivAbsences >= 5.5 && equivAbsences <= 6.0) || (attendanceScore >= 0.0 && attendanceScore <= 1.0));
      st.equivAbsences = equivAbsences;
      st.attendanceScore = attendanceScore;
      st.examStatus = isBanned ? 'BAN' : (isWarn ? 'WARN' : 'OK');
    }

    cache.put(cacheKey, JSON.stringify(statsMap), 1800);
    return st;
  } catch (e) {
    return fallback;
  }
}

/**
 * Computes and caches per-student attendance counters, official attendance score, exam ban status,
 * and merged per-day history `{ P, L, E, A, total, equivAbsences, attendanceScore, manualExamBan, examStatus, history }`
 * for an entire class. Cached in CacheService for 1800s under `CLASS_ATT_STATS_V36_<classId>`.
 */
function getClassAttendanceStatsMap_(ss, classId, bypassCache) {
  if (!classId) return {};
  const cache = CacheService.getScriptCache();
  const cacheKey = 'CLASS_ATT_STATS_V36_' + classId;

  if (!bypassCache) {
    const cached = cache.get(cacheKey);
    if (cached) {
      try { return JSON.parse(cached); } catch (e) {}
    }
  }

  const rosterMap = getClassRosterMap(ss, classId, false);
  const studentIds = Object.keys(rosterMap);

  // Detect manual red-highlighted students from synced target sheet properties + local Master Sheet row colors
  const manualBanMap = {};
  try {
    const rawPropBan = PropertiesService.getScriptProperties().getProperty('QR_MANUAL_EXAM_BAN_MAP');
    if (rawPropBan) {
      const parsedBan = JSON.parse(rawPropBan);
      const classBan = parsedBan && parsedBan[classId] ? parsedBan[classId] : null;
      if (classBan) {
        Object.keys(classBan).forEach(k => {
          if (classBan[k]) manualBanMap[normalizeId(k)] = true;
        });
      }
    }
  } catch (e) {}

  const sheetCellOverrides = {}; // { [sid]: { [colIdx0]: 'E' | 'L' | 'A' | 'P' } }
  try {
    const classSheet = ss.getSheetByName(classId);
    if (classSheet && classSheet.getLastRow() > 1) {
      const numRows = classSheet.getLastRow() - 1;
      const lastCol = Math.max(3, classSheet.getLastColumn());
      const headerRow = classSheet.getRange(1, 1, 1, lastCol).getValues()[0];
      const vals = classSheet.getRange(2, 1, numRows, lastCol).getValues();
      const bgs = classSheet.getRange(2, 1, numRows, lastCol).getBackgrounds();
      const fgs = classSheet.getRange(2, 1, numRows, 3).getFontColors();

      for (let r = 0; r < numRows; r++) {
        const sid = normalizeId(vals[r][1]);
        if (!sid) continue;
        const isRowRed =
          classifySheetCellColorLocal_(bgs[r][0]) === 'RED' ||
          classifySheetCellColorLocal_(bgs[r][1]) === 'RED' ||
          classifySheetCellColorLocal_(bgs[r][2]) === 'RED' ||
          classifySheetCellColorLocal_(fgs[r][0]) === 'RED' ||
          classifySheetCellColorLocal_(fgs[r][1]) === 'RED' ||
          classifySheetCellColorLocal_(fgs[r][2]) === 'RED';
        if (isRowRed) {
          manualBanMap[sid] = true;
        }

        // Inspect final session columns (col index 5+, skipping _P1 and _Trạng_thái_P2 helper columns)
        if (lastCol > 5) {
          for (let c = 5; c < lastCol; c++) {
            const hdr = String(headerRow[c] || '').trim();
            if (/_P1$|_Trạng_thái_P2$/i.test(hdr)) continue;
            const mS = hdr.match(/^S(\d+)_/i);
            const dayIdx = mS ? (parseInt(mS[1], 10) - 1) : (c - 5);
            const cellBgClass = classifySheetCellColorLocal_(bgs[r][c]);
            const cellText = String(vals[r][c] || '').trim().toUpperCase();
            if (cellBgClass === 'ORANGE' || cellText === 'E' || cellText === 'P_EXCUSED') {
              if (!sheetCellOverrides[sid]) sheetCellOverrides[sid] = {};
              sheetCellOverrides[sid][dayIdx] = 'E';
            } else if (cellBgClass === 'YELLOW' || cellText === 'L') {
              if (!sheetCellOverrides[sid]) sheetCellOverrides[sid] = {};
              sheetCellOverrides[sid][dayIdx] = 'L';
            } else if (cellText === 'X') {
              if (!sheetCellOverrides[sid]) sheetCellOverrides[sid] = {};
              sheetCellOverrides[sid][dayIdx] = 'A_MANUAL';
            }
          }
        }
      }
    }
  } catch (e) {}

  const statsMap = {};
  studentIds.forEach(sid => {
    statsMap[sid] = {
      P: 0,
      L: 0,
      E: 0,
      A: 0,
      total: 0,
      equivAbsences: 0,
      attendanceScore: 10.0,
      manualExamBan: Boolean(manualBanMap[sid]),
      examStatus: manualBanMap[sid] ? 'BAN' : 'OK',
      history: []
    };
  });

  const validDays = buildGroupedClassDayRecords_(ss, classId);

  validDays.forEach((grp, dayIdx) => {
    studentIds.forEach(sid => {
      let mark = grp.marks[sid] || '-';
      if (sheetCellOverrides[sid] && sheetCellOverrides[sid][dayIdx] && !grp.manualLocked[sid]) {
        mark = sheetCellOverrides[sid][dayIdx];
      }

      let histStatus = '-';
      if (mark === 'P') {
        statsMap[sid].P++;
        statsMap[sid].total++;
        histStatus = 'P';
      } else if (mark === 'L') {
        statsMap[sid].L++;
        statsMap[sid].total++;
        histStatus = 'L';
      } else if (mark === 'E') {
        statsMap[sid].E++;
        statsMap[sid].total++;
        histStatus = 'E';
      } else if (mark === 'A_MANUAL' || !grp.hasOpenSession) {
        statsMap[sid].A++;
        statsMap[sid].total++;
        histStatus = '-';
      } else {
        histStatus = 'OPEN';
      }

      statsMap[sid].history.push({
        dayKey: grp.dayKey,
        date: grp.displayDate,
        label: grp.label,
        sessionId: grp.latestSessionId,
        p1Status: (grp.p1Marks && grp.p1Marks[sid]) ? grp.p1Marks[sid] : '-',
        p2Status: grp.p2Opened ? ((grp.p2Marks && grp.p2Marks[sid]) ? grp.p2Marks[sid] : '-') : '',
        p2Opened: Boolean(grp.p2Opened),
        finalStatus: histStatus,
        status: histStatus
      });
    });
  });

  studentIds.forEach(sid => {
    const st = statsMap[sid];
    const equivAbsences = Math.round((st.A + (st.E * 0.5)) * 10) / 10;
    const attendanceScore = Math.round((10.0 - (st.A * 1.0) - (st.L * 0.5) - (st.E * 0.5)) * 10) / 10;
    const isBanned = Boolean(st.manualExamBan) || (equivAbsences > 6.0) || (attendanceScore < 0.0);
    const isWarn = !isBanned && ((equivAbsences >= 5.5 && equivAbsences <= 6.0) || (attendanceScore >= 0.0 && attendanceScore <= 1.0));

    st.equivAbsences = equivAbsences;
    st.attendanceScore = attendanceScore;
    st.examStatus = isBanned ? 'BAN' : (isWarn ? 'WARN' : 'OK');
  });

  try { cache.put(cacheKey, JSON.stringify(statsMap), 1800); } catch (e) {}
  return statsMap;
}

function getStudentClassAttendanceStats(ss, classId, studentId, bypassCache) {
  const sid = normalizeId(studentId);
  const fallback = {
    P: 0, L: 0, E: 0, A: 0, total: 0,

  };

    }
  });
  return out;
}

function calculateHaversineMeters(lat1, lon1, lat2, lon2) {
  const R = 6371000;
  const toRad = deg => (deg * Math.PI) / 180;
  const dLat = toRad(lat2 - lat1);
  const dLon = toRad(lon2 - lon1);
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos(toRad(lat1)) * Math.cos(toRad(lat2)) *
    Math.sin(dLon / 2) * Math.sin(dLon / 2);
  return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

    code = generateNumericCode(6);
    exp  = Date.now() + 2 * 60 * 60 * 1000;
    props.setProperty('TEMP_PC_CODE', code);
    props.setProperty('TEMP_PC_EXPIRES', String(exp));
    props.setProperty('FAILED_AUTH_COUNT', '0');
    props.deleteProperty('AUTH_LOCKOUT_UNTIL');
    cache.removeAll(['AUTH_LOCKOUT_UNTIL', 'DASH_C9_CODE_V35']);
    const doc = ss || SpreadsheetApp.getActiveSpreadsheet();
    const sheet = doc ? doc.getSheetByName(TAB_DASHBOARD) : null;
    if (sheet) {
      sheet.getRange('C9').setValue('Temp PC Code:\n' + code + '\n(Valid 2 Hours)').setFontSize(14).setFontWeight('bold');
    }
  }
  return code;
}

function generateRandomToken(len) {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz23456789';
  let out = '';
  for (let i = 0; i < len; i++) {
    out += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return out;
}

function generateNumericCode(len) {
  let out = '';
  for (let i = 0; i < len; i++) {
    out += Math.floor(Math.random() * 10);
  }
  return out;
}

function sha256Hex(input) {
  const raw = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, input, Utilities.Charset.UTF_8);
  return raw.map(b => ('0' + (b & 0xff).toString(16)).slice(-2)).join('');
}

function getTodayCalendarEventsCached() {
  try {
    const cache = CacheService.getScriptCache();
    const todayKey = 'CAL_EVENTS_' + Utilities.formatDate(new Date(), TIMEZONE, 'yyyyMMdd');
    const cached = cache.get(todayKey);
    if (cached) {
      return JSON.parse(cached);
    }

    const cal = CalendarApp.getDefaultCalendar();
    if (!cal) return [];
    const events = cal.getEventsForDay(new Date());
    const out = events.map(ev => ({
      title: String(ev.getTitle() || '').trim(),
      startTime: Utilities.formatDate(ev.getStartTime(), TIMEZONE, 'HH:mm'),
      endTime: Utilities.formatDate(ev.getEndTime(), TIMEZONE, 'HH:mm')
    }));
    cache.put(todayKey, JSON.stringify(out), 600);
    return out;
  } catch (e) {
    return [];
  }
}

/**
 * Parses period blocks from Class_Name or metadata text:
 *  - 123 / 1-3: 06:30 - 09:00
 *  - 456 / 4-6: 09:00 - 11:30
 *  - 789 / 7-9: 12:30 - 15:00
 *  - 101112 / 10-12: 15:00 - 17:30
 *  - 131415 / 13-15: 17:30 - 20:00
 */
function parseClassScheduleInfoBackend(text) {
  const s = String(text || '').trim();
  if (!s) return null;

  let startPeriod = null;
  let endPeriod = null;

  const compactMatch = s.match(/(?:ti[eế]t|t|ca)?\s*\b(123|456|789|101112|131415|161718)\b/i);
  if (compactMatch) {
    const map = {
      '123': [1, 3],
      '456': [4, 6],
      '789': [7, 9],
      '101112': [10, 12],
      '131415': [13, 15],
      '161718': [16, 18]
    };
    const pair = map[compactMatch[1]];
    if (pair) {
      startPeriod = pair[0];
      endPeriod = pair[1];
    }
  } else {
    const rangeMatch = s.match(/(?:ti[eế]t|t|periods?)?\s*\b(\d{1,2})\s*[-–→to]\s*(\d{1,2})\b/i);
    if (rangeMatch) {
      const sp = parseInt(rangeMatch[1], 10);
      const ep = parseInt(rangeMatch[2], 10);
      if (sp >= 1 && ep >= sp && ep <= 18 && (ep - sp) <= 6) {
        startPeriod = sp;
        endPeriod = ep;
      }
    }
  }

  const daysOfWeek = [];
  const addDay = d => {
    if (typeof d === 'number' && d >= 0 && d <= 6 && daysOfWeek.indexOf(d) === -1) {
      daysOfWeek.push(d);
    }
  };
  const thuRegex = /(?:^|[\s,;|(\[/])(?:th[uứ]\s*|t)([2-7](?:\s*(?:[&,+]\s*(?:th[uứ]\s*|t)?|[-–]\s*(?:th[uứ]\s*|t))[2-7])*)\b(?!\s*[-–→to]\s*\d)/ig;
  let m;
  while ((m = thuRegex.exec(s)) !== null) {
    const digits = m[1].match(/[2-7]/g) || [];
    digits.forEach(d => addDay(parseInt(d, 10) - 1));
  }
  if (/\b(ch[uủ]\s*nh[aậ]t|cn|sunday|sun)\b/i.test(s)) addDay(0);
  if (/\b(monday|mon)\b/i.test(s)) addDay(1);
  if (/\b(tuesday|tue)\b/i.test(s)) addDay(2);
  if (/\b(wednesday|wed)\b/i.test(s)) addDay(3);
  if (/\b(thursday|thu)\b/i.test(s)) addDay(4);
  if (/\b(friday|fri)\b/i.test(s)) addDay(5);
  if (/\b(saturday|sat)\b/i.test(s)) addDay(6);

  const dayOfWeek = daysOfWeek.length > 0 ? daysOfWeek[0] : null;

  if (startPeriod === null && dayOfWeek === null) return null;
  return {
    startPeriod: startPeriod,
    endPeriod: endPeriod,
    dayOfWeek: dayOfWeek,
    daysOfWeek: daysOfWeek,
    rawText: s
  };
}
